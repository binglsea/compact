	\section{Generalization of Submetacompactness}  
	\label{sec:3-2}


{\definition{} \label{def:3-2-1}
	\textup{(Smith [1975]\textsuperscript{\cite{Smith1975}})} 
An open cover $\mathscr{U}$ of $X$ is a 
\uwave{weak $\overline{\theta}$-cover}
\index{weak $\overline{\theta}$-cover| \S\ref{sec:3-2}\hfill}
if $\mathscr{U}$ has an open refinement $\mathscr{G} = \bigcup_{n \in \omega} \mathscr{G}_n$ where each $\mathscr{G}_n = \{G_{n, \alpha} : \alpha \in \Gamma_n\}$ such that

\textup{(i)}
For each $x \in X$, there exists $m \in \omega$ such that 
$0 < |\operatorname{ind}(\mathscr{G}_m)_x| < \omega$.

\textup{(ii)}
$\mathscr{G}^* = \{\bigcup \mathscr{G}_n : n \in \omega\}$ is a point-finite open cover of $X$.

\noindent The family $\mathscr{G}$ is called a 
\uwave{weak $\overline{\theta}$-refinement}
\index{weak $\overline{\theta}$-refinement| \S\ref{sec:3-2}\hfil}
of $\mathscr{U}$.  $X$ is 
\uwave{$\lambda$-weakly $\overline{\theta}$-refinable}
\index{$\lambda$-weakly $\overline{\theta}$-refinable| \S\ref{sec:3-2}\hfil}
if every open $\lambda$-cover is a weak $\overline{\theta}$-cover.
}


{\definition{} \label{def:3-2-2}
\textup{(i)}
An open cover $\mathscr{U}$ of $X$ is a $D\omega$-cover if $\mathscr{U}$ 
has a refinement $\mathscr{F} = \bigcup_{n \in \omega} \mathscr{F}_n$ where each 
$\mathscr{F}_n$ is a family of relatively discrete and closed in the subspace 
$X \setminus\bigcup_{i< n}(\cup \mathscr{F}_i)$, and 
$(\cup\mathscr{F}_m)\bigcap(\cup\mathscr{F}_n) = \emptyset$ for any $m\ne n$.
The family $\mathscr{F}$ is called a 
\uwave{$D\omega$-refinement}
\index{$D\omega$-refinement| \S\ref{sec:3-2}\hfill}
of $\mathscr{U}.$

\textup{(ii)}
\textup{(Liu [1977]\textsuperscript{\cite{Liu}})} 
$X$ is 
\uwave{$\lambda$-strictly quasi-paracompact}
\index{$\lambda$-strictly quasi-paracompact| \S\ref{sec:3-2}\hfill}
or 
\uwave{$D\omega$-refinable}
\index{$D\omega$-refinable| \S\ref{sec:3-2}\hfill}
if every open $\lambda$-cover of $X$ is a $D\omega$-cover.
}

{\definition{}  \label{def:3-2-3}
\textup{(Smith [1980]\textsuperscript{\cite{Smith1980}})} 
Let $\omega^2 = \omega\cdot\omega$ be an ordinal product, $\xi\in\{\omega, \omega^2\}$. 
The properties $\mathcal{P}$ will be discrete $(\mathcal{D})$, locally finite $(\mathcal{LC})$
or closed $(\mathcal{C})$.
	
\textup{(i)} An open cover $\mathscr{U}$ of $X$ is a 
\uwave{$B(\mathcal{P}, \xi)$-cover}
\index{$B(\mathcal{P}, \xi)$-cover| \S\ref{sec:3-2}\hfill}
if $\mathscr{U}$ has a refinement 
$\mathscr{E} = \bigcup_{\beta < \xi} \mathscr{E}_\beta$, which satisfies
	
\textup{(a)} $(\bigcup\mathscr{E}_\beta) \bigcap (\bigcup\mathscr{E}_{\beta'}) = \emptyset$ 
if $\beta \ne \beta'$;

\textup{(b)} for each $\beta < \xi$, $\mathscr{E}_\beta$ is a relatively $\mathcal{P}$ family 
of closed subsets of the subspace 
$X\setminus \bigcup_{\mu<\beta}(\bigcup\mathscr{E}_\mu)$;

\textup{(c)} for each $\beta < \xi$, $\bigcup_{\mu<\beta}(\bigcup\mathscr{E}_\mu)$ 
is closed in $X$.

\noindent For the case $\mathcal{P} = \mathcal{C}$, we require $\mathscr{E}_\beta$ to be a 
one-to-one partial refinement of $\mathscr{U}$.
The family $\mathscr{E}$ is called a 
\uwave{$B(\mathcal{P}, \xi)$-refinement}
\index{$B(\mathcal{P}, \xi)$-refinement| \S\ref{sec:3-2}\hfill}
of $\mathscr{U}$.

\textup{(ii)} $X$ is 
\uwave{$\lambda$-$B(\mathcal{P}, \xi)$-refinable}
\index{$\lambda$-$B(\mathcal{P}, \xi)$-refinable| \S\ref{sec:3-2}\hfill}
if every open $\lambda$-cover of $X$ is a $B(\mathcal{P}, \xi)$-cover.
}

\vspace{8pt}
It is easy to see that $X$ is $\lambda$-strictly quasi-paracompact iff $X$ is $\lambda$-$B(\mathcal{P}, \xi)$-refinable.

{\lemma{} \label{lem:3-2-4} 
\textup{(Liu [1977]\textsuperscript{\cite{Liu}})} 
A point-finite open cover of $X$ is a $D\omega$-cover.
}

\begin{proof}
Suppose $\mathscr{U} = \{ U_\alpha : \alpha \in \Delta \}$ is a point-finite open 
cover of $X$. For $n < \omega$, let $D_0= \emptyset$ and
$$D_n = \{ x \in X: |\operatorname{ind}(\mathscr{U})_x| \le n \},  \textup{ when } n \ge 1.$$
\noindent Then $D_n$ is closed. Let $F(\emptyset, 0) = \emptyset$. 
For $n \ge 1$ and $S \in [\Delta]^n$, let
$$F(S, n) = D_n\bigcap(\cap_{\alpha \in S} U_\alpha), \quad
\mathscr{F}_n= \{F(S, n) : S \in [\Delta]^n\}$$

\noindent We show that $\mathscr{F} = \bigcup_{n<\omega}\mathscr{F}_n$ is a $D\omega$-refinement of $\mathscr{U}$. Let $n < \omega$, 
$Y = X \setminus \bigcup_{i<n}(\cup\mathscr{F}_i)$. To show that $\mathscr{F}_n\vert{}_Y$ is a discrete closed family in $Y$. 
By Lemma \ref{lem:1-1-4}, we need only to show that

\textup{(1)} 
$\bigcup\mathscr{F}_n\vert{}_Y$ is closed in $Y$, and

\textup{(2)}  $\mathscr{F}_n\vert{}_Y$ is discrete at each 
$x \in \bigcup\mathscr{F}_n\vert{}_Y$.

To see (1), let $x\in G = Y \setminus \bigcup(\mathscr{F}_n\vert{}_Y)$. 
Suppose $|\operatorname{ind}(\mathscr{U})_x| = j$. Then $x \in D_j$. 
Pick $S \in [A]^j$ such that 
$x \in D_j\bigcap (\cap_{\alpha \in S} U_\alpha) = F(S, j) \subset \bigcup \mathscr{F}_j$.
Since $x \in G$, $j > n$. Then $Y \setminus D_n$ is a neighborhood of $x$ in $Y$ 
such that $Y \setminus D_n \subset G$. Hence $G$ is open in $Y$.

To see (2), let $x \in \bigcup(\mathscr{F}_n\vert{}_Y)$. Suppose $S_0 \in [\Delta]^n$ 
such that $x \in F(S_0, n)$. Then $W = Y \bigcap (\cap_{\alpha \in s_0} U_\alpha)$ is a neighborhood of $x$ in $Y$. 
For each $S \in [\Delta]^n \setminus \{S_0\}$, $W \cap F(S, n) = \emptyset$. 
So, $\mathscr{F}_n\vert{}_Y$ is discrete at $x$.
\end{proof}

{\theorem{}  \label{th:3-2-5}
\textup{(Zhu [1984]\textsuperscript{\cite{Zhu1984}},
Long [1986]\textsuperscript{\cite{Long}})} 
A $\lambda$-submetacompact space is $\lambda$-strictly quasi-paracompact.
}

\begin{proof}
Assume $X$ is $\lambda$-submetacompact and let 
$\mathscr{U} = \{ U_\alpha : \alpha < \lambda \}$ be an open cover of $X$. 
By Lemma \ref{lem:3-1-4}, 
there exists a countable closed cover $\{ F_n \}$ of $X$ such that each $\mathscr{U}\vert{}_{F_n}$ has a point-finite open refinement 
$\mathscr{U}_n$ in $F_n$. By Lemma 3.2.4, let 
$\mathscr{E}_n = \bigcup_{k<\omega} \mathscr{E}_{n,k}$ be a 
$D\omega$-refinement of $\mathscr{U}_n$ in $F_n$. For each $n, k$, 
$\mathscr{E}_{n,k}$ is a discrete closed family in 
$F_n \setminus\bigcup_{i<k}(\cup \mathscr{E}_{n,i})$. Then each 
$\mathscr{E}_{n,k}$ is a discrete closed family in 
$X \setminus \bigcup_{i<k}(\cup \mathscr{E}_{n,i})$.

Let $\varphi$ be a one-to-one function from $\omega \times \omega$ 
onto $\omega$ defined by $\varphi(n, k) = \frac{1}{2}[(n+k)^2 + 3n + k]$. 
Define $\mathscr{F}_m = \mathscr{E}_{n,k}$ iff $m = \varphi(n, k)$. 
Then $\mathscr{F} = \bigcup_{m<\omega} \mathscr{F}_m$ is a 
$D\omega$-refinement of $\mathscr{U}$.
\end{proof}


{\theorem{}  \label{th:3-2-6}
\textup{(Smith [1980]\textsuperscript{\cite{Smith1980}})} 
Every $\lambda$-$D\omega$-refinable space is 
$\lambda$-weakly $\bar{\theta}$-refinable.
}

\begin{proof}
Suppose $\mathscr{U} = \{U_\alpha : \alpha < \lambda\}$ be an open cover of a $\lambda$-$D\omega$-refinable space $X$. 
Let $\mathscr{F} = \bigcup_{n<\omega}\mathscr{F}_n$ be a $D\omega$-refinement of $\mathscr{U}$, where each $\mathscr{F}_n = \{F_{n,\alpha}: \alpha < \lambda\}$ is a discrete family of closed subsets of $X \setminus \bigcup_{i<n}(\cup \mathscr{F}_i)$
with $F_{n, \alpha}\subset U_\alpha$. Define
$$G_{n,\alpha}=\left(U_\alpha\setminus\bigcup_{\gamma\ne\alpha}F_{n,\gamma}\right)
\cap\left(X\setminus\bigcup_{i<n}(\cup \mathscr{F}_i)\right)
$$
\noindent for $n <\lambda$, and let
$\mathscr G_n=\{G_{n,\alpha}:\alpha<\lambda\}$.
Then $F_{n,\alpha}\subset G_{n,\alpha}\subset U_\alpha$.

(1) $\mathscr G=\bigcup_{n<\omega}\mathscr G_n$ is an open refinement 
of $\mathscr{U}$.

\noindent\textsc{Proof of (1)} Let $n<\omega,\ \alpha<\lambda$, and let
$C=\bigcup_{i<n}(\mathscr{F}_i)$.
Since \(\mathscr F_n\setminus\{F_{n,\alpha}\}\) is a discrete closed family in $X\setminus C$,
$\bigcup_{\gamma\ne\alpha}F_{n,\gamma}=K\cap(X\setminus C)$
for some closed set $K$ in $X$. Thus

$$
G_{n,\alpha}
=
U_\alpha-\bigcup_{\beta\ne\alpha}F_{n,\beta}\cup C
=
U_\alpha-K\cup C
$$

is an open set in \(X\).

For each \(x\in X\), there is \(m<\omega\) such that

$$
x\in\bigcup\mathcal G_m.
$$

Then

$$
x\in\bigcup G_m\subset\bigcup G.
$$

**(2)** \(\forall x\in X\ \exists m<\omega\quad
0<|\operatorname{ind}(\mathcal G_m)_x|<\omega.\)
\end{proof}

**Proof.** Let \(x\in X\). There are \(m<\omega\) and \(\beta<\lambda\) such that

$$
x\in F_{m,\beta}.
$$

Then

$$
x\in G_{m,\beta}
$$

and

$$
\operatorname{ind}(\mathcal G_m)_x\subset\{\beta\}.
$$

(2) holds.

**(3)**

$$
\left\{\bigcup\mathcal G_n:n<\omega\right\}
$$

is point-finite.

**Proof.** If \(x\in X\), then

$$
x\in\bigcup G_m
$$

for some \(m<\omega\). For \(k>m\),

$$
x\notin\bigcup\mathcal G_k.
$$

Hence \(x\) belongs to only finitely many of the unions. (3) holds.

It follows from (1)–(3) that \(\mathcal G\) is a weakly \(\bar{\sigma}\)-refinement of \(U\).
