{\lemma{} \label{le:3_1_1} 
	If every directed $($increasing$)$ open $\lambda$-cover of $X$ has a locally finite 
	open refinement, then every directed $($increasing$)$ open $\lambda$-cover has a 
	locally open shrink.
}
\begin{proof}
Suppose $\mathscr{U} = \{U_\alpha : \alpha < \lambda\}$ is a directed (increasing) open cover of $X$. By hypothesis, $\mathscr{U}$ has a locally finite open refinement $\mathscr{G} = \{G_\alpha : \alpha < \lambda\}$ with $G_\alpha \subset U_\alpha$. 
For $\alpha < \lambda$, let

$$W_\alpha = X \setminus \bigcup \{ \overline{G_\gamma} : \gamma < \lambda,  
G_\gamma \not\subset U_\alpha\}.$$
Then $\mathscr{W} = \{W_\alpha : \alpha < \lambda\}$ is an open cover of $X$ 
with $\overline{W_\alpha} \subset U_\alpha$. 
Since $\mathscr{U}$ is directed (increasing), $\mathscr{W}$ is also 
directed (increasing). Let $\mathscr{V} = \{V_\alpha : \alpha < \lambda\}$ be  a locally finite open refinement of $\mathscr{W}$ such that $V_\alpha \subset W_\alpha$. Then $\mathscr{V}$ is a locally finite open shrink of $\mathscr{U}$. 
\end{proof} 

{\theorem{} \label{lth:3_1_2} 
	\textup{(Mack [1967]\textsuperscript{\cite{Mack}})} For any $X$, the following are equivalent:

\textup{(i)} $X$ is $\lambda$-paracompact.

\textup{(ii)} Every directed open $\lambda$-cover of $X$ has a  locally finite open refinement.

\textup{(iii)} Every increasing open $\lambda$-cover of $X$ has a  locally finite open refinement.

\textup{(iv)}  Every increasing open $\lambda$-cover has a locally finite open shrink.

\textup{(v)}  Every increasing open $\lambda$-cover $\mathscr{U}$
has a $\sigma$-locally finite open refinement $\mathscr{V}$ such that 
$\overline{\mathscr{V}}$ refines $\mathscr{U}$.
}

\begin{proof}
It is clear that (i) $\Rightarrow$ (ii) $\Rightarrow$ (iii) and (iv) $\Rightarrow$ (v).

(iii) $\Rightarrow$ (iv). Lemma \ref{le:3_1_1}.

(v) $\Rightarrow$ (i). From Theorem \ref{th:8_2_18}8.2.18. % TO_DO 8.2.18
\end{proof} 

{\theorem{}  \label{th:3_1_3} 
For any $X$ the following are equivalent:

\textup{(i)} $X$ is $\lambda$-paracompact.

\textup{(ii)} Every directed open $\lambda$-cover has a locally finite open shrink.

\textup{(iii)} Every directed open $\lambda$-cover has a locally finite closed refinement.
}

\begin{proof}
(i) $\Rightarrow$ (ii). Lemma \ref{le:3_1_1}.

(ii) $\Rightarrow$ (iii). Clear.

(iii) $\Rightarrow$ (i). Assume (iii) and let $\mathscr{U} = \{U_\alpha : \alpha < \lambda\}$ 
be a directed open cover of $X$. Let 
$\mathscr{F} = \{F_\alpha : \alpha < \lambda\}$ be a locally finite closed refinement of 
$\mathscr{U}$ with $F_\alpha \subset U_\alpha$. 
For $S \in [\lambda]^{<\omega}$, let $V_S = X \setminus \bigcup_{\gamma \notin S} F_\gamma$. 
Then $\mathscr{V} = \{V_S : S \in [\lambda]^{<\omega}\}$ is a directed open 
$\lambda$-cover of $X$. Let $\mathscr{C} = \{C_S : S \in [\lambda]^{<\omega}\}$ 
be a locally finite closed refinement of $\mathscr{V}$ with $C_S \subset V_S$ 
for each $S \in [\lambda]^{<\omega}$. For $\alpha < \lambda$, let

$$G_\alpha = X \setminus \bigcup \{C_S : S \in [\lambda]^{<\omega}, 
C_S \cap F_\alpha = \emptyset\}.$$

Then $F_\alpha \subset G_\alpha$ for each $\alpha < \lambda$. 
It is easy to see that $\mathscr{W} = \{G_\alpha \cap U_\alpha : \alpha < \lambda\}$ 
is a locally finite open refinement of $\mathscr{U}$, so $X$ is 
$\lambda$-paracompact.
\end{proof}

{\lemma{} \label{le:3_1_4}
An open cover $\mathscr{U}$ of $X$ has a $\theta$-sequence of open refinements iff 
there exists a countable closed cover $F_n$ of $X$ such that 
for each $n \in \omega$, $\mathscr{U} \vert{} F_n$ has a point-finite open refinement 
in the subspace $F_n$.
}

\begin{proof}
($\Rightarrow$). Suppose $\mathscr{U}$ has a $\theta$-sequence 
$\{\mathscr{V}_n = \{V_{n,\alpha} : \alpha \in \Delta_n\}\}_{n \in \omega}$ of open refinements. For $m, n < \omega$, let 
\begin{gather*}
	F_{m, 0} = \emptyset \\
	F_{m,n} = \{x \in X: \vert{}\operatorname{ind}(\mathscr{V}_m)_x\vert{} \le n\}, n\ge 1.
\end{gather*}
Then $\{F_{m,n}\}$ is a closed cover of $X$ such that for each $m, n < \omega$, 
$\mathscr{U} \vert{}_{F_{m,n}}$ has a point-finite open refinement 
$\mathscr{V}_m \vert{} F_{m,n}$ in subspace $F_{m,n}$.

($\Leftarrow$). Suppose $\mathscr{U} = \{U_\alpha : \alpha \in \Delta\}$ is an 
open cover of $X$ and $\{F_n : n \in \omega\}$ is a countable closed cover of $X$ 
such that for each $n \in \omega$, $\mathscr{U} \vert{}_{F_n}$ has a point-finite 
open refinement $\mathscr{V}_n = \{V_{n,\alpha} : \alpha \in \Delta\}$ in the subspace 
$F_n$ with $V_{n,\alpha} \subset U_\alpha\cap F_n$. 
Let $V_{n,\alpha} = V_{n,\alpha}' \cap F_n$ for some open subset $V_{n,\alpha}'$ of $X$. 
For $n \in \omega$, let
$$\mathscr{G}_n = \{V_{n,\alpha}'\cap U_\alpha: \alpha \in \Delta\} \cup 
\{(V_{m,\alpha}'\setminus F_n)\cap U_\alpha: m\ne n, \alpha\in\Delta\}.$$
Then $\{\mathscr{G}_n : n \in \omega\}$ is a $\theta$-sequence of open refinements 
of $\mathscr{U}$.
\end{proof}

{\theorem{} \label{th:3_1_5}
$X$ is $\lambda$-submetacompact iff for every open $\lambda$-cover
 $\mathscr{U}$ of $X$ there exists a countable closed cover $\{F_n\}$ of $X$ 
 such that for each $n \in \omega$, $\mathscr{U} \vert{}_{ F_n}$ has a point-finite 
 open refinement in the subspace $F_n$.\hfill\qedsymbol
}

{\definition{} \label{def:3_1_6}
Let $\mathcal{U} = \{ U_\alpha : \alpha \in A \}$ and $\mathcal{V}$ be covers of $X$.

\textup{(i)} $\mathcal{V}$ is a **point-wise (local) $W$-refinement** of $\mathcal{U}$ at $x \in X$ if there is $s \in [A]^\omega$ (and $\exists \, e \in \text{top}(x)$) such that $(\mathcal{V})_x$ (and $\mathcal{V}$ refines $\{U_\alpha : \alpha \in s\}$, or is a point-wise (local) $W$-refinement of $\mathcal{U}$ as if $\mathcal{V}_x$ is a point-wise (local) $W$-refinement of $\mathcal{U}$ at each $x \in X$.

\textup{(ii)} A sequence $\{\mathcal{U}_n\}$ of covers of $X$ is a **point-wise $W$-refining sequence** for $\mathcal{U}$ if $\forall x \in X \; \exists \, m < \omega$ such that $\mathcal{U}_m$ is a point-wise $W$-refinement of $\mathcal{U}$ at $x$.
}

 3.1.10. Theorem. (Sconyers 1970)

$X$ is $\mathfrak{m}$-metacompact iff every increasing open $\mathfrak{m}$-cover of $X$ has a point-finite open refinement.

**Proof.** The necessity is clear. We prove the sufficiency.

Suppose every increasing open $\mathfrak{m}$-cover of $X$ has a point-finite open refinement. For each $\kappa \le \mathfrak{m}$ denote by $P_\kappa$ the proposition "every open $\kappa$-cover of $X$ has a point-finite open refinement." We use induction to show that $P_\kappa$ holds for each $\kappa \le \mathfrak{m}$.

If $\kappa$ is finite, $P_\kappa$ clearly holds.

Let $\kappa \le \mathfrak{m}$ such that $P_\mu$ holds for each $\mu < \kappa$. To show $P_\kappa$ holds, let $\mathcal{U} = \{U_\alpha : \alpha < \kappa\}$ be an open cover of $X$.

$\mathcal{U}^* = \{ \bigcup_{\gamma < \alpha} U_\gamma : \alpha < \kappa \}$ is an increasing open $\kappa$-cover of $X$.

Let $\mathcal{W} = \{W_\alpha : \alpha < \kappa\}$ be a point-finite open refinement of $\mathcal{U}^*$ with $W_\alpha \subset \bigcup_{\gamma < \alpha} U_\gamma$. For $\alpha < \kappa$, let

$$\mathcal{G}_\alpha = \{ U_\gamma : \gamma < \alpha \} \cup \{ \bigcup_{\gamma > \alpha} W_\gamma \}$$

Then $\mathcal{G}_\alpha$ is an open cover of $W_\alpha$ with power $< \kappa$. By induction assumption, $\mathcal{G}_\alpha$ has a point-finite open refinement $\mathcal{H}_\alpha$. Let $\mathcal{H}_\alpha = \{ H_\gamma : \gamma < \alpha \} \cup \{ R_\alpha \}$ such that $H_\gamma \subset U_\gamma$ for $\gamma < \alpha$ and $R_\alpha \subset \bigcup_{\gamma > \alpha} W_\gamma$.

Let

$$\mathcal{V} = \{ H_\gamma : \gamma < \alpha, \alpha < \kappa \}$$

Then $\mathcal{V}$ is a point-finite open refinement of $\mathcal{U}$. So $P_\kappa$ holds.

This completes the induction. Thus $P_\kappa$ holds, so $X$ is $\mathfrak{m}$-metacompact. $\square$
