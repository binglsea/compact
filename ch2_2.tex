	\section{Paracompactness with Separation Axioms}  
	\label{sec:ch2_para_t}


 {\theorem{} \textup{(Michael [1953, 1957]\textsuperscript{\cite{Michael1953}\cite{Michael1957}})}
For any $T_1$ regular space $X$, the following are equivalent:

\textup{(i)} $X$ is $\kappa$-paracompact.

\textup{(ii)}  Every open cover of $X$ has a $\sigma$-discrete open refinement.

\textup{(iii)} . Every open cover has a LF open refinement.

\textup{(iv)}  Every open cover has a $\sigma$-cp open refinement.
}
