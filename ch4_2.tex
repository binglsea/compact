\section{Paracompact and Metacompact Spaces} \label{sec:4-2}

{\theorem{} \label{th:4.2.1}
\textup{([Boone 1973]\textsuperscript{\cite{Boone}})}
$X$ is $\theta$-metacompact iff $X$ is $\sigma$-submetacompact and $\sigma$-discretely almost expandable.
}

\begin{proof}
See Corollary 8.1.6\ref{cor:2.1.6}.
\end{proof}

{\theorem{} \label{th:4.2.2}
\textup{([Junnila 1978, 1979a]\textsuperscript{\cite{Junnila1978}\cite{Junnila1979a}})}
For any $X$, the following are equivalent:

\textup{(i)} $X$ is metacompact.

\textup{(ii)} Every directed open $\lambda$-cover of $X$ has an interior-preserving 
open  point-star refinement.

\textup{(iii)} Every interior-preserving directed open $\lambda$-cover of $X$ has an interior-preserving
open  point-star refinement.

\textup{(iv)} Every  directed open $\lambda$-cover of $X$ has an closure-preserving refinement.

\textup{(v)}  Every interior-preserving  directed open $\lambda$-cover of $X$ has a closure-preserving
closed refinement.

\textup{(vi)} Every open $\lambda$-cover of $X$ has a semi-open point-star $\dot{F}$-refinement.
}

\begin{proof} 
	Assume (i) and let
$\mathcal{U}$ be a directed open
$\alpha$-cover of $X$. Let $\mathcal{V}$ be a point-finite open refinement of
$\mathcal{U}$. Then $\mathcal{V}$ is
an
IP open point-star refinement of $X$ ...
(iv) and (i) are clear.
(ii) $\iff$ (iv) and ... By Lemma 4.1.6.
...
Assume ... By Theorem 4.1.7,
$X$ is $\sigma$-submetacompact. By Theorem 4.2.1,
we need only to show that $X$ is $\sigma$-discretely almost
expandable. To this end, let $\mathcal{F}$ be
a discrete
family
of closed subsets of $X$ and let $\mathcal{S}$ be the induced cover.
Then $\mathcal{S}^{*\mathcal{F}}$ is
an IP directed
open cover of $X$. By ...
$\mathcal{S}^{*\mathcal{F}}$ has a cushioned refinement; or by axiom $\mathcal{S}^{*\mathcal{F}}$ has a
Semi-open point-star refinement $\mathcal{L}$. Then $\mathcal{L}$ is also
a point-star refinement of $\mathcal{S}^{*\mathcal{F}}$. By Lemma 4.1.9, $\mathcal{F}$
has
a cushioned refinement. By Lemma 1.2.4, $\mathcal{F}$ has
point-finite open expansion. Hence $X$ is discretely
almost expandable.
\end{proof}
