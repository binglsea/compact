\section{Paracompact and Metacompact Spaces} \label{sec:4-2}

{\theorem{} \label{th:4.2.1}
\textup{([Boone 1973]\textsuperscript{\cite{Boone}})}
$X$ is $\lambda$-metacompact iff $X$ is $\lambda$-submetacompact and $\lambda$-discretely almost expandable.
}

\begin{proof}
See Corollary 8.1.6\ref{cor:2.1.6}.
\end{proof}

{\theorem{} \label{th:4.2.2}
\textup{([Junnila 1978, 1979a]\textsuperscript{\cite{Junnila1978}\cite{Junnila1979a}})}
For any $X$, the following are equivalent:

\textup{(i)} $X$ is $\lambda$-metacompact.

\textup{(ii)} Every directed open $\lambda$-cover of $X$ has an interior-preserving 
open  point-star refinement.

\textup{(iii)} Every interior-preserving directed open $\lambda$-cover of $X$ has an interior-preserving
open  point-star refinement.

\textup{(iv)} Every  directed open $\lambda$-cover of $X$ has an closure-preserving closed refinement.

\textup{(v)}  Every interior-preserving  directed open $\lambda$-cover of $X$ has a closure-preserving
closed refinement.

\textup{(vi)} Every open $\lambda$-cover of $X$ has a semi-open point-star $\dot{F}$-refinement.
}

\begin{proof} 
(i) $\Rightarrow$ (ii). Assume (i) and let $\mathscr{U}$ be a directed open
$\lambda$-cover of $X$. Let $\mathscr{V}$ be a point-finite open refinement of
$\mathscr{U}$. Then $\mathscr{V}$ is an
interior-preserving open point-star refinement of $\mathscr{U}$.

(ii) $\Rightarrow$ (iii), (iv) $\Rightarrow$ (v) and (i) $\Rightarrow$ (iv). Clear.

(ii) $\Rightarrow$ (iv) and (iii) $\Rightarrow$ (v). By Lemma \ref{lem:4.1.6}.

(v)  (vi) $\Rightarrow$ (i). Assume (v) (vi). By Theorem \ref{th:4.1.7}, \ref{th:4.1.2},
$X$ is $\lambda$-submetacompact. By Theorem \ref{th:4.2.1},
we need only to show that $X$ is $\lambda$-discretely almost expandable. To this end, let 
$\mathscr{D}$ be a discrete $\lambda$-family of closed subsets of $X$ and
$\mathscr{D}^*$ be the induced cover. Then $\mathscr{D}^{*F}$ is
an interior-preserving directed open cover of $X$. By (v),
$\mathscr{D}^{*F}$ has a cushioned refinement; or by (vi) $\mathscr{D}^{*F}$ has a
semi-open point-star $\dot{F}$ refinement $\mathscr{L}$. 
Then $\mathscr{L}$ is also a point-star refinement of $\mathscr{D}^{*F}$. 
By Theorem \ref{th:4.1.9}, $\mathscr{D}^{*F}$ has a cushioned refinement. 
By Lemma \ref{lem:1.2.4}, $\mathscr{D}$ has
point-finite open expansion. Hence $X$ is $\lambda$-discretely almost expandable.
\end{proof} 

{\theorem{} \label{th:4.2.3}
\textup{([Krajewski 1971]\textsuperscript{\cite{Krajewski}})}
$X$ is $\lambda$-paracompact
iff $X$ is $\lambda$-submetacompact
and $\lambda$-expandable.
}

\begin{proof}
	See Corollary \ref{cor:4.2.7}.
\end{proof}

{\theorem{} \label{th:4.2.4}
\textup{([Junmila 1979a]\textsuperscript{\cite{Junnila1979a}})}
For any $X$ the following are equivalent.
\textup{(i)} $X$ is $\lambda$-paracompact.

\textup{(ii)}  Every directed open $\lambda$-cover of $X$ has an interior-preserving open
local-star refinement.

\textup{(iii)} Every interior-preserving directed open $\lambda$-cover of $X$ has a local-star 
refining sequence of interior-preserving open covers of $X$.

\textup{(iv)} Every directed open $\lambda$-cover of $X$ has closure-preserving closed
refinement $\mathscr{F}$ such that $\mathscr{F}^\circ$ covers $X$.

\textup{(v)} Every interior-preserving directed open $\lambda$-cover of $X$ has
a $\sigma$-closure-preserving  closed refinement $\mathscr{F}$ such that $\mathscr{F}^\circ$ covers $X$.
}

\begin{proof}
(i) $\Rightarrow$ (ii). Assume (i) and let $\mathscr{U}$ be a directed open
$\lambda$-cover of $X$. Let $\mathscr{V}$ be a locally finite open refinement of $\mathscr{U}$. Then
$\mathscr{V}$ is an interior-preserving open local-star refinement of $\mathscr{U}$.

(ii) $\Rightarrow$ (iii) and (iv) $\Rightarrow$ (v). Clear.

(ii) $\Rightarrow$ (iv) and (iii) $\Rightarrow$ (v) By Lemma \ref{lem:4.1.6}.



(v) $\Rightarrow$ (i).
Assume (v). By Theorem \ref{th:4.1.7}, $X$ is $\lambda$-submetacompact. 
By Theorem \ref{th:4.2.3}, we need only to show that
$X$ is $\lambda$-expandable. Let $\mathscr{B}$ be a locally finite $\lambda$-family of subsets.
of $X$. The induced cover
$\mathscr{B}^{*}$ is an interior-preserving directed open
$\lambda$-cover of $X$. By (v), $\mathscr{B}^*$ has a $\sigma$-cushioned open 
refinement. By Lemma \ref{lem:1.2.3}, $\mathscr{B}$ has a $\theta^*$-sequence 
of openvexpansions. Thus $X$ is $\lambda$-$\theta^*$-expandable. By 
Theorem \ref{th:3.3.4}, $X$ is $\lambda$-expandable.
\end{proof}

{\lemma{} \label{lem:4.2.5}
If $\mathscr{P}$ is a locally finite semi-open
cover of $X$, then $\mathscr{P}^F$ has
a locally finite closed refinement.
}

\begin{proof}
Let $\mathscr{P} = \bigcup_{n} \mathscr{P}_n$. For $S \in [\mathscr{P}]^{<\omega}$, let
$F_S = (\overline{\bigcap_{\alpha \in S} P_\alpha}) \setminus (\overline{X \setminus \bigcup_{\alpha \in S} P_\alpha})$.
For each $x \in X$, find $\alpha$ and $x \in P_\alpha$.
$\mathscr{F} = \{F_S : S \in [\mathscr{P}]^{<\omega}\}$ is a closed
cover of $X$. If $S \in [\mathscr{P}]^{<\omega}$,
then $F_S \subset \bigcup_{\alpha \in S} P_\alpha$. $\mathscr{F}$ is a refinement of $\mathscr{P}^{*\mathscr{F}}$. To see that
$\mathscr{F}$ is locally finite, let $x \in X$. Then $S(x) = \text{ind}(\mathscr{P})_x \in [\mathscr{P}]^{<\omega}$ and
$W = X \setminus \overline{\bigcup \{P_\gamma : \gamma \notin S(x)\}}$ is an open neighborhood of $x$. It is easy to see that
$\text{ind}(\mathscr{F})_W \subset \{S : S \subset S(x)\}$. It follows that $\text{ind}(\mathscr{F})_W$ is finite,
so $\mathscr{F}$ is locally finite.
\end{proof}


{\theorem{} \label{th:4.2.6}
$X$ is $\sigma$-paracompact iff every increasing
$\alpha$-cover of $X$ has
a locally finite semi-open refinement.
}

\begin{proof}
($\implies$) clear.
($\impliedby$) Assume that every increasing open $\alpha$-cover of $X$ has
a locally finite semi-open refinement. For every cardinal $\kappa$,


denote by $Q_\kappa$ the proposition: "every directed open
$\kappa$-cover
transfinite
of $X$ has a locally finite closed refinement." We use
induction
to
show that $Q_\kappa$ holds for each $\kappa$.
Clearly, $Q_\kappa$
is
trivially true for $\kappa$ finite. Let
$\omega \le \kappa \le \lambda$ such that $Q_\mu$ holds for each $\mu < \kappa$. To show
that $Q_\kappa$ holds, let
$\mathscr{U} = \{\mathscr{U}_\alpha : \alpha < \kappa\}$ be a directed open
$\alpha$-cover of $X$. $\mathscr{U}^* = \{\bigcup_{\gamma < \alpha} U_\gamma : \alpha < \kappa\}$ is an increasing open
$\alpha$-cover of $X$. Then $\mathscr{U}^*$ has
a locally finite semi-open refinement
$\mathscr{P} = \{P_\alpha : \alpha < \kappa\}$ such that
$P_\alpha \subset \bigcup_{\gamma < \alpha} U_\gamma$. By Lemma 4.2.5,
$\mathscr{P}^{*\mathscr{F}} = \{\bigcup_{\alpha \in S} P_\alpha : S \in [\kappa]^{<\omega}\}$ has
a locally finite closed refinement $\mathscr{K} = \{K_S : S \in [\kappa]^{<\omega}\}$ such that
$K_S \subset \bigcup_{\alpha \in S} P_\alpha$. For $S \in [\kappa]^{<\omega}$, let
$\delta(S) = \max S$. Then $K_S \subset \bigcup_{\gamma < \delta(S)} U_\gamma$. For $S \in [\kappa]^{<\omega}$ let
$\mathscr{W}_S = \{U_\gamma : \gamma < \delta(S)\} \cup \{X - K_S\}$
or
$\mathscr{W}_S = \{U_t : t \in [\delta(S)]^{<\omega}\} \cup \{U_t \cup (X - K_S) : t \in [\delta(S)]^{<\omega}\}$.
$\mathscr{W}_S$ is an
open cover of $X$ so $\mathscr{W}_S^\mathscr{F}$ is a directed open
$\alpha$-cover where $\mu = |[\delta(S)]^{<\omega}| = |\delta(S)| < \kappa$. By the induction
assumption, $\mathscr{W}_S^\mathscr{F}$ has a locally finite closed refinement $\mathscr{C}_S$. Let
$\mathscr{D}_S = \{C \cap K_S : C \in \mathscr{C}_S\}$
such that
$C_t \in \mathscr{C}_S$ and $C'_v = U_\psi \cup (X - K_S)$. Let
$\mathscr{D} = \bigcup \{D_S : S \in [\kappa]^{<\omega}\}$.
Since $\mathscr{K}$ and each $\mathscr{C}_S$ are locally finite, it is easy to see that $\mathscr{D}$ is
a locally finite closed refinement of $\mathscr{U}$. This completes the proof that
$Q_\kappa$ holds. By Theorem 3.1.3 $X$ is $\sigma$-paracompact.
\end{proof}
