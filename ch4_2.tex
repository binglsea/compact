\section{Paracompact and Metacompact Spaces} \label{sec:4-2}

{\theorem{} \label{th:4.2.1}
\textup{(\hstretch{1.8}{[}Boone 1973\hstretch{1.8}{]}\textsuperscript{\cite{Boone}})}
$X$ is $\lambda$-metacompact iff $X$ is $\lambda$-submetacompact and $\lambda$-discretely almost expandable.
}

\begin{proof}
See Corollary \ref{cor:7.1.6} (iii).
\end{proof}

{\theorem{} \label{th:4.2.2}
\textup{(\hstretch{1.8}{[}Junnila 1978, 1979a\hstretch{1.8}{]}\textsuperscript{\cite{Junnila1978}\cite{Junnila1979a}})}
For any $X$, the following are equivalent:

\textup{(i)} $X$ is $\lambda$-metacompact.

\textup{(ii)} Every directed open $\lambda$-cover of $X$ has an interior-preserving 
open  point-star refinement.

\textup{(iii)} Every interior-preserving directed open $\lambda$-cover of $X$ has an interior-preserving
open  point-star refinement.

\textup{(iv)} Every  directed open $\lambda$-cover of $X$ has a closure-preserving closed refinement.

\textup{(v)}  Every interior-preserving  directed open $\lambda$-cover of $X$ has a closure-preserving
closed refinement.

\textup{(vi)} Every open $\lambda$-cover of $X$ has a semi-open point-star $\dot{F}$-refinement.
}

\begin{proof} 
(i) $\Rightarrow$ (ii). Assume (i) and let $\mathscr{U}$ be a directed open
$\lambda$-cover of $X$. Let $\mathscr{V}$ be a point-finite open refinement of
$\mathscr{U}$. Then $\mathscr{V}$ is an
interior-preserving open point-star refinement of $\mathscr{U}$.

(ii) $\Rightarrow$ (iii), (iv) $\Rightarrow$ (v) and (i) $\Rightarrow$ (iv). Clear.

(ii) $\Rightarrow$ (iv) and (iii) $\Rightarrow$ (v). By Lemma \ref{lem:4.1.6}.

(v), (vi) $\Rightarrow$ (i). Assume (v), (resp. (vi)). By Theorem \ref{th:4.1.7}, (resp. \ref{th:4.1.2}),
$X$ is $\lambda$-submetacompact. By Theorem \ref{th:4.2.1},
we need only show that $X$ is $\lambda$-discretely almost expandable. To this end, let 
$\mathscr{D}$ be a discrete $\lambda$-family of closed subsets of $X$ and
$\mathscr{D}^*$ be the induced cover. Then $\mathscr{D}^{*F}$ is
an interior-preserving directed open cover of $X$. By (v),
$\mathscr{D}^{*F}$ has a cushioned refinement; or by (vi) $\mathscr{D}^{*F}$ has a
semi-open point-star $\dot{F}$ refinement $\mathscr{L}$. 
Then $\mathscr{L}$ is also a point-star refinement of $\mathscr{D}^{*F}$. 
By Theorem \ref{th:4.1.9}, $\mathscr{D}^{*F}$ has a cushioned refinement. 
By Lemma \ref{lem:1.2.4}, $\mathscr{D}$ has
point-finite open expansion. Hence $X$ is $\lambda$-discretely almost expandable.
\end{proof} 

{\theorem{} \label{th:4.2.3}
\textup{(\hstretch{1.8}{[}Krajewski 1971\hstretch{1.8}{]}\textsuperscript{\cite{Krajewski}})}
$X$ is $\lambda$-paracompact
iff $X$ is $\lambda$-submetacompact
and $\lambda$-expandable.
}

\begin{proof}
See Corollary \ref{cor:7.2.8}.
\end{proof}

{\theorem{} \label{th:4.2.4}
\textup{(\hstretch{1.8}{[}Junnila 1979a\hstretch{1.8}{]}\textsuperscript{\cite{Junnila1979a}})}
For any $X$ the following are equivalent.

\textup{(i)} $X$ is $\lambda$-paracompact.

\textup{(ii)}  Every directed open $\lambda$-cover of $X$ has an interior-preserving open
local star refinement.

\textup{(iii)} Every interior-preserving directed open $\lambda$-cover of $X$ has a local star 
refining sequence of interior-preserving open covers of $X$.

\textup{(iv)} Every directed open $\lambda$-cover of $X$ has closure-preserving closed
refinement $\mathscr{F}$ such that $\mathscr{F}^\circ$ covers $X$.

\textup{(v)} Every interior-preserving directed open $\lambda$-cover of $X$ has
a $\sigma$-closure-preserving  closed refinement $\mathscr{F}$ such that $\mathscr{F}^\circ$ covers $X$.
}

\begin{proof}
(i) $\Rightarrow$ (ii). Assume (i) and let $\mathscr{U}$ be a directed open
$\lambda$-cover of $X$. Let $\mathscr{V}$ be a locally finite open refinement of $\mathscr{U}$. Then
$\mathscr{V}$ is an interior-preserving open local star refinement of $\mathscr{U}$.

(ii) $\Rightarrow$ (iii) and (iv) $\Rightarrow$ (v). Clear.

(ii) $\Rightarrow$ (iv) and (iii) $\Rightarrow$ (v) By Lemma \ref{lem:4.1.6}.

(v) $\Rightarrow$ (i).
Assume (v). By Theorem \ref{th:4.1.7}, $X$ is $\lambda$-submetacompact. 
By Theorem \ref{th:4.2.3}, we need only to show that
$X$ is $\lambda$-expandable. Let $\mathscr{B}$ be a locally finite $\lambda$-family of subsets
of $X$. The induced cover
$\mathscr{B}^{*}$ is an interior-preserving directed open
$\lambda$-cover of $X$. By (v), $\mathscr{B}^*$ has a $\sigma$-cushioned open 
refinement. By Lemma \ref{lem:1.2.3}, $\mathscr{B}$ has a $\theta^*$-sequence 
of open expansions. Thus $X$ is $\lambda$-$\theta^*$-expandable. By 
Theorem \ref{th:3.3.4}, $X$ is $\lambda$-expandable.
\end{proof}

{\lemma{} \label{lem:4.2.5}
If $\mathscr{P}$ is a locally finite semi-open
cover of $X$, then $\mathscr{P}^F$ has
a locally finite closed refinement.
}

\begin{proof}
Let $\mathscr{P} = \{P_\alpha: \alpha\in \Delta\}$ be a locally finite semi-open cover of $X$.
For $S \in \hstretch{1.8}{[}\Delta\hstretch{1.8}{]}^{<\omega}$, let
$$F_S = (\overline{\raisebox{-3pt}{\scalebox{1.5}{$\cap$}}_{\alpha \in S} P_\alpha}) \cap
 (\overline{X \setminus \raisebox{-3pt}{\scalebox{1.5}{$\cup$}}_{\gamma \in\Delta\setminus S} P_\gamma})$$
\noindent For each $x \in X$,  $S = \operatorname{ind}(\mathscr{P})_x \in \hstretch{1.8}{[}\Delta\hstretch{1.8}{]}^{<\omega}$ 
and $x \in F_S$. Hence
$\mathscr{F} = \{F_S: S \in \hstretch{1.8}{[}\Delta\hstretch{1.8}{]}^{<\omega}\}$ is a closed
cover of $X$. If $S \in \hstretch{1.8}{[}\Delta\hstretch{1.8}{]}^{<\omega}$,
then $F_S \subset \bigcup_{\alpha \in S} P_\alpha$. 
So, $\mathscr{F}$ is a refinement of $\mathscr{P}^{F}$. To see that
$\mathscr{F}$ is locally finite, let $x \in X$. Then 
$S(x) = \text{ind}(\overline{\mathscr{P}})_x \in \hstretch{1.8}{[}\Delta\hstretch{1.8}{]}^{<\omega}$ and
$W = X \setminus \overline{\bigcup \{P_\gamma: \gamma \in\Delta\setminus S(x)\}}
\in \operatorname{top}(x)$. It is easy to see that
$\text{ind}(\mathscr{F})_W \subset \{S: S \subset S(x)\}$. 
It follows that $\text{ind}(\mathscr{F})_W$ is finite,
so $\mathscr{F}$ is locally finite.
\end{proof}


{\theorem{} \label{th:4.2.6}
$X$ is $\lambda$-paracompact iff every increasing
$\lambda$-cover of $X$ has
a locally finite semi-open refinement.
}

\begin{proof}
($\Rightarrow$). Clear.

($\Leftarrow$) Assume that every increasing open $\lambda$-cover of $X$ has
a locally finite semi-open refinement. For every cardinal $\kappa$,
denote a proposition  by

$\mathcal{P}_\kappa$: \textit{Every directed open
$\kappa$-cover of $X$ has a locally finite closed refinement.}

\noindent We use transfinite induction to show that $\mathcal{P}_\kappa$ holds for each $\kappa$.
Clearly, $\mathcal{P}_\kappa$ is trivially true for $\kappa$ finite. Let
$\omega \le \kappa \le \lambda$ such that $\mathcal{P}_\mu$ holds for each $\mu < \kappa$. To show
that $\mathcal{P}_\kappa$ holds, let
$\mathscr{U} = \{U_\alpha: \alpha < \kappa\}$ be a directed open
cover of $X$. 
$\mathscr{U}^* = \{\bigcup_{\gamma < \alpha} U_\gamma: \alpha < \kappa\}$ is an increasing open
$\lambda$-cover of $X$. Then $\mathscr{U}^*$ has
a locally finite semi-open refinement
$\mathscr{V} = \{V_\alpha: \alpha < \kappa\}$ such that
$V_\alpha \subset \bigcup_{\gamma < \alpha} U_\gamma$. By Lemma \ref{lem:4.2.5},
$\mathscr{V}^{F} = \{\bigcup_{\alpha \in S} V_\alpha: S \in \hstretch{1.8}{[}\kappa\hstretch{1.8}{]}^{<\omega}\}$ has
a locally finite closed refinement $\mathscr{K} = \{K_S: S \in \hstretch{1.8}{[}\kappa\hstretch{1.8}{]}^{<\omega}\}$ 
such that
$K_S \subset \bigcup_{\alpha \in S} V_\alpha$. For $S \in \hstretch{1.8}{[}\kappa\hstretch{1.8}{]}^{<\omega}$, let
$\delta(S) = \max \{\alpha < \kappa: \alpha\in S\}$. Then 
$K_S \subset \bigcup_{\gamma < \delta(S)} U_\gamma$. 
For $S \in \hstretch{1.8}{[}\kappa\hstretch{1.8}{]}^{<\omega}$ let
$$\mathscr{W}_S = \{U_\gamma: \gamma < \delta(S)\} \cup \{X\setminus K_S\}.$$ 
\noindent As a convention adopted since Definiton \ref{def:1.1.1} and Definition \ref{def:1.2.1},
$U_t  = \bigcup_{\alpha\in t}U_\alpha$ for $t\in \hstretch{1.8}{[}\delta(S)\hstretch{1.8}{]}^{<\omega}$ or $t\in \hstretch{1.8}{[}\Delta\hstretch{1.8}{]}^{<\omega}$.
Then
\begin{gather}
\mathscr{W}_S^F = \{U_t: t \in \hstretch{1.8}{[}\delta(S)\hstretch{1.8}{]}^{<\omega}\} \cup \{U_t \cup (X \setminus K_S): 
t \in \hstretch{1.8}{[}\delta(S)\hstretch{1.8}{]}^{<\omega}\}. \label{eq:4.2.6.1}
\end{gather}
\noindent $\mathscr{W}_S$ is an
open cover of $X$, so $\mathscr{W}_S^{F}$ is a directed open
$\mu$-cover where $\mu = |\hstretch{1.8}{[}\delta(S)\hstretch{1.8}{]}^{<\omega}| = |\delta(S)| < \kappa$. 
By the induction hypothesis, $\mathscr{W}_S^{F}$ has a locally finite closed refinement $\mathscr{C}_S$. 
We may assume the refinement $\mathscr{C}_S$ is precise, so from \eqref{eq:4.2.6.1} we can let
\begin{gather}
\mathscr{C}_S = \{C_t: t\in \hstretch{1.8}{[}\delta(S)\hstretch{1.8}{]}^{<\omega} \}
\cup \{{C'}_t: t\in \hstretch{1.8}{[}\delta(S)\hstretch{1.8}{]}^{<\omega}\} \label{eq:4.2.6.2}
\end{gather}
such that
\begin{gather}
	C_t \subset U_t, \; {C'}_t \subset U_t\cup (X \setminus K_S) \text{ for all } t\in \hstretch{1.8}{[}\delta(S)\hstretch{1.8}{]}^{<\omega}. \label{eq:4.2.6.3}
\end{gather}
Let
\begin{gather}
\mathscr{D} _S=\{C_t\cap K_S: t\in \hstretch{1.8}{[}\delta(S)\hstretch{1.8}{]}^{<\omega}\} \cup
  \{C'_t\cap K_S: t\in \hstretch{1.8}{[}\delta(S)\hstretch{1.8}{]}^{<\omega}\}, \\
\mathscr{D} = \raisebox{-3pt}{\scalebox{1.5}{$\cup$}} \{D_S: S \in \hstretch{1.8}{[}\kappa\hstretch{1.8}{]}^{<\omega}\}. 
\end{gather}
\noindent Since $\mathscr{K}$ and each 
$\mathscr{C}_S$ are locally finite, it is easy to see that $\mathscr{D}$ is
a locally finite closed refinement of $\mathscr{U}$. This completes the proof of
$\mathcal{P}_\kappa$. By Theorem \ref{th:3.1.3},  $X$ is $\lambda$-paracompact.
\end{proof}
