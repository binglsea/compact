\section{Subparacompact and Discretely Subexpandable spaces} \label{sec:6.2}

{\theorem{\label{th:6.2.1}} For any space $X$, the following are equivalent:

\textup{(i)} $X$ is  $\lambda$-discretely subexpandable.

\textup{(ii)} Every $D\omega$ $\lambda$-cover of $X$ has a $\sigma$-discrete closed refinement.

\textup{(iii)}  Every $D\omega$ $\lambda$-cover of $X$ has an $\varepsilon$-sequence of open refinements.

\textup{(iv)} Every interior-preserving $D\omega$ $\lambda$-cover of $X$ has an 
$\varepsilon$-sequence of interior-preserving open covers of $X$.

\textup{(v)} Every interior-preserving $D\omega$ $\lambda$-cover of $X$ has a 
point-star refining sequence of interior-preserving  open covers of $X$.

\textup{(vi)} Every interior-preserving $D\omega$ $\lambda$-cover of $X$ has a 
point-star refining sequence of  semi-open covers of $X$.

\textup{(vii)}  Every interior-preserving  $D\omega$ $\omega$-cover of $X$ has a 
$\sigma$-cushioned refinement.
}

\begin{proof} (i) $\Rightarrow$ (ii). Assume (i) and let $\mathscr{U}$ be a 
$D\omega$ $\lambda$-cover of $X$. Then $X$ is $\lambda$-discretely $\theta$-subexpandable. 
By Theorem \ref{th:6.1.2} and \ref{th:3.1.5}, there exists a countable closed cover 
$\{F_n: n < \omega\}$ of $X$ such that each $\mathscr{U}|_{F_n}$ has a 
point-finite open refinement $\mathscr{V}_n$ in the subspace $F_n$. 
Since the closed subspace $F_n$ of $X$ is $\lambda$-discretely subexpandable by 
Theorem \ref{th:3.3.7}, $\mathscr{V}_n$ has a $\sigma$-discrete closed refinement 
$\mathscr{P}_n$ in $F_n$. Then $\bigcup_{n<\omega} \mathscr{P}_n$ is a 
$\sigma$-discrete closed refinement of $\mathscr{U}$.

(ii) $\Rightarrow$ (iii) and (iv). Assume (ii) and let $\mathscr{U}=\{U_\alpha: \alpha < \lambda\}$ 
be a $D\omega$ cover of $X$. Then $\mathscr{U}$ has a refinement 
$\mathscr{D} = \bigcup_{n < \omega}\mathscr{D}_n$, where each
 $\mathscr{D}_n = \{D_{n,\alpha}: \alpha< \lambda\}$ is a discrete family of closed subsets 
 of $X$ with $D_{n,\alpha}\subset U_\alpha$ for each $\alpha < \lambda$. 
 For $n < \omega, \alpha < \lambda$, let 
 $G_{n,\alpha} = U_\alpha\setminus \bigcup_{\gamma\in \lambda\setminus\{\alpha\}}D_{n,\gamma}$
 Then $D_\alpha \subset G_{n, \alpha}\subset U_\alpha$ and 
 each $\mathscr{G} = \{G_{n,\alpha}: \alpha< \lambda\}$ is an open refinement o $\mathscr{U}$.
 For each $x\in X$ there exists $m<\omega$ and $\alpha_0 < \alpha$ such that 
 $x\in D_{m,\alpha_0}$. Then
 $\operatorname{ind}(\mathscr{G}_m)_x \subset \{\alpha_0\}$, so $\{\mathscr{G}_n: n < \omega\}$ is
 an $\varepsilon$-sequence. We have shown that (ii) $\Rightarrow$ (iii). If $\mathscr{U}$ is
 interior-preserving open, then each $\mathscr{G}_n$ is interior-preserving open,
 so (iv) holds. 

(iii) $\Rightarrow$ (vi) and (iv) $\Rightarrow$ (v) $\Rightarrow$ (vi). Clear.

(v) $\Rightarrow$ (vii) is clear by Theorem \ref{th:4.1.9}. 

(vii) $\Rightarrow$ (i). Assume (vii) and let $\mathscr{D}$ be a discrete $\lambda$-family 
of closed subsets of $X$. The induced cover $\mathscr{D}^*$ is an interior-preserving 
$D\omega$-cover. By (vii), $\mathscr{D}^*$ has a $\sigma$-cushioned refinement. By Lemma
\ref{lem:1.2.4}, $\mathscr{D}$ has a $\varepsilon$-sequence of open expansions, so (i) holds.
\end{proof}

{\corollary{} \label{cor:6.2.2}
$\lambda$-subparacompact spaces are $\lambda$-discretely subexpandable. \hfill
}


{\theorem{\label{th:6.2.3}}  $X$ is $\lambda$-subparacompact if and only if $X$ is 
$\lambda$-discretely subexpandable and $\lambda$-strictly quasi-paracompact.
}

\begin{proof} 
	By Theorem \ref{th:6.2.1}.
\end{proof}

{\corollary{} \label{cor:6.2.4}
\textup{(i)} \textup{([Jiang 1988]\textsuperscript{\cite{Jiang1988}}, 
	[Zhu 1988]\textsuperscript{\cite{Zhu1988}})}
 $X$ is subparacompact if and only if $X$ is $\lambda$-discretely subexpandable and 
 strictly quasi-paracompact. 
 
\textup{(ii)} \textup{([Chaber 1979]\textsuperscript{\cite{Chaber}})}
$X$ is subparacompact if and only if $X$ is $\lambda$-discretely subexpandable 
and $\lambda$-submetacompact.
}
