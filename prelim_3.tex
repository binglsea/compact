\section{Inrerior-Preserving Families and Semi-Open Covers}  
\label{sec:prelim_more_covers}



{\definition{}\label{def_prelim3_cp_ip}
	 Let Let $\mathcal{U} = \{U_{\alpha} : \alpha \in\Delta\}$ be a family of subsets of $X$.

\textup{(i) (Michael [1957]\cite{Michael1957})}
$\mathcal{U}$ is 
\uwave{closure-preserving}\index{closure-preserving, \S\ref{sec:prelim_more_covers}} 
(simply denoted by CP) if for any $\Gamma \subset \Delta$,
$$\overline{\bigcup_{\alpha\in \Gamma} U_{\alpha}} = \bigcup_{\alpha\in \Gamma} \overline{U_{\alpha}}$$

\textup{(ii) (Junnila [1979a]\cite{Junnila1979a})} 
$\mathcal{U}$ is 
\uwave{interior-preserving}\index{interior-preserving, \S\ref{sec:prelim_more_covers}} 
 (simply denoted by IP) if for any $\Gamma \subset \Delta$,
$$\left(\bigcap_{\alpha\in \Gamma} U_{\alpha}\right)^{\circ} = \bigcap_{\alpha\in \Gamma} U_{\alpha}^{\circ}$$
} %end of def

{\lemma{}  \label{lemma_prelim3_ip_equiv}
For a family $\mathcal{U} = \{U_\alpha : \alpha \in \Delta\}$ of open sets in $X$, the following are equivalent.

\textup{(i)} $\mathcal{U}$ is interior-preserving.
	
\textup{(ii)} For each  $\Gamma \subset \Delta$, $\bigcap_{\alpha \in \Gamma} U_\alpha$ is open.
	
\textup{(iii)}  For each $x \in \bigcup\mathcal{U}$, $\bigcap(U)_x$ is open.
\qedsymbol
} %end of lemma

{\theorem{} Let $\mathcal{U} = \{U_\alpha : \alpha \in \Delta\}$ be an interior-preserving
 open family in $X$. For each $\beta \in B$, let $\Delta_\beta \subset \Delta$. Let

$$\mathcal{U}' = \left\{ \bigcap_{\alpha \in \Delta_\beta} U_\alpha : \beta \in B \right\}, \quad \mathcal{U}'' = \left\{ \bigcup_{\alpha \in \Delta_\beta} U_\alpha : \beta \in B \right\}.$$
Then the families $\mathcal{U}'$ and $\mathcal{U}''$ are interior-preserving open.
} %end of teorem

\vspace{10pt}
Proof. If $\mathcal{U}$ is IP open, then clearly $\mathcal{U}''$ is IP open. To see that $\mathcal{U}'$ is IP open, let $P \subset B$. It is easily verified that

$$\bigcap_{\beta \in P} \left( \bigcup_{\alpha \in A_\beta} U_\alpha \right) = \bigcup \left\{ \bigcap_{\beta \in P} U_{f(\beta)} : f \in \prod_{\beta \in P} A_\beta \right\}.$$

By Lemma \ref{lemma_prelim3_ip_equiv}, $\mathcal{U}'$ is interior-preserving open.
\qedsymbol

{\definition{} (\textup{Junnila [1979b]\cite{Junnila1979b})} 
	A cover $\mathcal{E}$ of $X$ is 
\uwave{semi-open}\index{semi-open, \S\ref{sec:prelim_more_covers}} 
if for every $x \in X$, $x \in  \operatorname{st}(x, \mathcal{E})^\circ$.

} %end of def

{\definition{}  If $\mathcal{E}$ is a cover of $X$, then the following are equivalent.

\textup{(i)}  $\mathcal{E}$ is semi-open.

\textup{(ii)} For each $A \subset X$, 
$\overline{A} \subset \operatorname{st}(A, \mathcal{E})$.

\textup{(iii} For each $A \subset X$, $A \subset  \operatorname{st}(A, \mathcal{E})^\circ$.
}

\vspace{8pt}
Clearly, every closure-preserving closed cover, as well as open covers, are semi-open.


{\definition{} A subset $A$ of $X$ is 
\uwave{functionally open}\index{functionally open, \S\ref{sec:prelim_more_covers}} 
or a cozero set 
\uwave{cozero set}\index{cozero set, \S\ref{sec:prelim_more_covers}} 
if $A = f^{-1}(0, 1] = \{x \in X : f(x) > 0\}$ for some continuous function 
$f : X \rightarrow [0, 1]$.
A family of subsets of $X$ is functionally open if every member of it is functionally open.
}%end of def


{\definition{} Let $\Phi = \{\varphi_\alpha : \alpha \in \Delta\}$ be a family of continuous functions from $X$ into $[0, 1]$ such that the family 
$\{\varphi_\alpha^{-1}(0, 1] : \alpha \in \Delta\}$ is point finite. 
For $x \in X$, let $\Delta(x) = \{\alpha \in \Delta : x \in\varphi_\alpha^{-1}(0, 1]\}$.

\textup{(i)} Define $\varphi : X \rightarrow \mathbb{R}$ by 
%$\varphi = \sum_{\alpha \in \Delta} \varphi_\alpha$.

$$\varphi(x) = \sum_{\alpha \in \Delta} \varphi_\alpha(x) = 
\begin{cases} 0, 
	& \text{if } \Delta(x) = \emptyset, 
	\\ \varphi_{\alpha_1}(x) + \dots + \varphi_{\alpha_n}(x), 
	& \text{if } \Delta(x) = \{\alpha_1, \dots, \alpha_n\}. 
\end{cases}$$

\textup{(ii)} A family $\Phi$ is a 
\uwave{partition of unity}\index{partition of unity, \S\ref{sec:prelim_more_covers}} 
 on $X$ if for each $x \in X$, $\sum_{\alpha \in \Delta} \varphi_\alpha(x) = 1$.
}


{\lemma{}
\textup{(i)} If $\mathcal{V}$ is a locally finite family of functionally open subsets of $X$, 
then $\bigcup \mathcal{V}$ is functionally open.

\textup{(ii)}  If $\mathcal{V}$ is a locally finte functionally open cover of $X$, 
then there is a partition of unity $\{\varphi_\alpha: \alpha\in\Delta\}$ on $X$ 
such that $\mathcal{V} = \{\varphi_\alpha^{-1}(0, 1]:  \alpha\in\Delta\}$.
}

Proof. 
 Let $\mathcal{V} = \{f_\alpha^{-1}(0, 1] : \alpha \in \Delta\}$ be a locally finite
 family of functionally open subsets of $X$ where each $f_\alpha : X \rightarrow [0, 1]$ is continuous.

(i) For each $x \in X$, $\Delta(x) = \{\alpha \in \Delta : x \in f_\alpha^{-1}(0, 1]\}$ is finite.
