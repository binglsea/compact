\section{Interior-Preserving Families and Semi-Open Covers}  
\label{sec:1_3}

{\definition{}\label{def:1-3-1}
	 Let Let $\mathscr{U} = \{U_{\alpha} : \alpha \in\Delta\}$ be a family of subsets of $X$.

\textup{(i) (Michael [1957]\textsuperscript{\cite{Michael1957}})}
$\mathscr{U}$ is 
\uwave{closure-preserving}\index{closure-preserving| \S\ref{sec:1_3}\hfill} 
if for any $\Gamma \subset \Delta$,
$$\overline{\bigcup_{\alpha\in \Gamma} U_{\alpha}} = \bigcup_{\alpha\in \Gamma} \overline{U_{\alpha}}$$

\textup{(ii) (Junnila [1979a]\textsuperscript{\cite{Junnila1979a}})} 
$\mathscr{U}$ is 
\uwave{interior-preserving}\index{interior-preserving| \S\ref{sec:1_3}\hfill} 
 if for any $\Gamma \subset \Delta$,
$$\left(\bigcap_{\alpha\in \Gamma} U_{\alpha}\right)^{\circ} = \bigcap_{\alpha\in \Gamma} U_{\alpha}^{\circ}$$
} %end of def

{\lemma{}  \label{lem:1-3-2}
For a family $\mathscr{U} = \{U_\alpha : \alpha \in \Delta\}$ of open sets in $X$, the following are equivalent.

\textup{(i)} $\mathscr{U}$ is interior-preserving.
	
\textup{(ii)} For each  $\Gamma \subset \Delta$, $\bigcap_{\alpha \in \Gamma} U_\alpha$ is open.
	
\textup{(iii)}  For each $x \in \bigcup\mathscr{U}$, $\bigcap(\mathscr{U})_x$ is open.
\hfill \qedsymbol
} %end of lemma

{\theorem{} \label{th:1-3-3}
Let $\mathscr{U} = \{U_\alpha : \alpha \in \Delta\}$ be an interior-preserving
 open family in $X$. For each $\beta \in B$, let $\Delta_\beta \subset \Delta$. Let

$$\mathscr{U}' = \left\{ \bigcap_{\alpha \in \Delta_\beta} U_\alpha : \beta \in B \right\}, \quad \mathscr{U}'' = \left\{ \bigcup_{\alpha \in \Delta_\beta} U_\alpha : \beta \in B \right\}.$$
Then the families $\mathscr{U}'$ and $\mathscr{U}''$ are interior-preserving open.
} %end of teorem

\begin{proof} 
If $\mathscr{U}$ is interior-preserving open, then clearly $\mathscr{U}''$ is 
interior-preserving open. To see that $\mathscr{U}'$ is interior-preserving open, let $D \subset B$. It is easily verified that

$$\bigcap_{\beta \in D} \left( \bigcup_{\alpha \in \Delta_\beta} U_\alpha \right) 
= \bigcup \left\{ \bigcap_{\beta \in D} U_{f(\beta)} : f \in \prod_{\beta \in D} \Delta_\beta \right\}.$$
By Lemma \ref{lemma_prelim3_ip_equiv}, $\mathscr{U}'$ is interior-preserving open.
\end{proof} 

{\definition{} \label{def:1-3-4}
(\textup{Junnila [1979b]\textsuperscript{\cite{Junnila1979b}})} 
	A cover $\mathscr{E}$ of $X$ is 
\uwave{semi-open}\index{semi-open| \S\ref{sec:1_3}\hfill} 
if for every $x \in X$, $x \in  \operatorname{st}(x, \mathscr{E})^\circ$.

} %end of def

{\lemma{} \label{lem:1-3-5}
If $\mathscr{E}$ is a cover of $X$, then the following are equivalent.

\textup{(i)}  $\mathscr{E}$ is semi-open.

\textup{(ii)} For each $A \subset X$, 
$\overline{A} \subset \operatorname{st}(A, \mathscr{E})$.

\textup{(iii} For each $A \subset X$, $A \subset  \operatorname{st}(A, \mathscr{E})^\circ$.
\hfill \qedsymbol
}

\vspace{8pt}
Clearly, every closure-preserving closed cover, as well as open covers, are semi-open.


{\definition{} \label{def:1-3-6}
A subset $A$ of $X$ is 
\uwave{functionally open}\index{functionally open| \S\ref{sec:1_3}\hfill} 
or a 
\uwave{cozero set}\index{cozero set| \S\ref{sec:1_3}\hfill} 
if $A = f^{-1}(0, 1] = \{x \in X : f(x) > 0\}$ for some continuous function 
$f : X \rightarrow [0, 1]$.
A family of subsets of $X$ is functionally open if every member of it is functionally open.
}%end of def


{\definition{} \label{def:1-3-7}
Let $\Phi = \{\varphi_\alpha : \alpha \in \Delta\}$ be a family of continuous functions from $X$ into $[0, 1]$ such that the family 
$\{\varphi_\alpha^{-1}(0, 1] : \alpha \in \Delta\}$ is point finite. 
For $x \in X$, let $\Delta(x) = \{\alpha \in \Delta : x \in\varphi_\alpha^{-1}(0, 1]\}$.

\textup{(i)} Define $\varphi : X \rightarrow \mathbb{R}$ by 
%$\varphi = \sum_{\alpha \in \Delta} \varphi_\alpha$.

$$\varphi(x) = \sum_{\alpha \in \Delta} \varphi_\alpha(x) = 
\begin{cases} 0, 
	& \text{if } \Delta(x) = \emptyset, 
	\\ \varphi_{\alpha_1}(x) + \dots + \varphi_{\alpha_n}(x), 
	& \text{if } \Delta(x) = \{\alpha_1, \dots, \alpha_n\}. 
\end{cases}$$

\textup{(ii)} A family $\Phi$ is a 
\uwave{partition of unity}\index{partition of unity| \S\ref{sec:1_3}\hfill} 
 on $X$ if for each $x \in X$, $\sum_{\alpha \in \Delta} \varphi_\alpha(x) = 1$.
}


{\lemma{}\label{lem:1-3-8}
\textup{(i)} If $\mathscr{V}$ is a locally finite family of functionally open subsets of $X$, 
then $\bigcup \mathscr{V}$ is functionally open.

\textup{(ii)}  If $\mathscr{V}$ is a locally finte functionally open cover of $X$, 
then there is a partition of unity $\{\varphi_\alpha: \alpha\in\Delta\}$ on $X$ 
such that $\mathscr{V} = \{\varphi_\alpha^{-1}(0, 1]:  \alpha\in\Delta\}$.
}

\begin{proof} 
 Let $\mathscr{V} = \{f_\alpha^{-1}(0, 1] : \alpha \in \Delta\}$ be a locally finite
 family of functionally open subsets of $X$ where each $f_\alpha : X \rightarrow [0, 1]$ is continuous.

(i) For each $x \in X$, $\Delta(x) = \{\alpha \in \Delta : x \in f_\alpha^{-1}(0, 1]\}$ is finite.
Define a continuous function $f : X \rightarrow [0, 1]$ by
$f(x) = \max \{ f_\alpha : x \in \Delta \}$ for $x\in X$.
Then $\bigcup \mathscr{V} = f^{-1}(0, 1]$ is functionally open.

(ii) If $\mathscr{V}$ covers $X$, then 
$g = \sum_{\alpha\in\Delta}f_\alpha : X \rightarrow \mathbb{R}$ 
is a continuous function such that $g(x) > 0$ for all $x\in X$. 
For each $\alpha \in \Delta$, define a continuous function 
$\varphi_\alpha : X \rightarrow [0, 1]$ by $\varphi_\alpha(x) = f_\alpha(x) /g(x)$. 
Then $\sum_{\alpha\in\Delta}\varphi_\alpha(\alpha) = 1$ for each $x \in X$,
and  $\varphi_\alpha^{-1}(0, 1] = f_\alpha^{-1}(0, 1]$.
Therefore, $\{\varphi_\alpha: \alpha\in \Delta\}$ is a partition of unity on $X$.
\end{proof}
