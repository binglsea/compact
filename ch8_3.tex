\section{Strongly Paracompact and Ultraparacompact Spaces} \label{sec:8.3}


{\theorem{\label{th:8.3.1}} 
For any \(\lambda\) the following are equivalent:

\textup{(i)} \(X\) is \(\lambda\)-ultraparacompact.

\textup{(ii)} Every open \(\lambda\)-cover of \(X\) has a discrete closed refinement.

\textup{(iii)}  Every open \(\lambda\)-cover of \(X\) has a disjoint open refinement.

\textup{(iii)}  Every open \(\lambda\)-cover of \(X\) has a disjoint functionally open refinement.

\textup{(iv)} Every open \(\lambda\)-cover of \(X\) has a disjoint semi-open refinement.
}

\begin{proof}
(i) \(\Rightarrow\) (ii) \(\Rightarrow\) (iii)  \(\Rightarrow\)  (v) 
and (iv) \(\Rightarrow\) (iii).  Clear.

(ii) \(\Rightarrow\) (iv) Assume (ii). By theorem \ref{th:8.1.5},
$X$ is $\lambda$-collectionwise normal.
Let \(\mathscr{U}= \{U_\alpha: \alpha < \lambda\}\) be an open \(\lambda\)-cover of \(X\). 
Let \(\mathscr{F}= \{F_\alpha: \alpha < \lambda\}\) be a discrete closed cover such that 
$F_\alpha\subset U_\alpha$ for all $\alpha < \lambda$.
Let  $\mathscr{H}= \{H_\alpha: \alpha < \lambda\}$ be a disjoint functionally open 
expansion of \(\mathscr{F}\) such that $F_\alpha \subset H_\alpha \subset U_\alpha$.
Then $\mathscr{H}$ is a refinement of $\mathscr{U}$.

(v) \(\Rightarrow\) (iii). Suppose \(\mathscr{U}\) is an open \(\lambda\)-cover of \(X\). 
By (v), let \(\mathscr{V} = \{V_\beta: \beta\in \Gamma\}\) be a disjoint semi-open refinemnt 
of \(\mathscr{U}\). To see that each $V_\beta$ is open, let $x\in V_\beta$. Then
$\operatorname{st}(x, \mathscr{V}) = V_\alpha$. 
Hence $x\in \operatorname{st}(x, \mathscr{V}) ^\circ\subset V_\beta^\circ$.
Then $V_\beta = V_\beta^\circ$ is open.
\end{proof} 

{\theorem{\label{th:8.3.2}} 
 For any $X$ the following are equivalent:

\textup{(i)} \(X\) is \(\lambda\)-ultraparacompact.

\textup{(ii)} \(X\) is countably ultraparacompact and $\lambda$-screenable.

\textup{(iii)} \(X\) is countably ultraparacompact and $\lambda$-fully normal.

\textup{(iv)}  \(X\) is countably ultraparacompact and every open \(\lambda\)-cover 
of \(X\) has a  cushioned refinement.

\textup{(v)}  \(X\) is countably ultraparacompact, $\lambda$-collectionwise normal
and $\lambda$-$D\omega^2$-refinable.
}

\begin{proof}
(i) \(\Rightarrow\) (ii) and (i) \(\Rightarrow\) (iv) Clear.

(ii) \(\Rightarrow\) (iii). Assume (ii), then $X$ is countably shrinkable. By
Theorem \ref{th:8.2.1}, $X$ is $\lambda$-fully normal.

(iii) \(\Rightarrow\) (v). Assume (iii). By Corollary \ref{cor:8.1.6}, $X$ is  
$\lambda$-collectionwise normal. Since $X$ is $\lambda$-paracompact, it is 
$D\omega^2$-refinable.

(iv) \(\Rightarrow\) (v). Assume (iv). By Theorem \ref{th:8.1.5}, $X$ is  
$\lambda$-collectionwise normal. By Theorem \ref{th:4.1.1}, $X$ is $\lambda$-subparacompacr, so it is $\lambda$-$D\omega^2$-refinabel.

(v) \(\Rightarrow\) (i) Assume (v), and let 
\(\mathscr{U}=\{U_\alpha: \alpha < \lambda\}\) be an open cover of \(X\). 
Then $\mathscr{U}$ is a $D\omega^2$-cover.
By Theorem \ref{th:8.1.5}, $\mathscr{U}$ is a normal cover.
By Theorem \ref{th:2.1.6}, $\mathscr{U}$ has a refinement
$\mathscr{V}=\{\bigcup_{n<\omega}\mathscr{V}_n\}$ where each
$\mathscr{V}_n=\{V_{n, \alpha}:\alpha < \omega\}$ is a discrete open
with $V_{n, \alpha}\subset U_\alpha$ for all $\alpha$.
Let $\{W_n: n<\omega\}$ be a discrete open cover of $X$ 
such that $W_n=\bigcup\mathscr{V}_n$ for each $n<\omega$. For $\alpha < \lambda$,
let $G_\alpha = \bigcup_{n < \omega}(W_n\cap V_{n,\alpha})$. 
Then $\{G_\alpha:\alpha < \lambda\}$ is a disjoint open refinement of $\mathscr{U}$,
so $X$ is  $\lambda$-ultraparacompact.
\end{proof} 

Let $\mathscr{E}$ be a family of subsets of $X$ and $n\ge 1$. A $n$-sequence 
$C_1, C_2, \cdots, C_n$ in $\mathscr{E}$ is a 
\uwave{$n$-\textit{link from} $C_1$ \textit{to} $C_n$}
\index{$n$-link from $C_1$ to $C_n$| \S\ref{sec:8.3}\hfill}
if for each $1 \le i \le n-1$, $C_i\bigcap C_{i+1}\ne \emptyset$.
A
\uwave{\textit{link in} $\mathscr{E}$ \textit{from} $A$ \textit{to} $B$}
\index{link in $\mathscr{E}$ from $A$ to $B$| \S\ref{sec:8.3}\hfill} is a $n$-link from 
$C_1$ to $C_n$ for some $n\ge 1$ such that $C_1 = A$ and $C_n=B$.
$\mathscr{E}$ is
\uwave{\textit{connected}}
\index{connected| \S\ref{sec:8.3}\hfill} 
if for every $A, B\in \mathscr{E}$ there
exists a link in $\mathscr{E}$ from $A$ to $B$. For any family 
$\mathscr{E}, D\in \mathscr{E}$ and $n\ge 1$, let
$$P_n(D)=\{A\in \mathscr{E}: \text{ there exists an $n$-link in } 
\mathscr{E} \text{ from } A \text{ to } B\}.$$

{\lemma{\label{lem:8.3.3}} 
Let $\mathscr{E}$ be a family in $X$.

\textup{(i)} 	If $\mathscr{E}$ is connected, then for any $D\in\mathscr{E}$,
	\[
	\mathscr{E}=\bigcup_{n\geq1}P_n(D),
	\]
	and for every $n\geq1$,
	\[
	P_{n+1}(D)
	=
	\bigcup\{(\mathscr{E})_E: E\in P_n(D)\}.
	\]
	
\textup{(ii)} 
	If $\mathscr{E}$ is connected and star-countable, then
	$\mathscr{E}$ is countable.
}

\begin{proof}
(i) By a routine operation.

(ii) Suppose $\mathscr{E}$ is connected and star-countable. Then for
each $E\in\mathscr{E}$, $(\mathscr{E})_E$ is countable.
If $D\in\mathscr{E}$, then
$P_1(D)=\{D\}, P_2(D)=(\mathscr{E})_{D}$
are countable. If $P_n(D)$ is countable, then by (i),
$P_{n+1}(D)=\bigcup\{(\mathscr{E})_E: E\in P_n(D)\}$
is countable. Thus each $P_n(D)$ is countable, then each
$P_n(D)$ is is countable, so $\mathscr{E} = \bigcup_{n\ge 1}P_n(D)$
is countable.
\end{proof} 


{\lemma{\label{lem:8.3.4}} 
Any family $\mathscr{B}$ of subsets of $X$ can be expressed as
$\mathscr{B}=\bigcup_{\alpha\in \Delta}\mathscr{E}_{\alpha}$
such that $\left(\bigcup\mathscr{E}_{\alpha}\right)\cap
\left(\bigcup\mathscr{E}_{\beta}\right)
=\emptyset$ whenever $\alpha\ne\beta$.
If $\mathscr{B}$ is star-countable, then each
$\mathscr{E}_{\alpha}$ is countable.
}

\begin{proof}
Let $\mathscr{B}$ be a family in $X$. Let
$$\hat{\mathscr{B}}=\{\mathscr{E}:\mathscr{E} \text{ is a connected subfamily of } \mathscr{B}\}.$$
Then for every chain 
$\mathbb{A}\subseteq\hat{\mathscr{B}}, \; \bigcup \mathbb{A}\in \hat{\mathscr{B}}$. By Zorn's lemma
for each $B\in \hat{\mathscr{B}},$ there exists a maximal element 
$\mathscr{E}^*$ in $(\hat{\mathscr{B}}, \subseteq)$
such that $\{B\}\subseteq\mathscr{E}^*$. 
Let $\{\mathscr{E}_\alpha:\alpha\in\Delta\}$ be the collection of all
maximal elements of $\hat{\mathscr{B}}$. Then there exists $\alpha <\Delta, \; 
 \mathscr{E}^* = \mathscr{E}_\alpha$, so
 $B\in\mathscr{E}_\alpha$.
It follows that $\mathscr{B}=\bigcup_{\alpha\in\Delta}\mathscr{E}_\alpha$. 
Now if there exists 
$x\in(\bigcup\mathscr{E}_\alpha)\cap(\bigcup\mathscr{E}_\beta)$, 
then $\mathscr{E}_\alpha\cup\mathscr{E}_\beta\in \hat{\mathscr{B}}$, so $\mathscr{E}_\alpha=\mathscr{E}_\alpha\cup\mathscr{E}_\beta=\mathscr{E}_\beta$, so $\alpha=\beta$.
Finally, if $\mathscr{B}$ is star-countable, then each $\mathscr{E}_\alpha$ 
is star-countable and connected. By Lemma \ref{lem:8.3.4},
$\mathscr{E}_\alpha$  countable.
\end{proof}

{\theorem{\label{th:8.3.5}} A regular space is normal if every open
cover has a $\sigma$-closure-preserving open refinement.
}

\begin{proof} 
Suppose $X$ is regular and every open cover of $X$ has a
$\sigma$-closure-preserving open refinement. Let $A$ and $B$ be disjoint closed subsets
of $X$. By regularity of $X$, let $\mathscr{G}$ be an open cover of $X$ such that $\mathscr{\overline{G}}$
refines $\{X-A, X-B\}$. Let $\mathscr{V}$ be a $\sigma$-closure-preserving open refinement of $\mathscr{G}$. Then $\mathscr{V}$ is a $\sigma$-cushioned open refinement of $\mathscr{U}$. By
Lemma \ref{lem:8.1.4}, $\mathscr{U}$ has a cushioned refinement.
Hence there exist disjoint open sets $U$ and $V$ such that $A\subset U$ 
and $B\subset V$,
so $X$ is normal.
\end{proof}

{\lemma{\label{lem:8.3.6}}  
Suppose $X$ is countably shrinkable and 
$\mathscr{U} = \{U_\alpha: \alpha\in \Delta\}$
be an open cover of $X$. Let $\mathscr{K} = \{K_\beta: \beta\in\Gamma\}$ be a
disjoint open and closed cover of $X$ such that for each $\beta\in\Gamma$,
there exists a countable subfamily $\mathscr{U}_\beta$ of $\mathscr{U}$ 
which covers $K_\beta$.
Then $\mathscr{U}$ has a star-finite functionally open refinement.
}

\begin{proof} 
Let $\mathscr{U}, \mathscr{K}, \mathscr{U}_\beta$ be as in the lemma, and let 
$\mathscr{U}_\beta = \{U_{\beta_n} : n < \omega\}$.
Then $K_\beta = \bigcup\{U_{\beta_n}\cap K_\beta: n < \omega\}$.
Since the closed subspace $K_\beta$ of $X$ is countable shrinkable, there exists a countable functionally  open cover 
$\mathscr{G}=\{G_n: n<\omega\}$ of $K_\beta$ such that 
$\overline{G}_n \subset U_{\beta,n}$ for each $n < \omega$.
By Theorem \ref{th:2.1.3}, $\mathscr{G}$ has a countable star-finite functionally 
open refinement $\mathscr{L}_\beta$ in $K_\beta$.
For each $L \in \mathscr{L}_\beta$ there exists a continuous function $f_L : K_\beta \to [0,1]$ such that $L = f_L^{-1}(0,1]$.
Define a continuous function $g_L : X \to [0,1]$ by
\[ g_L(x) = 
\begin{cases} 
	f_L(x) & \text{if } x \in K_\beta, \\ 
	0 & \text{if } x \in X - K_\beta 
\end{cases} \]
Then $L = g_L^{-1}(0,1]$, so it is a functionally open subset of $X$.
Thus $\mathscr{L} = \bigcup_{\beta \in \Gamma} \mathscr{L}_\beta$ is a star-finite functionally open refinement of $\mathscr{U}$.
\end{proof}
