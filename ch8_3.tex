\section{Strongly Paracompact and Ultraparacompact Spaces} \label{sec:8.3}


{\theorem{\label{th:8.3.1}} 
For any \(\lambda\) the following are equivalent:

\textup{(i)} \(X\) is \(\lambda\)-ultraparacompact.

\textup{(ii)} Every open \(\lambda\)-cover of \(X\) has a discrete closed refinement.

\textup{(iii)}  Every open \(\lambda\)-cover of \(X\) has a disjoint open refinement.

\textup{(iii)}  Every open \(\lambda\)-cover of \(X\) has a disjoint functionally open refinement.

\textup{(iv)} Every open \(\lambda\)-cover of \(X\) has a disjoint semi-open refinement.
}

\begin{proof}
(i) \(\Rightarrow\) (ii) \(\Rightarrow\) (iii)  \(\Rightarrow\)  (v) 
and (iv) \(\Rightarrow\) (iii).  Clear.

(ii) \(\Rightarrow\) (iv) Assume (ii). By theorem \ref{th:8.1.5},
$X$ is $\lambda$-collectionwise normal.
Let \(\mathscr{U}= \{U_\alpha: \alpha < \lambda\}\) be an open \(\lambda\)-cover of \(X\). 
Let \(\mathscr{F}= \{F_\alpha: \alpha < \lambda\}\) be a discrete closed cover such that 
$F_\alpha\subset U_\alpha$ for all $\alpha < \lambda$.
Let  $\mathscr{H}= \{H_\alpha: \alpha < \lambda\}$ be a disjoint functionally open 
expansion of \(\mathscr{F}\) such that $F_\alpha \subset H_\alpha \subset U_\alpha$.
Then $\mathscr{H}$ is a refinement of $\mathscr{U}$.

(v) \(\Rightarrow\) (iii). Suppose \(\mathscr{U}\) is an open \(\lambda\)-cover of \(X\). 
By (v), let \(\mathscr{V} = \{V_\beta: \beta\in \Gamma\}\) be a disjoint semi-open refinemnt 
of \(\mathscr{U}\). To see that each $V_\beta$ is open, let $x\in V_\beta$. Then
$\operatorname{st}(x, \mathscr{V}) = V_\alpha$. 
Hence $x\in \operatorname{st}(x, \mathscr{V}) ^\circ\subset V_\beta^\circ$.
Then $V_\beta = V_\beta^\circ$ is open.
\end{proof} 

{\theorem{\label{th:8.3.2}} 
 For any $X$ the following are equivalent:

\textup{(i)}\(X\) is \(\lambda\)-ultraparacompact.

\textup{(ii)} \(X\) is countably ultraparacompact and $\lambda$-screenable.

\textup{(iii)} \(X\) is countably ultraparacompact and $\lambda$-fully normal.

\textup{(iv)}  \(X\) is countably ultraparacompact and every open \(\lambda\)-cover 
of \(X\) has a  cushioned refinement.

\textup{(v)}  \(X\) is countably ultraparacompact, $\lambda$-collectionwise normal
and $\lambda$-$D\omega^2$-refinable.
}

\begin{proof}
(i) \(\Rightarrow\) (ii) and (i) \(\Rightarrow\) (iv) Clear.

(ii) \(\Rightarrow\) (iii). Assume (ii), then $X$ is countably shrinkable. By
Theorem \ref{th:8.2.1}, $X$ is $\lambda$-fully normal.

(iii) \(\Rightarrow\) (v). Assume (iii). By Corollary \ref{cor:8.1.6}, $X$ is  
$\lambda$-collectionwise normal. Since $X$ is $\lambda$-paracompact, it is 
$D\omega^2$-refinable.

(iv) \(\Rightarrow\) (v). Assume (iv). By Theorem \ref{th:8.1.5}, $X$ is  
$\lambda$-collectionwise normal. By Theorem \ref{th:4.1.1}, $X$ is $\lambda$-subparacompacr, so it is $\lambda$-$D\omega^2$-refinabel.

(v) \(\Rightarrow\) (i) Assume (v), and let 
\(\mathscr{U}=\{U_\alpha: \alpha < \lambda\}\) be an open cover of \(X\). 
Then $\mathscr{U}$ is a $D\omega^2$-cover.
By Theorem \ref{th:8.1.5}, $\mathscr{U}$ is a normal cover.
By Theorem \ref{th:2.1.6}, $\mathscr{U}$ has a refinement
$\mathscr{V}=\{\bigcup_{n<\omega}\mathscr{V}_n\}$ where each
$\mathscr{V}_n=\{V_{n, \alpha}:\alpha < \omega\}$ is a discrete open
with $V_{n, \alpha}\subset U_\alpha$ for all $\alpha$.
Let $\{W_n: n<\omega\}$ be a discrete open cover of $X$ 
such that $W_n=\bigcup\mathscr{V}_n$ for each $n<\omega$. For $\alpha < \lambda$,
let $G_\alpha = \bigcup_{n < \omega}(W_n\cap V_{n,\alpha})$. 
Then $\{G_\alpha:\alpha < \lambda\}$ is a disjoint open refinement of $\mathscr{U}$,
so $X$ is  $\lambda$-ultraparacompact.
\end{proof}

{\lemma{\label{lem:8.3.3}} 
Let $\mathscr{E}$ be a family in $X$.

\textup{(i)} 	If $\mathscr{E}$ is connected, then for any $D\in\mathscr{E}$,
	\[
	\mathscr{E}=\bigcup_{n\geq1}P_n(D),
	\]
	and for every $n\geq1$,
	\[
	P_{n+1}(D)
	=
	\bigcup\{(\mathscr{E})_E: E\in P_n(D)\}.
	\]
	
\textup{(ii)} 
	If $\mathscr{E}$ is connected and star-countable, then
	$\mathscr{E}$ is countable.
}

\begin{proof}
(i) By a routine operation.

(ii) Suppose $\mathscr{E}$ is connected and star-countable. Then for
each $E\in\mathscr{E}$, $(\mathscr{E})_E$ is countable.
If $D\in\mathscr{E}$, then
$P_1(D)=\{D\}, P_2(D)=(\mathscr{E})_{D}$
are countable. If $P_n(D)$ is countable, then by (i),
$P_{n+1}(D)=\bigcup\{(\mathscr{E})_E: E\in P_n(D)\}$
is countable. Thus each $P_n(D)$ is countable, then each
$P_n(D)$ is is countable, so $\mathscr{E} = \bigcup_{n\ge 1}P_n(D)$
is countable.
\end{proof}

8.3.4. Lemma. Let \(\mathscr{U}\) be a star-countable open cover of \(X\). Then \(\{C(U,\mathscr{U}): U\in\mathscr{U}\}\) is a disjoint open cover of \(X\) such that each \(C(U,\mathscr{U})\) is a open-closed set in \(X\) and \(C(U,\mathscr{U})\) is covered by a countable subcollection of \(\mathscr{U}\).

Proof. It is clear that \(\{C(U,\mathscr{U}): U\in\mathscr{U}\}\) is a disjoint open cover of \(X\). Since \(X\setminus C(U,\mathscr{U})=\bigcup\{C(V,\mathscr{U}): V\in\mathscr{U}, C(V,\mathscr{U})\cap C(U,\mathscr{U})=\emptyset\}\) is open, \(C(U,\mathscr{U})\) is closed. For \(U\in\mathscr{U}\), let \(K_0(U)=\{U\}\), \(K_n(U)=\{V\in\mathscr{U}: V\cap(\bigcup K_{n-1}(U))\neq\emptyset\}\) for \(n\ge 1\). Since \(\mathscr{U}\) is star-countable, \(K_n(U)\) is countable for each \(n\in\omega\). Thus, \(\mathscr{U}(U)=\bigcup_{n\in\omega}K_n(U)\) is a countable subcollection of \(\mathscr{U}\). It is easy to see that \(C(U,\mathscr{U})=\bigcup\mathscr{U}(U)\). \(\square\)

8.3.12. Theorem. For any \(\lambda\) the following are equivalent:
