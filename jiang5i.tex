\section{Normal Covers and $D\omega^2$-refinable Spaces}
\label{sec:5.1}

{\theorem{\label{th:5.1.1}}
For any open cover $\mathscr{U}$ of $X$, the following are equivalent.
	
\textup{(i)}  $\mathscr{U}$ is a normal cover.

\textup{(ii)} $\mathscr{U}$ has a $\sigma$-disjoint functionally open refinement $\mathscr{V} = \bigcup_{n<\omega} \mathscr{V}_n$ such that each $\bigcup \mathscr{V}_n$ is functionally open.

\textup{(iii)}  $\mathscr{U}$ has a $\sigma$-star finite functionally open refinement 
$\mathscr{V} = \bigcup_{n<\omega} \mathscr{V}_n$ such that each $\bigcup \mathscr{V}_n$ is functionally open.
}

\begin{proof}
(i) $\Rightarrow$ (ii). Suppose $\mathscr{U}$ is normal. By Theorem \ref{th:2.1.6}, $\mathscr{U}$ has a functionally open refinement $\mathscr{V} = \bigcup_{n<\omega} \mathscr{V}_n$ where each $\mathscr{V}_n$ is discrete, so is disjoint, and by Lemma \ref{lem:1.3.8}, $\bigcup \mathscr{V}_n$ is functionally open.

(ii) $\Rightarrow$ (iii). Trivial.

(iii) $\Rightarrow$ (i). Suppose $\mathscr{U}$ is an open cover of $X$ and 
$\mathscr{V} = \bigcup_{n<\omega} \mathscr{V}_n$ is a functionally open refinement 
of $\mathscr{U}$ such that each 
$\mathscr{V}_n=\{V_{n,\alpha}: \alpha\in\Delta_n\}$ is star-finite and 
$\bigcup \mathscr{V}_n$ is functionally open. By Theorem \ref{th:2.1.3}, 
$\mathscr{V}' = \{\bigcup \mathscr{V}_n: {n<\omega}\}$ has a star-finite 
functionally open refinement $\mathscr{L}$. Since $\mathscr{L}$ is locally finite, 
$\mathscr{V}'$ is a normal cover of $X$. Let $\mathscr{W}$ 
be a normal cover of \(X\) such that 
\(\overline{\mathscr{W}}\) refines \(\mathscr{V}'\). 
By Lemma \ref{lem:1.1.5}, 
there exists a locally finite functionally open cover
\(\mathscr{H}=\{H_n: n < \omega\) such that 
\(\overline{H_n}\subset\cup\mathscr{V}_n\) for each \(n<\omega\).
For each \(n<\omega\), let \(\mathscr{G}_n=\{H_n\cap V_{n,\alpha}: \alpha\in\Delta_n\}\) 
where each set \(H_n\cap V_{n,\alpha}\) is
functionally open. To see that \(\mathscr{G}_n\) is locally finite, let \(x\in X\). If
\(x\notin\cup \mathscr{V}_n\), then 
\(X\setminus\overline{H}_n\in\operatorname{top}(x)\) 
which intersects no member of \(\mathscr{G}_n\). 
If \(x\in V_{n,\beta}\) for some \(\beta\in\Delta_n\), then the set
\(\{\alpha\in\Delta_n: V_{n,\beta}\cap (H_n\cap V_{n,\alpha})\neq\emptyset\}\) 
is finite because \(\mathscr{V}_n\) is
star-finite. Hence each \(\mathscr{G}_n\) is locally finite. 
By Theorem \ref{th:2.1.3}, 
there exists a countable locally finite functionally open cover 
\(\mathscr{S}=\{S_n: n<\omega\}\)
such that \(S_n\subset\bigcup\mathscr{G}_n \forall \, n\). 
Let
$$\mathscr{R}=\{S_n\cap H_n\cap V_{n,\alpha}:  \alpha\in\Delta_n, n < \omega\}.$$

\noindent Then \(\mathscr{R}\) is a functionally open refinement of \(\mathscr{U}\). To see
that \(\mathscr{R}\) is locally finite, let \(x\in X\). Since \(\mathscr{S}\) is locally finite, 
\(\exists \, G_0\in\operatorname{top}(x)\) such that 
$\operatorname{ind}(\mathscr{S})_{G_0} = \{n_1, \cdots, n_k\}$
 is finite. For each \(i\le k\), \(\mathscr{S}_{n_i}\)
is locally finite, so \(\exists G_i\in\operatorname{top}(x)\) such that the set
$$D_i=\{\alpha\in\Delta_{n_i}: G_i\cap H_{n_i}\cap V_{n_i,\alpha}\neq\emptyset\}$$

\noindent is finite. Let \(G=\bigcap_{i=0}^k G_i\), and let
\[C=\{(\alpha, n):  \alpha\in \Delta_n, n < \omega,  
G\cap(S_{n}\cap H_n\cap V_{n,\alpha})\neq\emptyset\}.\]
\noindent It is easy to see that \(C\subset\bigcup_{i=1}^k (D_i\times\{n_i\})\). Thus \(C\) is finite,
so \(\mathscr{R}\) is locally finite. By Theorem \ref{th:2.1.6}, \(\mathscr{U}\) is a normal cover.
\end{proof}

{\theorem{\label{th:5.1.2}}
 For any open cover \(\mathscr{U}\) of \(X\) the following
are equivalent:

\textup{(i)} \(\mathscr{U}\) is a normal cover.

\textup{(ii)} There exists a sequence $\{\mathscr{E}_n: n < \omega\}$ of semi-open covers of $X$ 
such that $\mathscr{E}_0$ refines $\mathscr{U}$ and each $\mathscr{E}_{n+1}$ is 
a star refinement of $\mathscr{E}_n$.

\textup{(iii)} There exists a sequence $\{\mathscr{E}_n: n < \omega\}$ of semi-open covers of $X$ 
such that $\mathscr{E}_0$ refines $\mathscr{U}$ and each $\mathscr{E}_{n+1}$ is 
a point-star refinement of $\mathscr{E}_n$.
}

\begin{proof}
 (i) $\Rightarrow$ (ii) $\Rightarrow$ (iii). Clear

(iii) $\Rightarrow$ (i). Assume (iii). For each $n < \omega$, let 
$$\mathscr{U}_n = \{\operatorname{st}(x, \mathscr{E})^{\circ}: x \in X\}.$$
Then each $\mathscr{U}_n$ is an open cover of $X$. To see that $\mathscr{U}_0$ refines $\mathscr{U}$, let $x \in X$. There exists $E \in \mathscr{E}_0$ and 
$U \in \mathscr{U}$ such that $\operatorname{st}(x, \mathscr{E}_1) \subset E \subset U$. 
Then $\operatorname{st}(x, \mathscr{E}_1)^{\circ} \subset U$, 
so $\mathscr{U}_0$ refines $\mathscr{U}$.
To see that $\mathscr{U}_{n+1}$ point-star refines $\mathscr{U}_n$, let $x \in X$. 
It suffices to prove that 
$\mathrm{st}(x, \mathscr{U}_{n+1}) \subset \mathrm{st}(x, \mathscr{E}_{n+1})^{\circ}$. 
Suppose $y \in \mathrm{st}(x, \mathscr{U}_{n+1})$, then 
$x, y \in \operatorname{st}(u, \mathscr{E}_{n+2})^{\circ}$ 
for some $u \in X$. Let $E \in \mathscr{E}_{n+1}$ be such that 
$\mathrm{st}(u, \mathscr{E}_{n+2}) \subset E$. Then $x, y \in E^\circ$, so 
$y \in E^{\circ} \subset \mathrm{st}(x, \mathscr{E}_{n+1})^{\circ}$. 
This completes the proof of 
$\mathrm{st}(x, \mathscr{U}_{n+1}) \subset \mathrm{st}(x, \mathscr{E}_{n+1})^{\circ}$. 
Hence $\mathscr{U}_{n+1}$ point-star refines $\mathscr{U}_n$ and  $\mathscr{U}$ is normal.
\end{proof}


{\definition{} \label{def:5.1.3}
\textup{(i)} Let $<$ be the anti-lexicographic order on $\omega \times \omega$, 
$$(p, q) < (m, n) \text{ iff } (q < n) \text{ or } (q = n \text{ and } p < m).$$

\textup{(ii)} An open cover $\mathscr{U}$ of $X$ is a 
\uwave{$D\omega^2$-cover}
\index{$D\omega^2$-cover| \S\ref{def:5.1.3}\hfill}
if it has a refinement 
$\mathscr{V} = \bigcup_{m,n<\omega} \mathscr{V}_{m,n}$ such that for each $m, n < \omega$, 
the family 
$$\mathscr{V}_{m,n}|_{(
		X\setminus \bigcup\{\cup\mathscr{V}_{(p,q)}: (p,q)<(m,n)\}
		)}$$
is relatively discrete and closed in the subspace 
$X\setminus \bigcup\{\cup\mathscr{V}_{(p,q)}: (p,q)<(m,n)\}$.
The family $\mathscr{V}$ is called a
\uwave{$D\omega^2$-refinement}
\index{$D\omega^2$-refinement| \S\ref{def:5.1.3}\hfill}
of $\mathscr{U}$.

\textup{(iii)} $X$ is 
\uwave{$\lambda$-$D\omega^2$-refinable}
\index{$\lambda$-$D\omega^2$-refinable| \S\ref{def:5.1.3}\hfill}
\index{$D\omega^2$-refinable| \S\ref{def:5.1.3}\hfill}
if every open $\lambda$-cover of $X$ is a  $D_\omega^2$-cover.
}

{\lemma{}  \label{lem:5.1.4}
An open cover $\mathscr{U}= \{U_\alpha: \alpha < \Delta\}$ of $X$ is a 
$D\omega^2$-cover iff $\mathscr{U}$ has a refinement 
$\mathscr{F} = \bigcup_{m,n<\omega} \mathscr{F}_{m,n}$ such that for each 
$m, n < \omega$,
$\mathscr{F}_{m,n} = \{F_{m,n,\alpha}: \alpha \in \Delta\}$ is a discrete family of 
closed subsets of the subspace 
$X \setminus \bigcup\{\cup \mathscr{F}_{(p,q)}: (p, p)<(m,n)\}$ and 
$F_{m,n,\alpha}\subset U_\alpha$ for each $\alpha\in \Delta$. 
}

\begin{proof}
$(\Rightarrow)$ Clear.

$(\Leftarrow)$ Suppose that $\mathscr{U} =\{ U_\alpha: \alpha\in\Delta\}$ is an open cover 
of $X$, 
and that $\mathscr{V} = \bigcup_{m,n<\omega} \mathscr{V}_{m,n}$ 
is a $D\omega^2$-refinement of $\mathscr{U}$ with 
$\mathscr{V}_{m,n} = \{V_{m,n,\gamma}: \gamma \in \Gamma_{m,n}\}$. 
For each $\gamma\in \Gamma_{m,n}$, pick $\delta(\gamma) \in \Delta$ 
such that $\mathscr{V}_{m,n,\gamma} \subset U_{\delta(\gamma)}$. 
For each $\alpha$, let 
$$F_{m,n,\alpha} = V_{m,n,\alpha} \setminus \bigcup_{(p,q)<(m,n)} (\bigcup \mathscr{V}_{p,q}).$$ 
Then $\mathscr{F}_{m,n}$ covers... 
Let $\mathscr{F}_{m,n} = \{F_{m,n,\alpha}:  \alpha \in \Gamma_{m,n}\}$ 
and $\mathscr{F} = \bigcup_{m,n<\omega} \mathscr{F}_{m,n}$.

 If $(p, q) \neq (m, n)$, then...

Since $\mathscr{V}$ is a $D_\omega$-refinement of..., it follows from... and... that each $\mathscr{F}_{m,n}$ is a discrete family of closed subsets of $X \setminus \bigcup\{\bigcup \mathscr{V}_{p,q}: (p,q) < (m,n)\}$.
\end{proof}

{\theorem{} \label{th:5.1.5}
A $\lambda$-weakly $\overline{\theta}$-refinable space is a $D\omega^2$-refinable.
}

\begin{proof}
Suppose that $X$ is a weakly $\overline{\theta}$-refinable 
and that $\mathscr{U}$ is an open $\lambda$-cover of $X$. 
Let $\mathscr{G} = \bigcup_{n<\omega} \mathscr{G}_n$ where each 
$\mathscr{G}_n = \{G_{n,\alpha}: \alpha \in \Gamma_n\}$ is a weak 
$\overline{\theta}$-refinement of $\mathscr{U}$.
 We show that $\mathscr{U}$ is a $D\omega^2$-cover. 
 Define an one-to-one function 
 $\varphi: \omega \times \omega \to \omega$ by 
 $\varphi(k, l) = \frac{1}{2}[(k+l)^2 +3k+l]$.
For $m < \omega$, there exists a unique pair $(k, l)$ such that $\varphi(k, l) = m$.
Let
\begin{gather*}
	D_k^0 = \emptyset \\
	D_k^l = \{x\in X: 0 < |\{\alpha\in \Gamma_k: x\in G_{k, \alpha}\}| \le l\} \text { for } l \ge 1 \\
	H_0 = \emptyset \\
	H_n = \{x\in X: |\{i \in \omega: x\in \cup\mathscr{G}_i\}| \le n\} \text { for } n \ge 1 
\end{gather*}
For $B\in [\Gamma_k]^l, S\in[\omega]^n$, let
\begin{gather*}
	G(\emptyset, k, 0) = \emptyset \\
	G(B, k, l) = \cap_{\alpha\in B} G_{k, \alpha} \text{ for } n \ge 1 \\
	L(\emptyset, 0) = \emptyset \\
	L(S, n) = \cap (\cup \mathscr{G}_i), \text{ for } n \ge 1, \\
	P(, S, m, n) = G(B, k, l) \cap D_k^l \cap L(S, n) \cap H_n, \\
	\mathscr{P}_{m, n} = \{P(B, S, m, n): B\in [\Gamma_k]^l, 
	   S\in[\omega]^n, \text{ and } \varphi(k, l) = m\}.
\end{gather*}
We show that $\mathscr{P} = \bigcup_{m,n<\omega} \mathscr{P}_{m,n}$ is a 
$D\omega^2$-refinement of $\mathscr{U}$.
 Since $\mathscr{P}$ is a weak $\overline{\theta}$-refinement of $\mathscr{U}$, 
 $\mathscr{P}$ is a refinement of $\mathscr{U}$.
To show $\mathscr{P}$ is a $D\omega^2$-refinement, let $m, n < \omega$ with $\varphi(k, l) = m$.
Let 
$Y = X \setminus \bigcup\{\cup \mathscr{P}_{p,q}: (p, q) < (m, n)\}$. 
It suffices to show that $\mathscr{P}_{m,n}|_Y$ is discrete and closed in $Y$. 
By Lemma \ref{lem:1.1.5}, we need only to show the following:

\textsc{Claim 1.} $\cup(\mathscr{P}_{m,n}|_Y)$ is closed in Y, and

\textsc{Claim 2.}  $\mathscr{P}_{m,n}|_Y$ is discrete at each point $x \in \mathscr{P}_{m,n}|_Y)$.

\noindent To show Claim 1, first we show that $G = Y \setminus \bigcup (\mathscr{P}_{m,n}|_Y)$ 
is open in $Y$. Let $x \in G$. We have 
\begin{gather}
	x \in X \setminus \bigcup \{\cup \mathscr{P}_{p,q}: (p,q) < (m,n)\}. \label{eq:5.1.5.1}
\end{gather}

\noindent Since $\mathscr{G} = \bigcup_{n<\omega} \mathscr{G}_n$ is a weak $\overline{\theta}$-refinement, there exist $r, j < \omega$ such that 
\begin{gather*}
	|\{\alpha\in \Gamma_r: x\in G_{r, \alpha}\}| = i > 0, \\
	|\{t\in \omega: x\in \cup\mathscr{G}_t\}| = j. 
\end{gather*}

\noindent Suppose $h = \varphi(r, i)$. Take $B \in [\Gamma_r]^i$ and $S \in [\omega]^j$ 
such that 
$$x \in G(B, \Gamma, i) \cap D_r^l \cap L(S, j) \cap H_j = P(B, S, h, j)\subset
\cup\mathscr{P}_{h, j}.$$
If $j < r$, then $(h, j) < (m, n)$.
It follows that 
$x\in \cup\{\cup\mathscr{P}_{p. q}: (p, q) < (m, n)\}\subset X\setminus Y$, a contradiction. 
Hence $j \ge n$.

\textsc{Case 1}. Suppose $j > n$. Then $x\notin H_n$ and $W = Y \setminus H_n$ 
is a neighborhood of $x$ in $Y$ such that 
$W \cap (\cup \mathscr{P}_{m, n})\subset (X\setminus H_n)\cap H_n = \emptyset$,
so $W\subset G$.

\textsc{Case 2}. Suppose $j = n$. There exists $S_0\in[\omega]^n$ such that 
\begin{gather}
x\in L(S_0, n) \cap H_n. \label{eq:5.1.5.2}
\end{gather}

\textsc{Case 2}a. 
If $x \notin \cup\mathscr{G}_k$, then $k \notin S_0$. The set 
$W = Y\cap L(S_0, n)$ is a neighborhood of $x$ in $Y$ such that 
for each $B\subset [\Gamma_r]^l$, $S_0\in [\omega]^r$,
\begin{gather*}
	W\cap P(B, S_0, m, n) \subset\cap_{i\in S_0}((\cup \mathscr{G}_i))\cap
	(\cup \mathscr{G}_h) \cap H_n = \emptyset.
\end{gather*}
Then $W\subset G$.

\textsc{Case 2}b.
If $x\in \mathscr{G}_k$, then $A_{k} = \{\alpha\in \Gamma_k: x\in G_{k, \alpha}\} \ne \emptyset$. 
We have 
\begin{gather}
	\vert{}A_k\vert{} > l. \label{eq:5.1.5.3}
\end{gather}
If $A_k$ is infinite, then this holds. If $A_k$ is finite, suppose $\vert{}A_k\vert{} = q > 0$. 
Then there exists $B_1 \in [\Gamma_k]^q$ such that 
$x \in G(B_1, k, q)\cap D_k^q$. Let $p = \varphi(k, q)$. If follows from
\eqref{eq:5.1.5.2} that $x\in P(B, S_0, p, n) \subset \cup\mathscr{P}_{p, n}$.
If $q \le l$, then $p =\varphi(k, q) \le \varphi(k, l) = m$, so $(p, q) \le (m, n)$. 
But $x \in \bigcup \mathscr{P}_{p,q}$, which contradicts to \eqref{eq:5.1.5.1}. 
Hence $q > l$ and the \eqref{eq:5.1.5.3} holds.

The set $W = Y\cap G(B_1, k, q)$ is a neighborhood of $x$ in $Y$.
For each $B\in [\Gamma_k]^l, S\in [\omega]^n$, 
since $B_1  = q > l$, 
$W \cap P(B, S, m, n) \subset \bigcap_{\alpha \in B_1} G_{k,\alpha} \cap D_k^l = \emptyset$. 
This completes the proof of \textsc{Claim 1}.


To show Claim \textsc{Claim 2}, let $x \in \bigcup (\mathscr{P}_{m,n}|_Y)$. 
There exists $B_0 \in [\Gamma_k]^l$ and $S_0\in [\omega]^n$ 
such that $x \in P(B_0, S_0, m, n)$.
The set $W = Y\cap G(B_0, k, l) \cap L(S_0, n)$ is a neighborhood of $x$ in $Y$.
Let  $B\in [\Gamma_k]^l$ and $S\in [\omega]^n$ with $(B, S) \neq (B_0, S_0)$.
If $B \neq B_0$, then 
$$W \cap P(\beta, S, m, n) \subset G(B_0, k, l) \cap G(B, k, l) \cap D_k^l = \emptyset.$$
If $S \neq S_0$, then 
$$W \cap P(B, S, m, n) \subset L(S_0, n) \cap L(S,n)\cap H_n = \emptyset.$$
Thus, $\mathscr{P}_{m,n}|_Y$ is discrete at $x$. We completed the proof of \textsc{Claim 2}.
\end{proof}
