\section{Shrinkable and Normal Spaces} \label{sec:5-2}

{\definition{\label{def:5.2.1}}
Recall Definition \ref{def:1.1.2}. 
Let $\mathscr{U} = \{U_\alpha: \alpha \in \Delta\}$ and $\mathscr{B} = \{B_\alpha: \alpha \in \Delta\}$ 
be covers of $X$. $\mathscr{B}$ is a 
\uwave{shrink}\index{shrink| \S\ref{def:5.2.1}\hfill} 
of $\mathscr{U}$ if $\overline{B_\alpha} \subset U_\alpha$ for each $\alpha\in \Delta$.

$X$ is 
\uwave{$\lambda$-shrinkable}\index{$\lambda$-shrinkable| \S\ref{def:5.2.1}\hfill} 
\index{shrinkable| \S\ref{def:5.2.1}\hfill} 
 every open $\lambda$-cover of $X$ has an open shrink.
}

{\lemma{\label{lem:5.2.2}}
\textup{(i)} A countably shrinkable space is normal and countable paracompact.

\textup{(ii)} For any $X$ the following are equivalent.

$\quad$ \textup{(a)} $X$ is $\lambda$-shrinkable.

$\quad$ \textup{(b)} Every open $\lambda$-cover of $X$ has a closed shrink.

$\quad$ \textup{(c)} Every open $\lambda$-cover of $X$ has a functionally open shrink.
}

\begin{proof}
(i) Suppose $X$ is countably shrinkable. Then $X$ is normal. By Theorem \ref{th:3.3.2}, 
$X$ is countably paracompact.

(ii) (a) $\Rightarrow$ (b) and (c) $\Rightarrow $(a). Clear.

(b) $\Rightarrow$ (c).
Assume (b). It is easy to show that $X$ is normal. Let 
$\mathscr{U} = \{U_\alpha: \alpha \in \Lambda\}$ be an open cover of $X$, and 
let $\{F_\alpha: \alpha \in \Lambda\}$ be a closed cover of $X$ 
such that $F_\alpha \subset U_\alpha$ for each $\alpha \in \Lambda$. 
Let $H_\alpha$ be a functionally open set such that 
$F_\alpha \subset H_\alpha \subset \overline{H_\alpha} \subset U_\alpha$. 
Then $\mathscr{H} = \{H_\alpha\}_{\alpha \in \Lambda}$ be a functionally open 
shrink of $\mathscr{U}$.
\end{proof}

{\definition{} \label{def:5.2.3}
\textup{(i)}  An open cover \(\mathscr{U} = \{ U_\alpha : \alpha \in \Delta \}\) of $X$ is a 
\uwave{$C\omega^2$-cover}\index{$C\omega^2$-cover| \S\ref{def:5.2.3}\hfill} 
if \(\mathscr{U}\) has a refinement \(\mathscr{F} = \bigcup_{m,n<\omega} \mathscr{F}_{m,n}\) 
where each \(\mathscr{F}_{m,n} = \{ F_{m,n,\alpha} : \alpha \in \Delta \}\) is a family of 
closed subsets of the subspace  \(X\setminus \cup \{\bigcup \mathscr{F}_{p, q}: (p, q) < (m,n) \}\)
and $F_{m, n, \alpha} \subset U_\alpha$ for all \(\alpha \in \Delta\).
The family \(\mathscr{F}\) is called a 
\uwave{$C\omega^2$-refinement}\index{$C\omega^2$-refinement| \S\ref{def:5.2.3}\hfill} 
of \(\mathscr{U}\).
\(X\) is 
\uwave{$\lambda$-$C\omega^2$-refinable}
\index{$\lambda$-$C\omega^2$-refinable| \S\ref{def:5.2.3}\hfill} 
\index{$C\omega^2$-refinable| \S\ref{def:5.2.3}\hfill} 
if every open \(\lambda\)-cover of \(X\) is a $C\omega^2$-cover.

\textup{(ii)}  An open cover \(\mathscr{U}\) is a 
\uwave{$C^*\omega^2$-cover}
\index{$C^*\omega^2$-cover| \S\ref{def:5.2.3}\hfill} 
 if \(\mathscr{U}\) has a $C\omega^2$-refinement 
 \(\mathscr{F} = \bigcup_{m,n<\omega} \mathscr{F}_{m,n}\) 
 such that each \(\bigcup \mathscr{F}_{m,n} \) is a closed subset of 
 $X\setminus \bigcup\{\mathscr{F}_{p, q}: (p, q) < (m, n)\}$.
}

{\theorem{} \label{the:5.2.4}
\(X\) is $\lambda$-shrinkable iff \(X\) is normal and $\lambda$-$C\omega^2$-refinable.
}

\begin{proof}
\((\Rightarrow)\) Suppose \(X\) is $\lambda$-shrinkable, then \(X\) is normal. 
Let \(\mathscr{U} = \{ U_\alpha : \alpha \in \Delta \}\) be an open cover of \(X\) and 
\(\mathscr{F}\) be a closed shrink of \(\mathscr{U}\). Let $\mathscr{F}_{0,0} = \mathscr{F}$ and 
 for each $(m, n) > (0,0), \mathscr{F}_{m,n}=\{\emptyset\}$.
 Then \(\bigcup_{m,n<\omega} \mathscr{F}_{m,n}\) is a \(C\omega^2\)-refinement of 
 \(\mathscr{U}\), so \(X\) is \(C\omega^2\)-refinable.
	
\((\Leftarrow)\) Suppose \(X\) is normal and \(C\omega^2\)-refinable. 
We first establish the following:
	
\textsc{Claim 1}. For any open \(\lambda\)-cover \(\mathscr{U} = \dots\) of \(X\), there exists a sequence \(\{\mathscr{V}_{m,n} \}_{m,n<\omega}\) of \(\lambda\)-families of open 
subsets of $X$, where each \(\mathscr{V}_{m,n} = \{ V_{m,n,\alpha} : \alpha < \lambda\}\) 
such that \(\overline{V}_{m,n,\alpha} \subset U_\alpha\) for all \(\alpha < \lambda\) and 
\(\mathscr{V} = \bigcup_{m,n<\omega} \mathscr{V}_{m,n}\) covers \(X\).[cite: 2]

	We use induction on the well-ordered set \(\omega \times \omega\) (\(<\))[cite: 2]. Let \(\mathscr{U} = \{ U_\alpha : \alpha < \kappa \}\) be an open \(\kappa\)-cover of \(X\) and let \(\mathscr{F} = \bigcup_{m,n<\omega} \mathscr{F}_{m,n}\) be a \(\sigma\)-Caz refinement of \(\mathscr{U}\), where each \(\mathscr{F}_{m,n} = \{ F_{m,n,\alpha} : \alpha < \kappa \}\) is a family of closed subsets of the subspace \(X \setminus \bigcup_{\langle p,q \rangle < \langle m,n \rangle} U^{\langle p,q \rangle}\) with \(F_{m,n,\alpha} \subset U_\alpha\) for all \(\alpha < \kappa\)[cite: 2].
	
	Let \(\mathscr{F}_{0,0} = \{\emptyset\}\) and let \(\mathscr{V}_{0,0} = \{\emptyset\}\)[cite: 2]. Let \(\langle m,n \rangle \in \omega \times \omega\)[cite: 2]. Assume that the families \(\mathscr{V}_{p,q} = \{ V_{p,q,\alpha} : \alpha < \kappa \}\) of open sets which satisfy the conditions[cite: 2]:
	\begin{enumerate}
		\item[(1)] \(\overline{V_{p,q,\alpha}} \subset U_\alpha\) for all \(\alpha < \kappa\)[cite: 2].
		\item[(2)] \(\bigcup_{\langle i,j \rangle \le \langle p,q \rangle} (\bigcup \mathscr{F}_{i,j}) \subset \bigcup_{\langle i,j \rangle \le \langle p,q \rangle} (\bigcup \mathscr{V}_{i,j})\)[cite: 2]
	\end{enumerate}
	have been constructed for all \(\langle p,q \rangle < \langle m,n \rangle\)[cite: 2]. Then we have
	\[
	\bigcup_{\langle p,q \rangle < \langle m,n \rangle} (\bigcup \mathscr{F}_{p,q}) \subset \bigcup_{\langle p,q \rangle < \langle m,n \rangle} U^{\langle p,q \rangle}. \text{[cite: 2]}
	\]
	
	Let \(L = \bigcup_{\langle p,q \rangle < \langle m,n \rangle} (\bigcup \mathscr{V}_{p,q})\)[cite: 2]. Since each \(F_{m,n,\alpha} \setminus L\) is a closed subset of \(X\), there exists an open set \(V_{m,n,\alpha}\) of \(X\) such that[cite: 2]
	\[
	F_{m,n,\alpha} \setminus L \subset V_{m,n,\alpha} \subset \overline{V}_{m,n,\alpha} \subset U_\alpha. \text{[cite: 2]}
	\]
	Let \(\mathscr{V}_{m,n} = \{ V_{m,n,\alpha} : \alpha < \kappa \}\)[cite: 2]. Then
	\[
	\bigcup_{\langle p,q \rangle \le \langle m,n \rangle} (\bigcup \mathscr{F}_{p,q}) \subset \bigcup_{\langle p,q \rangle \le \langle m,n \rangle} (\bigcup \mathscr{V}_{p,q}). \text{[cite: 2]}
	\]
	Hence the open families \(\mathscr{V}_{m,n}\) satisfy the conditions (1) and (2)[cite: 2]. By (2), \(\mathscr{V}\) covers \(X\)[cite: 2]. Claim 1 is true[cite: 2].

		Let \(A\) and \(B\) be disjoint closed sets in \(X\).[cite: 4] Then the open cover \(\{ X \setminus A, X \setminus B \}\) has an open refinement \(\mathscr{V} = \{ V_{n,0}, V_{n,1} : n < \omega \}\) such that \(\overline{V}_{n,0} \subset X \setminus B\) and \(\overline{V}_{n,1} \subset X \setminus A\) for each \(n < \omega\).[cite: 4]

{\theorem{} \label{the:5.2.7}
For any space \(X\), the following are equivalent[cite: 5]:
\begin{enumerate}
	\item[(i)] \(X\) is \(\kappa\)-shrinkable[cite: 5].
	\item[(ii)] Every open \(\kappa\)-cover \(\mathscr{U}\) of \(X\) has a functionally open refinement \(\mathscr{V} = \{ V_{n,\alpha} : n < \omega, \alpha < \kappa \}\) such that \(\overline{V}_{n,\alpha} \subset U_\alpha\)[cite: 5].
	\item[(iii)] Every open \(\kappa\)-cover \(\mathscr{U}\) of \(X\) has an open refinement \(\mathscr{V} = \{ V_{n,\alpha} : n < \omega, \alpha < \kappa \}\) such that \(\overline{V}_{n,\alpha} \subset U_\alpha\)[cite: 5].
	\item[(iv)] \(X\) is normal and every open \(\kappa\)-cover \(\mathscr{U}\) of \(X\) has a closed refinement \(\mathscr{B} = \{ B_{n,\alpha} : n < \omega, \alpha < \kappa \}\) such that \(B_{n,\alpha} \subset U_\alpha\) for each \(n, \alpha\)[cite: 5].
	\end{enumerate}
}

		
		\begin{proof}
			(i) \(\Rightarrow\) (ii). By Lemma 5.2.2[cite: 5].
			
			(ii) \(\Rightarrow\) (iii). Clear[cite: 5].
			
			(iii) \(\Rightarrow\) (iv). By Lemma 5.2.6[cite: 5].
			
			(iv) \(\Rightarrow\) (i).
			
			Assume (iv)[cite: 5]. By Theorem 5.2.4, it suffices to prove that \(X\) is \(\sigma\)-Caz refinable[cite: 5]. Let \(\mathscr{U} = \{ U_\alpha : \alpha < \kappa \}\) be an open cover of \(X\)[cite: 5]. Let \(\{ B_{n,\alpha} : n < \omega, \alpha < \kappa \}\) be a closed cover of \(X\) such that \(B_{n,\alpha} \subset U_\alpha\) for each \(n < \omega, \alpha < \kappa\)[cite: 5].
			
			For \(n < \omega\), let \(\mathscr{B}_n = \{ B_{n,\alpha} : \alpha < \kappa \}\)[cite: 5].
			
			\textbf{Claim.} For each \(m < \omega\), there exists a family \(\mathscr{F}_{m,0} = \{ F_{m,0,\alpha} : \alpha < \kappa \}\) such that[cite: 5]
			\begin{enumerate}
				\item[(1)] \(F_{m,0,\alpha} \subset U_\alpha\) for each \(\alpha < \kappa\)[cite: 5],
				\item[(2)] \(\mathscr{F}_{m,0}\) is a family of closed subsets of \(X\), and[cite: 5]
				\item[(3)] \(\bigcup_{i \le m} (\bigcup \mathscr{F}_{i,0}) = \bigcup_{i \le m} (\bigcup \mathscr{B}_i)\)[cite: 5].
				\end{enumerate}
\end{proof}
