\section{Paracompact and Expandable Spaces} \label{sec:7.2}



{\definition{\label{def:7.2.1}} 
Let $X$ be a space and $\mathscr{U}, \mathscr{V}$ be covers of $X$.

\textup{(i)} $\mathscr{V}$ is a 
\uwave{local $W$-refinement}
\index{local $W$-refinement| \S\ref{def:7.2.1}\hfill}
of $\mathscr{U}$ at $x \in X$ if there exists $G \in \operatorname{top}(x)$ and 
$F\in[\Delta]^{<\omega}$ such that $(\mathscr{V})_G$ refines $\{U_\alpha: \alpha \in F\}$.

\textup{(ii)} $\mathscr{V}$ is a 
\uwave{local-star $\overset{*}{F}$-refinement}
\index{local-star $\overset{*}{F}$-refinement| \S\ref{def:7.2.1}} 
of $\mathscr{U}$ at $x$ if there exists  $F\in[\Delta]^{<\omega}$ such that

\quad \textup{(a)} $\operatorname{st}(x, \mathscr{V}) \subset 
\operatorname{st}(x, \{U_\alpha: \alpha\in F\})$,

\quad \textup{(b)} $\exists \, W \in \operatorname{top}(x) \, 
\forall \, G\in \operatorname{top}(x)$
$$G\subset W \Rightarrow \operatorname{st}(G, \mathscr{V}) \subset 
\operatorname{st}(G, \{U_\alpha: \alpha\in F\}).$$

\textup{(iii)} $\mathscr{V}$ is a 
\uwave{local\, $W$-refinement}
 $($ or \uwave{local-star $\overset{*}{F}$-refinement}$)$ of $\mathscr{U}$ 
 if $\mathscr{V}$ is a local $W$-refinement $($or local-star refinement$)$ of $\mathscr{U}$ 
 at each $x \in X$.

A sequence of covers $\{\mathscr{V}_n: n < \omega\}$ is a 
\uwave{local $W$-refining sequence}
\index{local $W$-refining sequence| \S\ref{def:7.2.1}\hfill}
$($or 
\uwave{local-star $\overset{*}{F}$-refining sequence}$)$ 
\index{local-star $\overset{*}{F}$-refining sequence| \S\ref{def:7.2.1}\hfill}
for $\mathscr{U}$ 
if for each $x \in X$ there exists $m < \omega$ such that $\mathscr{V}_m$ is a 
local $W$-refinement $($or local-star$\overset{*}{F}$-refinement$)$ of $\mathscr{U}$ at $x$.
}


{\lemma{} \label{lem:7.2.2}
Let $\mathscr{U}, \mathscr{V}$ be covers of $X$.

\textup{(i)} If $\mathscr{V}$ is a local $W$-refinement of $\mathscr{U}$ at $x$, then $\mathscr{V}$ is a local-star $\overset{*}{F}$-refinement of $\mathscr{U}$ at $x$.


\textup{(ii)} If $\mathscr{V}$ is a local-star $\overset{*}{F}$-refinement of $\mathscr{U}$ at $x$, then $\mathscr{V}$ is a point-star $\dot{F}$-refinement of $\mathscr{U}$ at $x$.
}

\begin{proof}
Let $\mathscr{U}=\{U_\alpha: \alpha \in \Delta\}$.

(i)  Assume that $\mathscr{V}$ is a local $W$-refinement of 
$\mathscr{U}$ at $x$. there exists $W \in \operatorname{top}(x)$ and 
$F\in [\Delta]^{<\omega}$ such that $(\mathscr{V})_W$ refines $\{U_\alpha: \alpha \in F\}$. Then 
$\operatorname{st}(x, \mathscr{V}) \subset \operatorname{st}(x, \{U_\alpha: \alpha \in F\})$. 
If $G \in \operatorname{top}(x)$ with $G \subset W$, then 
$\operatorname{st}(G, \mathscr{V}) \subset \operatorname{st}(G, \{U_\alpha: \alpha \in F\})$, 
 so $\mathscr{V}$ is a local-star $\overset{*}{F}$-refinement of $\mathscr{U}$ at $x$.

(ii) If $\mathscr{V}$ is a local-star $\overset{*}{F}$-refinement of $\mathscr{U}$ at $x$, 
then there exists $F\in [\Delta]^{<\omega}$ such that 
$\operatorname{st}(x, \mathscr{V}) \subset \operatorname{st}(x, \{U_\alpha: \alpha \in F\})$. 
For each 
$y \in \operatorname{st}(x, \mathscr{V})$, there exists $\alpha(y)\in F$ such that 
$x, y\in U_{\alpha(y)}$. Let $F' = \{\alpha(y) : y \in \operatorname{st}(x, \mathscr{V})\}$. 
Then $F' \subset F$, $x \in \bigcap_{\beta\in F'} U_\beta$ and 
$\operatorname{st}(x, \mathscr{V})\subset \bigcup_{\beta\in F'}U_\beta$. 
Hence $\mathscr{V}$ is a point-star $\dot{F}$-refinement 
of $\mathscr{U}$ at $x$.
\end{proof}

{\theorem{} \label{th:7.2.3}
For any $X$, the following are equivalent:

\textup{(i)} $X$ is $\lambda$-expandable.

\textup{(ii)}  Every $A\omega^2$ $\lambda$-cover of $X$ has a locally finite open refinement.

\textup{(iii)} Every $A\omega^2$ $\lambda$-cover of $X$ has an interior-preserving 
open local-star $\overset{*}{F}$-refinement.


\textup{(iv)} Every directed $A\omega^2$ $\lambda$-cover of $X$ has 
an interior-preserving open local-star refinement.


\textup{(v)}  Every interior-preserving directed $A\omega^2$ 
$\lambda$-cover of $X$ has a semi-open local-star refinement.


\textup{(vi)}  Every directed $A\omega^2$ $\lambda$-cover of $X$ 
has a closure-preserving closed refinement $\mathscr{F}$ such that 
$\mathscr{F}^\circ$ covers $X$.


\textup{(vii)}  Every interior-preserving directed $A\omega^2$ 
$\lambda$-cover of $X$ has a cushioned open refinement.
}

\begin{proof}
(i) $\Rightarrow$ (ii): By Lemma \ref{lem:7.1.1}.

(ii) $\Rightarrow$ (iii) $\Rightarrow$ (iv) $\Rightarrow$ (v) and (vi) $\Rightarrow$ (vii). Clear. 

(v) $\Rightarrow$ (vii). By Theorem \ref{th:4.1.9}.

(iv) $\Rightarrow$ (vi). By Lemma \ref{lem:4.1.6}.

(vii) $\Rightarrow$ (i). Suppose $\mathscr{B}$ is a locally finite $\lambda$-family of subsets 
of $X$. The induced cover $\mathscr{B}^*$ is an interior-preserving directed $A\omega^2$ 
$\lambda$-cover of $X$. By (vii), $\mathscr{B}^*$ has a cushioned open refinement. 
By Lemma \ref{lem:1.2.3}, $\mathscr{B}$ has a locally finite open expansion, so $X$ is $\lambda$-expandable.
\end{proof}

{\corollary{} \label{cor:7.2.4}
Any $\lambda$-paracompact space is $\lambda$-expandable. \hfill\qedsymbol
}

{\theorem{} \label{th:7.2.5}
For any $X$, the following are equivalent:

\textup{(i)}  $X$ is $\lambda$-expandable.


\textup{(ii)}  Every directed $A\omega^2$ $\lambda$-cover of $X$ has a $\sigma$-closure-preserving 
closed refinement $\mathscr{F}$ such that $\mathscr{F}^\circ$ covers $X$.


\textup{(iii)}  Every interior-preserving directed $A\omega^2$ $\lambda$-cover of $X$ has a $\sigma$-closure-preserving closed refinement $\mathscr{F}$ such that 
$\mathscr{F}^\circ$ covers $X$.

\textup{(iv)} Every interior-preserving directed $A\omega^2$ $\lambda$-cover of $X$ has a $\sigma$-cushioned open refinement.
}

\begin{proof}
(i) $\Rightarrow$ (ii). By Theorem \ref{th:7.2.3}.

(ii) $\Rightarrow$ (iii)  $\Rightarrow$ (iv). Clear.

(ii) $\Rightarrow$ (i).  Assume (iv) and let $\mathscr{B}$ be a locally finite $\lambda$-family
of subsets of  $X$. The induced cover $\mathscr{B}^*$ is an interior-preserving directed 
$A\omega^2$ $\lambda$-cover of  $X$. Then $\mathscr{B}^* $ has a 
$\sigma$-cushioned open refinement. By Lemma \ref{lem:1.2.3}, $\mathscr{B}$ 
has $\theta^*$-sequence of open expansions. Then 
$X$ is $\lambda$-$\theta^*$-expandable. by Theorem \ref{th:3.3.4},  
$X$ is $\lambda$-expandable.
\end{proof}

{\theorem{} \label{th:7.2.6}
For any  $X$ the following are equivalent:

\textup{(i)}  $X$ is $\lambda$-expandable.

\textup{(ii)}   Every $A\omega^2$ $\lambda$-cover of  $X$ has a 
$\theta^*$-sequence of open refinements.

\textup{(iii)}   Every $A\omega^2$ $\lambda$-cover of  $X$ has a local $W$-refining sequence by interior-preserving open covers of  $X$.

\textup{(iv)}   Every $A\omega^2$ $\lambda$-cover of  $X$ has 
a local-star $\overset{*}{F}$-refining sequence by  interior-preserving open covers.

\textup{(v)}   Every interior-preserving directed $A\omega^2$ 
$\lambda$-cover of  $X$ has a local-star refining sequence by semi-open covers.
}

\begin{proof}
 (i) $\Rightarrow$ (ii) and (i) $\Rightarrow$ (iii). By Theorem \ref{th:8.2.5}. % TO_DO

(ii) $\Rightarrow$ (v) and (iii) $\Rightarrow$ (iv) $\Rightarrow$ (v). Clear.


(v) $\Rightarrow$ (i)  Assume (v) and let $\mathscr{U} = \{ U_\alpha: \alpha < \lambda\}$ be an 
interior-preserving directed $A\omega^2$ $\lambda$-cover of  $X$. 
Then $\mathscr{U}$ has a local-star refining sequence of semi-open covers. 
By Theorem \ref{th:4.1.9}, $\mathscr{U}$ has a $\sigma$-cushioned open refinement. 
By Theorem \ref{th:7.2.5},  $X$ is $\lambda$-expandable.
\end{proof}


{\theorem{} \label{th:7.2.7}
\textup{([Jiang 1987a]\textsuperscript{\cite{Jiang1987a}})}
$X$ is $\lambda$-paracompact iff  $X$ is $\lambda$-expandable and
 $\lambda$-$A\omega^2$-refinable.
}

\begin{proof}
($\Rightarrow$) By Corollary \ref{cor:8.2.6} % TO_DO (8.2.6) and 

($\Leftarrow$) By Theorem \ref{th:8.2.5}. % TO_DO
\end{proof}

{\corollary{} \label{cor:7.2.8}
\textup{(i)}  $X$ is $\lambda$-paracompact iff  $X$ is $\lambda$-expandable  
and $\lambda$-submetacompact.

\textup{(ii)}  \textup{([Krajewski 1971]\textsuperscript{\cite{Krajewski}})}.   
$X$ is paracompact iff  $X$ is expandable and submetacompact. \hfill\qedsymbol
}


{\theorem{} \label{th:7.2.9}
\textup{([Jiang 1987b]\textsuperscript{\cite{Jiang1987b}})}.   
$X$ is $\lambda$-paracompact iff  $X$ is $\lambda$-$\theta^*$-refinable.
}

\begin{proof}
$(\Rightarrow)$ Clear. 

$(\Leftarrow)$ Suppose  $X$ is $\sigma$-expandable. Suppose  $X$ is $\sigma$-$\mathscr{A}\omega$-refinable. By Corollary 7.2.9,  $X$ is $\sigma$-expandable. Since  $X$ is $\sigma$-submetacompact, by Corollary 8.2.12,  $X$ is $\sigma$-paracompact.
\end{proof}
