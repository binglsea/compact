\section{Submetacompact and $\theta$-expandable Spaces} \label{sec:6.1}

In order to characterize expandable spaces, [Katuta 1975]\textsuperscript{\cite{Katuta1975}}
introduced the $A$-cover. An open cover $\mathscr{U}$ of a space $X$ is an $A$-cover if it has a 
locally finite refinement. We now generalize this notion as follows.

{\definition{\label{def:6.1.1}} 
\textup{(i)} An open cover $\mathscr{U} = \{U_\alpha: \alpha \in \Delta\}$ of $X$ 
is an 
\uwave{$A\omega$-cover}
\index{$A\omega$-cover| \S\ref{def:6.1.1}\hfill}
if it has a refinement 
$\mathscr{B} = \bigcup_{n<\omega} \mathscr{B}_n$ 
where each $\mathscr{B}_n = \{B_{n,\alpha}: alpha \in \Delta\}$ 
is a relatively locally finite family of subsets of the subspace 
$X \setminus \bigcup_{i<n} (\bigcup \mathscr{B}_i)$ with 
$B_{n,\alpha} \subset U_\alpha$ for each $\alpha \in \Lambda$ 
and  
$\bigcup_{i<n} (\bigcup \mathscr{B}_i)$ is closed in $X$ for each $n < \omega$.

\textup{(ii)} An open cover $\mathscr{U}$ is an 
\uwave{$A\omega^2$-cover}
\index{$A\omega^2$-cover| \S\ref{def:6.1.1}\hfill}
if it has a refinement 
$\mathscr{A} = \bigcup_{m,n<\omega} \mathscr{A}_{m,n}$ 
where each $\mathscr{A}_{m,n}$ is a locally finite family of subsets of the  subspace 
$X \setminus \bigcup\{\bigcup \mathscr{A}_{p,q}: (p,q)< (m,n)\}$
where < is a well ordering on 
$\omega\times\omega$ as in Definition \ref{def:6.1.3}.

\textup{(iii)} A space $X$ is 
\uwave{$\lambda$-$A\omega$-refinable}
\index{$\lambda$-$A\omega$-refinable| \S\ref{def:6.1.1}\hfill}
$($\uwave{$\lambda$-$A\omega^2$-refinable}$)$
\index{$\lambda$-$A\omega^2$-refinable| \S\ref{def:6.1.1}\hfill}
if every open  $\lambda$-cover of the space is an $A\omega$-cover
$(A\omega^2$-cover$)$).
}

{\theorem{} \label{th:6.1.2}
For any $X$ the following are equivalent.

\textup{(i)} $X$ is $\lambda$-$\theta$-expandable $($or $\lambda$-discretely $\theta$-expandable$)$. 

\textup{(ii)} Every $A\omega$ $($or $D\omega)$ $\lambda$-cover of $X$ has
a $\theta$-sequence of open refinements.

\textup{(iii)} Every directed $A\omega$ $($or $D\omega)$ $\lambda$-cover of $X$ has
a $\sigma$-closure-preserving closed refinements.

\textup{(iv)} Every interior-preserving directed $A\omega$ $($or $D\omega)$ $\lambda$-cover of 
$X$ has a $\sigma$-cushioned refinements.
}

\begin{proof}
(i) $\Rightarrow$  (ii). Assume (i) and let $\mathscr{U} = \{U_\alpha: \alpha < \lambda\}$
be an $A\omega$-cover  (or $D\omega$-cover) of $X$. Then $\mathscr{U}$
 has a refinement $\mathscr{B}=\bigcup_{n<\omega}\mathscr{B}_n$ where each $\mathscr{B}_n=\{B_{n,\alpha}: \alpha\in \lambda\}$ is a
locally finite (or discrete closed) family of subsets of the
subspace $X\setminus \bigcup_{i<n}(\bigcup\mathscr{B}_i)$ 
with ${B}_{n,\alpha}\subset U_{\alpha}$ for each $n < \omega, \alpha\in \lambda$,
and $\bigcup_{i<n}(\bigcup\mathscr{B}_i)$ is a closed subset of $X$ for each $n < \omega$.
We have the following

\textsc{Claim 1}. For every $n<\omega$ and every $(k_0, \cdots, k_n)\in\omega^{n+1}$,
there exists a family $\mathscr{V}(k_0, \cdots, k_n) = \{V(k_0, \cdots, k_n,\alpha): \alpha<\lambda\}$
of open subsets of $X$ such that

(1)  $V(k_0, \cdots, k_n,\alpha)\subset U_\alpha$ for each $\alpha< \lambda$;

(2) $\bigcup_{i=0}^n (\bigcup\mathscr{B}_i)\subset
 \bigcup_{i=0}^n (\bigcup\mathscr{V}(k_0, \cdots, k_i))$;

(3) For $n \ge 1, (\bigcup\mathscr{V}(k_0, \cdots, k_n)) \bigcap 
(\bigcup_{i=0}^n (\bigcup\mathscr{B}_i))
= \emptyset$; and 

(4) If $n = 0$, then for each $x\in X$, there exists $k_0(x) < \omega$
such that $\text{ind}(\mathscr{V}(k_0(x))_x$ is finite.
If $n > 0$, then for each $x\in X$ and each $(k_0, \cdots, k_{n-1})\in\omega^{n}$
there exists $k_n(x) <\omega$ such that 
$\operatorname{ind}(\mathscr{V}(k_0, \cdots, k_{n-1}, k_n(x)))_x$ is finite.

\textsc{Proof of Claim 1}. We use induction on $n < \omega$. 
Let $\mathscr{B}_0 = \{B_{0, \alpha}: \alpha < \lambda\}$ be a locally finite (or discrete closed) 
family in $X$ with $B_{0, \alpha} \subset U_\alpha$. 
Then $\mathscr{B}_0$ has a $\theta$-sequence 
$\{\mathscr{V}(k_0): k_0 < \omega\}$ of open expansions where each 
$\mathscr{V}(k_0) = \{V(k_0, \alpha): \alpha < \lambda\}$ such that 
$B_{0, \alpha} \subset V(k_0, \alpha) \subset U_\alpha$. Then
$\mathscr{B}_0 \subset \bigcup \mathscr{V}(k_0)$. 
For every $x\in X$, there exists $k_0(x) < \omega$ such that 
$\operatorname{ind}(\mathscr{V}(k_0(x)))_x$
 is finite. Hence  $\mathscr{V}(k_0) $ satisfies Conditions (1), (2) and (4). 
 
 Let $n \ge 1$. Suppose that for each $i \le n$ and each $(k_0, \dots, k_i)\in\omega^{i+1}$, the family 
 $\mathscr{V}(k_0, \cdots, k_i) = \{V(k_0, \cdots, k_i, \alpha): \alpha < \lambda\}$ 
 has been constructed such that Conditions (1) -- (4) are satisfied. 
 We now define $\mathscr{V}(k_0, \dots, k_{n+1})$ for any 
 $(k_0, \dots, k_{n+1}) \in \omega^{n+2}$. 
 Let $L(k_0, \dots, k_n)=\bigcup_{i\le n}(\bigcup \mathscr{V}(k_0, \dots, k_i))$. 
 By inductive hypothesis, we have 
 $\bigcup_{i\le n}(\bigcup \mathscr{B}_i)\subset L(k_0, \dots, k_n)$. 
 Hence, $\mathscr{B}_{n+1}|_{(X\setminus L(k_0,\cdots, k_n))}$ 
 is a locally finite (or discrete closed) $\lambda$-family in $X$ such that 
 $B_{n+1, \alpha}\setminus  L(k_0,\cdots, k_n)\subset 
 U_\alpha \setminus \bigcup_{i\le n}(\bigcup \mathscr{B}_i)$.
 There exists a $\theta$-sequence 
 $\{\mathscr{V}(k_0, \dots, k_n, k_{n+1}): n+1 < \omega\}$ of open expansions,
 where each 
 $\mathscr{V}(k_0, \dots, k_n, k_{n+1}) = 
 \{V(k_0, \dots, k_n, k_{n+1}, \alpha): \alpha<\lambda\}$
 such that for each $\alpha < \lambda$,
 $$B_{n+1, \alpha}\setminus  L(k_0,\cdots, k_n)\subset 
 V(k_0,\cdots, k_n, k_{n+1}, \alpha) \subset
 U_\alpha \setminus  \bigcup_{i\le n}\left(\bigcup \mathscr{B}_i\right).$$
Then 
$$\left(\bigcup \mathscr{V}(k_0, \dots, k_n, k_{n+1})\right) \bigcap 
\left(\bigcup_{i\le n}\left(\bigcup \mathscr{B}_i\right)\right) = \emptyset.$$
Since $\bigcup \mathscr{B}_{n+1} \setminus L(k_0, \dots, k_n, k_{n+1}) \subset
\bigcup \mathscr{V}(k_0, \dots, k_n, k_{n+1})$,
\begin{equation*}
\begin{aligned}
 \bigcup_{i\le n+1}\left(\bigcup \mathscr{B}_{i}\right) &\subset L(k_0, \dots, k_n) \bigcup 
                    \left(\bigcup \mathscr{B}_{n+1}\setminus L(k_0, \dots, k_n, k_{n+1})\right) \\
                    &\subset \bigcup_{i\le n+1}\left(\bigcup \mathscr{V}(k_0, \dots, k_i)\right).
\end{aligned}
\end{equation*}


\end{proof}





Proof continuation: We use induction on $n$. For each $n \ge 1$ and each index, the family has been constructed such that the required conditions are satisfied. Let us define the open expansions and sequences such that for each step, the properties hold true and the resulting refinement satisfies the necessary covering properties.
