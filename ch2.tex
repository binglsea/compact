\section{Normal Covers}\label{sec:ch2_normal_covers}

{\definition{} 
A function $\rho : X \times X \to [0, \infty)$ is a 
\uwave{pseudo-metric}\index{pseudo-metric, \S\ref{sec:ch2_normal_covers}}
 on $X$ if for all $x, y, z \in X$, we have $\rho(x, x) = 0$, $\rho(x, y) = \rho(y, x)$, 
 and $\rho(x, z) \le \rho(x, y) + \rho(y, z)$. A pseudo-metric $\rho$ is a 
\uwave{metric}\index{metric, \S\ref{sec:ch2_normal_covers}} 
 if $\rho(x, y) = 0$ implies $x = y$. 
 The topology $\tau_\rho$ on $X$ generated by a pseudo-metric $\rho$ is 
 the topology which has the family $\mathcal{B} = \{B_\rho(x, r) : x \in X, r > 0\}$ 
 of balls as a base, where $B_\rho(x, r) = \{y \in X : \rho(x, y) < r\}$.
}

Tukey [1940]\cite{Tukey} proved that every metric space is fully normal.

{\theorem. Every pseudo-metric space is fully normal.
}

\vspace{10pt}
Proof. Let $(X, \rho)$ be a pseudo-metric space. Let $\mathcal{U}$ be an open cover of $X$. For each $x \in X$, pick some $U_x \in \mathcal{U}$ and 
an integer $m(x)\ge 1$ such that $x \in U_x$ and

\begin{equation}
	\mathclap{B_\rho\left(x, \frac{3}{m(x)}\right) \subset U_x\in \mathcal{U}.} \tag*{(1)}
\end{equation}

Let $\mathcal{V}$ be the sets of smaller balls $B_\rho(x, \frac{1}{m(x)})$
for all  $x \in X$ and its chosen $m(x)$ satisfying Condition (1).
Then $\mathcal{V}$ is an open refinement of $\mathcal{U}$. 
We will show that it is also a point-star refinement of $\mathcal{U}$, 
so that $X$ is fully normal. 
Let $x \in X$, then there exits at least one $u \in X$ such that
$x\in B_\rho\left(u, \frac{1}{m(u)}\right)\in\mathcal{V}$
since $\mathcal{V}$ covers $X$.
So, we are able to define
$$k(x) = \min\{m(u) : u\in X, x\in B_\rho\left(u, \frac{1}{m(u)}\right)\}.$$
and pick $u_x\in X$ such that $k(x) = m(u_x)$. Now, we have

\begin{equation}
	\mathclap{x\in B_\rho\left(u_x, \frac{1}{m(u_x)}\right)\in \mathcal{V}.} \tag*{(2)}
\end{equation}

It suffices to show that 

\begin{equation}
	\mathclap{\mathrm{st}(x, \mathcal{V}) \subset U_{u_x}.} \tag*{(3)}
\end{equation}

By (1), it suffices to show
\begin{equation}
	\mathclap{\mathrm{st}(x, \mathcal{V}) \subset B_\rho(u_x, \frac{3}{m(u_x)}).} \tag*{(4)}
\end{equation}

For if $y \in \mathrm{st}(x, \mathcal{V})$, then exits $t\in X$ 
such that $y, x \in B_\rho\left(t, \frac{1}{m(t)}\right)$.
Since $m(u_x)=k(x) \le m(t)$, we have

$$\rho(y, u_x) \le \rho(y, t) + \rho(t, x) + \rho(x, u_x) < \frac{3}{m(u_x)}.$$
Thus (4) holds. 
\qedsymbol

{\theorem{} \textup{(Dowker-Morita)} A countable functionally open cover of $X$ has a countable star-finite functionally open refinement.
}

\vspace{8pt}
Proof. Let $\mathcal{U} = \{U_n : n \ge 1\}$ be a functionally open cover of $X$, where each $U_n = f_n^{-1}(0, 1]$ for some continuous function $f_n : X \to [0, 1]$. Define $f : X \to [0, 1]$ by

$$f(x) = \sum_{n=1}^{\infty} f_n(x) / 2^n, x\in X$$

a continuous function.

For $n \ge 1$, let $G_n = f^{-1}\left(\frac{1}{2^n}, 1\right]$ and $F_n = f^{-1}\left[\frac{1}{2^n}, 1\right]$.

Here is the transcription of the manuscript excerpt from `jiang20.pdf`:

---

It is easy to check that $A_{n+1} \subset \bigcap_{k=1}^{n} U_k$ for each $n \ge 1$.

For $k \ge 1, 1 \le j \le k$, let

$$W_{n,k} = U_k \cap (G_{n+1} - F_{n-1}), \quad F_{-1} = \phi.$$

$W = \bigcup_{k=1}^{\infty} W_{n,k}$. To see that $W$ covers $X$, let $x \in X$. Since $\{G_n\}$ covers $X$, let $m$ be the smallest integer such that $x \in F_m$. Since $F_m \subset G_{m+1} \subset \bigcup_{k \le m} U_k$, there is $k \le m$ such that $x \in U_k \cap (G_{m+1} - F_{m-1}) = W_{m,k}$.

Hence $W$ is a countable functionally open refinement of $\mathcal{U}$.

To see that $W$ is star-finite, fix $k \ge 1$, $1 \le j \le k$. For any $m \ge 1$, $1 \le i \le m$, if $m \ge k + 2$, then $W_{k,j} \subset G_{k+1} \subset F_{k+1} \subset F_{m-1}$. Then $W_{k,j} \cap W_{m,i} = \emptyset$.

Hence if $W_{m,i} \cap W_{k,j} \neq \emptyset$, then $m < k+2$.

It follows that $\mathrm{St}(W_{k,j}, W) \subset \{W_{m,i} : m < k+2, 1 \le i \le m\}$, so $W$ is star-finite.

$$\square$$
