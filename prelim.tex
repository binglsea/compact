\chapter{Preliminaries}

\section{Basic	Covering	Properties}  
\label{sec:prelim_basic_prop}

We use the  usual terms and notations\cite{Alexandroff1924}. 
An \uwave{ordinal}\index{ordinal, \S\ref{sec:prelim_basic_prop}} 
is the set of all smaller ordinals, 
and a \uwave{cardinal}\index{cardinal, \S\ref{sec:prelim_basic_prop}} 
 is an initial ordinal. 
We use $\kappa, \mu$ for cardinals; $\lambda$
for an infinite cardinal;  $\alpha, \beta, \gamma, \theta$ for ordinal numbers. $\omega = \{0, 1, 2, \dots\}$ is the smallest infinite cardinal. 
$|A|$ denotes the \uwave{cardinality}\index{cardinal, \S\ref{sec:prelim_basic_prop}} 
 of a set $A$.
The \uwave{empty set}\index{empty set, \S\ref{sec:prelim_basic_prop}}  $\emptyset$ is the unique set $B$ such that $|B| = 0$.
$\mathbb{R}$ denotes the  
\uwave{real line}\index{real line, \S\ref{sec:prelim_basic_prop}},
and $[0, 1]$ is the   
\uwave{closed unit interval}\index{closed unit interval, \S\ref{sec:prelim_basic_prop}}  in $\mathbb{R}$.

For any sets $X, Y$, we denote 
$\mathcal{P}(X) = \{A : A \subset{X}\}$; 
$Y^X= \{f: f$  is a function from $X$  into $Y$\}. 
If $f \in Y^X$, we write $f: X \to Y$.

If $f: X \to Y$, $A \subset X$ and $B \subset Y$, then $f|_A: A \to Y$
 is defined by $f|_A(x)) = f(x)$ for each $x \in A$; 
 $f[A] = \{f(x): x \in A\}$. $f^{-1}B = f^{-1}[B] = \{x \in X: f(x) \in B\}$.

For a set $A$ and $n < \omega$,
let $[A]^{n} = \{ B \subset A: | B| = n \}$,
$[A]^{< \omega} = \bigcup \{ [A]^{n} : n < \omega \}$. 
Let $ \omega^0 = \{ \emptyset \}$. For  $n = 1, \dots, n$, let
$\omega^n = \{ f : f: n \to \omega \} = \{ ( k_{0}, k_{1}, \dots, k_{n-1} ) :  k_i \in\omega, i < n\} $.
If $h < \omega$, let $\emptyset\oplus h = (h)$, 
$(k_0, \cdots, k_{n-1})\oplus h = (k_0, \cdots, k_{n-1}, h)$.	
Let $\omega^{<\omega} = \bigcup_{n<\omega}\omega^{n}$.
	
Let  $\mathcal{U} = \{ U_\alpha\in \Delta\} $ be any family of subsets of  $X$. 
Without loss of generality we may assume the indexing set  $ \Delta $ is an ordinal or a set of ordinals. We call
cardinality $|\Delta|$ the 
 \uwave{power}\index{power, \S\ref{sec:prelim_basic_prop}}
of $\mathcal{U}$. A \uwave{$\lambda$-family} is a family with power $\le \lambda$. $\mathcal{U}$ is a \uwave{cover} of $X$ if $\bigcup \mathcal{U} = X$. A \uwave{$\lambda$-cover} is a cover with power $\le \lambda$.

Let $\mathcal{U} = \{U_\alpha : \alpha \in \Delta\}$, $A \subset X$, $x \in X$. We denote
$$
	\mathcal{U}|_A = \{U_\alpha \cap A : \alpha \in \Delta\}.$$
$$
	(\mathcal{U})_A = \{U_\alpha : \alpha \in \Delta, 
	\ U_\alpha \cap A \neq \emptyset\}; \quad (\mathcal{U})_x = \{U_\alpha : \alpha \in \Delta, \ x \in U_\alpha\}.$$
$$	
   \text{st}(A, \mathcal{U}) = \bigcup (\mathcal{U})_A, 
   \quad \text{st}(x, \mathcal{U}) = \bigcup (\mathcal{U})_x.$$
$$
   \text{ind}(\mathcal{U})_A = \{\alpha \in \Delta : 
   U_\alpha \cap A \neq \emptyset\}, 
  \text{ind}(\mathcal{U})_x = \{\alpha \in \Delta : x \in U_\alpha\}.$$

By a \uwave{space} \index{space, \S\ref{sec:prelim_basic_prop}} 
we mean a topological space without any of the separation axioms $T_i$, $i=0,1,2$ at all.

Throughout the following, $X$ denotes a space. For $A \subset X$, $\overline{A}$ denotes the 
\uwave{closure}\index{closure, \S\ref{sec:prelim_basic_prop}}  
of $A$ and $A^\circ$ the 
\uwave{interior}\index{interior, \S\ref{sec:prelim_basic_prop}} 
 of $A$. For $x \in X$, let
$$
	\text{top}(x) = \{G : G \text{ is an open subset of } X 
	\text{ with } x \in G\}.$$
	
A \uwave{normal}\index{normal, \S\ref{sec:prelim_basic_prop}} 
or \uwave{regular}\index{regular, \S\ref{sec:prelim_basic_prop}} 
space needs not be $T_1$.	

{\definition{}
	(i) A family $\mathcal{U} = \{ U_\alpha\in \Delta\}$  is
	\uwave{directed}\index{directed, \S\ref{sec:prelim_basic_prop}} 
	if for every $\alpha, \beta \in \Delta$, 
	$U_\alpha\cup U_\beta	\subset U_\gamma$ for some $\gamma\in\Delta$. 
	
	(ii) A family $\mathcal{U} = \{ U_\alpha: \alpha < \lambda\}$  is
   \uwave{increasing}\index{increasing, \S\ref{sec:prelim_basic_prop}} 
   if $U_\alpha\subset U_\beta$
   whenever $\alpha < \beta <\lambda$, 

} %end of defintion


	For any family $\mathcal{U} = \{ U_\alpha\in \Delta\}$, the family
	$\mathcal{U}^F = \{U_S: S\in[\Delta]^{<\omega}\}$, 
	where $U_S = \bigcup_{\alpha\in S}U_\alpha$, is directed, and 
	$\mathcal{U}$ is directed iff for each $S\in [\Delta]^{<\omega}$
	there is some $\alpha\in\Delta$ such that $U_S\subset U_\alpha$.

{\definition{} Let
	$\mathcal{U} = \{ U_\alpha\in \Delta\}$ and 
	$\mathcal{G} = \{ G_\beta\in \Gamma\}$
	be families of subsets of $X$.
	
	(i) $\mathcal{G}$ 
	   \uwave{refines}\index{refines, \S\ref{sec:prelim_basic_prop}} 
	   $\mathcal{U}$ or $\mathcal{G}$ is a 
	    \uwave{partial refinement}\index{partial refinement, \S\ref{sec:prelim_basic_prop}} 
	    of $\mathcal{U}$ if for every $\beta\in \Gamma$
	    there is some $\alpha\in \Delta$ such that 
	    $G_\beta\subset U_\alpha$.
  $\mathcal{G}$ is a 
  \uwave{refinement}\index{refinement, \S\ref{sec:prelim_basic_prop}} 
  of $\mathcal{U}$ if $\mathcal{G}$ is a partial refinement of 
  $\mathcal{U}$ and $\bigcup\mathcal{G} = \bigcup\mathcal{U}$.

	    Let $\Gamma = \Delta$. $\mathcal{U}$ is an 
	    	    \uwave{expansion}\index{expansion, \S\ref{sec:prelim_basic_prop}} of $\mathcal{G}$
	    	    if $G_\alpha \subset U_\alpha$ for all $\alpha\in \Delta$.
	   $\mathcal{G}$ is an 
	   \uwave{shrink}\index{shrink, \S\ref{sec:prelim_basic_prop}} of $\mathcal{U}$
	   if $\overline{G_\alpha} \subset U_\alpha$ for all $\alpha\in \Delta$.
	    
	(ii) $\mathcal{G}$ is a 
	\uwave{cushioned}\index{cushioned, \S\ref{sec:prelim_basic_prop}} 
	in $\mathcal{U}$ or
	$\mathcal{G}$ is a 
	\uwave{partial cushioned refinement}\index{cushioned, \S\ref{sec:prelim_basic_prop}} 
	of $\mathcal{U}$
	if for every $\beta\in\Gamma$ there is $\delta(\beta)\in\Delta$ such that for any $B\subset\Gamma$, $\overline{\bigcup_{\beta\in B}G_\beta} \subset \bigcup_{\beta\in B}U_{\delta()\beta)}$. 
	$\mathcal{G}$ is a 
	\uwave{cushioned refinement}\index{cushioned refinement, \S\ref{sec:prelim_basic_prop}}
	if $\mathcal{G}$ is a partial cushioned refinement
	of $\mathcal{U}$ and $\bigcup\mathcal{G} = \bigcup\mathcal{U} $.
} %end of defintion


{\definition{} Let  $\mathcal{V}_n, n<\omega$,  
	$\mathcal{U}=\{ U_\alpha: \alpha \in \Delta\}$ be covers of  $X$.
	
(i)  $ \mathcal{U}$ is  
  \uwave{disjoint}\index{disjoint, \S\ref{sec:prelim_basic_prop}}
    $($\uwave{point-finite}$)$\index{point-finite, \S\ref{sec:prelim_basic_prop}}
     at  $ x \in X$ if $|\operatorname{ind}(\mathcal{U})_x|\leq 1$ 
     $(|\operatorname{ind}(\mathcal{U})_x| < \infty)$.

(ii) $\mathcal{U}$ is  
  \uwave{discrete}\index{discrete, \S\ref{sec:prelim_basic_prop}}
  $($\uwave{\text{locally finite}}\index{locally finite, \S\ref{sec:prelim_basic_prop}}$)$ at $x$ 
  if there is some $G\in\operatorname{top}(x)$ such that
  $|\operatorname{ind}(\mathcal{U})_G| \leq 1 $ 
  $(|\operatorname{ind}(\mathcal{U})_G|  < \omega).$

(iii) $\mathcal{U}$ is  
 \uwave{disjoint} $($\uwave{point-finite},
  \uwave{discrete}, \uwave{LF}$)$ in $X$
  if $\mathcal{U}$ is  
  disjoint $($point-finite, discrete, locally finite,$)$ at each $x\in X$.
  Clearly, $\mathcal{U}$ is disjoint in  $X$ iff  
  $(\alpha \neq \beta \rightarrow  
  U_\alpha\bigcap U_\beta= \emptyset).$

(iv)  $\mathcal{U}$  is 
  \uwave{star-finite}\index{star-finite, \S\ref{sec:prelim_basic_prop}}
   $($\uwave{star-countable}\index{star-countable, \S\ref{sec:prelim_basic_prop}}$)$ if
   $\forall\beta\in\Delta, |\operatorname{ind}(
    \mathcal{U})_{U_{\beta}}| < \omega$   $ (\le\omega)$.

(v)  $\mathcal{U}$  is 
\uwave{$\sigma$-disjoint}\index{$\sigma$-disjoint, \S\ref{sec:prelim_basic_prop}}
$($\uwave{$\sigma$-locally finite}\index{$\sigma$-locally finite, \S\ref{sec:prelim_basic_prop}}, 
\uwave{$\sigma$-star finite}\index{$\sigma$-star finite, \S\ref{sec:prelim_basic_prop}}, $\cdots)$ if
$\mathcal{U} = \bigcup_{n < \omega}\mathcal{U}_n$,
where each $\mathcal{U}_n$ is disjoint $($locally finite, star-finite, $\cdots)$.

(vi)  $\{\mathcal{V}_n: n \in\omega\}$  is an 
\uwave{$\epsilon$-sequence}\index{$\epsilon$-sequence, \S\ref{sec:prelim_basic_prop}}
$($\uwave{$\theta$-sequence}\index{$\theta$-sequence, \S\ref{sec:prelim_basic_prop}}$)$
if $\forall x \in X, \exists m = m(x) < \omega$ such that 
$\mathcal{V}_m$ is disjoint $($point-finite$)$ at $x$.

(vii)  $\{\mathcal{V}_n: n \in\omega\}$  is an 
\uwave{$\epsilon^*$-sequence}\index{$\epsilon^*$-sequence, \S\ref{sec:prelim_basic_prop}}
$($\uwave{$\theta^*$-sequence}\index{$\theta^*$-sequence, \S\ref{sec:prelim_basic_prop}}$)$
if $\forall x \in X, \exists m  < \omega$ such that 
$\mathcal{V}_m$ is discrete $($locally finite$)$ at $x$.
} %end of defintion

{\lemma{}
	A family $\mathcal{D}$ of subsets of $X$ is discrete and closed iff $\bigcup \mathcal{D}$ is closed in $X$ and $\mathcal{D}$ is discrete at each $x \in \bigcup \mathcal{D}$. \qedsymbol
} %end of lemma

{\lemma{}
	Let $\mathcal{U} = \{U_\alpha : \alpha \in \Delta\}$ be a cover of $X$. 
	
	 $($i$)$ If $\mathcal{U}$ has a locally finite open partial refinement $\mathcal{V}$ such that $\overline{\mathcal{V}}$ refines $\mathcal{U}$, then $\mathcal{U}$ has a locally finite 
	 open partial refinement $\mathcal{H} = \{H_\alpha : \alpha \in \Delta\}$ such that
	  $\overline{H}_\alpha \subset U_\alpha$ for all $\alpha \in \Delta$, and 
	  $\bigcup \mathcal{H} = \bigcup \mathcal{V}$.  
	  
	 %\normalfont{
	  This result is also true if we replace ``locally finite" by ``discrete" or replace ``open" by ``closed".
	%}
	
	  $($ii$)$ If $\mathcal{U}$ has a partial cushioned $(\text{open})$ refinement $\mathcal{V}$, then $\mathcal{U}$ has a partial cushioned $\text{(open)}$ refinement $\mathcal{H} = \{H_\alpha : \alpha \in \Delta\}$ such that for any $\Gamma \subset \Delta$, $\overline{\bigcup_{\alpha \in \Gamma} H_\alpha} \subset \bigcup_{\alpha \in \Gamma} U_\alpha$ and $\bigcup \mathcal{H} = \bigcup \mathcal{V}$.  
} % end of lemma

{Proof of (i).} Suppose $\mathcal{U} = \{U_\alpha : \alpha \in \Delta\}$ has a locally finite open partial refinement $\mathcal{V}$ (such that 
$\overline{\mathcal{V}}$ refines $\mathcal{U}$). For every $V\in\mathcal{V}$, pick 
a $\delta(V)\in\Delta$ such that $V\subset U_{\delta(V)}$. For $\alpha \in \Delta$, let
$H_\alpha = \bigcup\{V\in\mathcal{V}: \delta(V) = \alpha\}$.   
Then $\mathcal{H} = \{H_\aleph : \alpha \in \Delta\}$ is a desired 
locally finite open partial refinement of $\mathcal{U}$.
\qedsymbol

{\definition{\textup{(Tukey[1940]\cite{Tukey})}}
	Let $\mathcal{U}$ and $\mathcal{V}$ be covers of $X$.
	
	\textup{(i)} $\mathcal{V}$ is a 
	\uwave{star-refinement}\index{star-refinement, \S\ref{sec:prelim_basic_prop}}
	of $\mathcal{U}$ if 
	$\{\operatorname{st}(V, \mathcal{V}) : V\in \mathcal{V}\}$ refines $\mathcal{U}$.
$\mathcal{V}$ is a 
\uwave{point-star refinement}\index{point- star refinement, \S\ref{sec:prelim_basic_prop}}
of $\mathcal{U}$ if 
$\{\operatorname{st}(x, \mathcal{V}) : x\in X\}$ refines $\mathcal{U}$.
	
	\textup{(ii)} An open cover $\mathcal{U}$ of $X$ is 
		\uwave{normal}\index{normal, \S\ref{sec:prelim_basic_prop}} 
	if there is a sequence $\{ \mathcal{U}_n: n\in\omega\}$ of open covers such that $\mathcal{U}_0$ refines $\mathcal{U}$ and $\mathcal{U}_{n+1}$ star-refines $\mathcal{U}_n$ for each $n<\omega$.
}%end of def

{\lemma{}
\textup{(i)} Let $\mathcal{U}, \mathcal{V}, \mathcal{W}$ be covers of $X$. If $\mathcal{W}$ point-star refines $\mathcal{V}$ and $\mathcal{V}$ point-star refines $\mathcal{U}$, then $\mathcal{W}$ star-refines $\mathcal{U}$.

\textup{(ii)}  An open cover $\mathcal{U}$ is normal iff there is a sequence $\{\mathcal{U}_n: n\in\omega\}$ of open covers such that $\mathcal{U}_0$ refines $\mathcal{U}$ and  $\mathcal{U}_{n+1}$ point-star refines $\mathcal{U}_n$ for each $n<\omega$..

\textup{(iii)}  If $\mathcal{U}$ is a normal cover of $X$, then there is a normal cover $\mathcal{U}_1$ of $X$ such that $\overline{\mathcal{U}_1}$ refines $\mathcal{U}$.
} %end of lemma

Proof (i) Suppose $\mathcal{W}$ point-star refines $\mathcal{V}$ and $\mathcal{V}$ point-star-refines $\mathcal{U}$. To show
that $\mathcal{W}$ star-refines $\mathcal{U}$, let $W \in \mathcal{W}$ and pick $x \in W$.
Then $\operatorname{st}(x, \mathcal{V}) \subset U$ for some $U \in \mathcal{U}$. 
For each $x \in W$ there is $V_x \in \mathcal{V}$ such that 
$\operatorname{st}(x, \mathcal{W}) \subset V_x$. Then 
$x \in W \subset \operatorname(x, \mathcal{W}) \subset V_x$, and 
$V_x \subset \operatorname{st}(x, \mathcal{V}) \subset U$. It follows that:
$$\operatorname{st}(W, \mathcal{W}) = \bigcup_{x \in W} 
\operatorname{st}(x, \mathcal{W}) \subset \bigcup_{x \in W} V_x \subset U$$  
Hence $\mathcal{W}$ star-refines $\mathcal{U}$.  

(ii) follows from (i). 

 (iii) Suppose $\mathcal{U}$ is a normal cover of $X$ and $\{\mathcal{U}_n, n\in\omega\}$ 
 is a sequence of open covers such that $\mathcal{U}_0$ refines $\mathcal{U}$ and each $\mathcal{U}_{n+1}$ star-refines $\mathcal{U}_n$ for each $n <\omega$. For each $U \in \mathcal{U}_1$, there are 
 $U_0 \in \mathcal{U}_0$ and $U \in \mathcal{U}$ such that 
 $\operatorname{st}(U_1, \mathcal{U}_1) \subset U_0 \subset U$. 
 Since $\overline{U_1} \subset \operatorname{st}(U_1, \mathcal{U}_1)$,
 $\overline{\mathcal{U}_1}$ refines $\mathcal{U}$. 
 Clearly, $\mathcal{U}_1$ is a normal cover of $X$.  \qedsymbol

{\definition{}
(i) $X$ is \uwave{$\lambda$-utraparacompact}
 \index{$\lambda$-utraparacompact, 
 	\S\ref{sec:prelim_basic_prop}} 
 	 if every open $\lambda$-cover of $X$ has a discrete open refinement.

(ii) $X$ is \uwave{$\lambda$-strongly paracompact}
 \index{$\lambda$-strongly paracompact, 
 \S\ref{sec:prelim_basic_prop}} 
 	 if every open $\lambda$-cover of $X$ has a star-finite open refinement.
 
(iii) $X$ is \uwave{$\lambda$-fully normal}
\index{$\lambda$-fully normal, 
	\S\ref{sec:prelim_basic_prop}} 
if every open $\lambda$-cover of $X$ is a normal cover.
 
(iv) $X$ is \uwave{$\lambda$-paracompact}
\index{$\lambda$-paracompact, 
	\S\ref{sec:prelim_basic_prop}} 
if every open $\lambda$-cover of $X$ has a locally finite refinement.

(v) $X$ is \uwave{$\lambda$-metacompact}
\index{$\lambda$-metacompact, 
	\S\ref{sec:prelim_basic_prop}} 
if every open $\lambda$-cover of $X$ has a point finite refinement.

(vi) $X$ is \uwave{$\lambda$-subcompact}
\index{$\lambda$-subcompact, 
	\S\ref{sec:prelim_basic_prop}}
