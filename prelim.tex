\chapter{Preliminaries}

\section{Basic	Covering	Properties}  
\label{sec:prelim_basic_prop}

We use the  usual terms and notations\cite{Alexandroff1924}. 
An \uwave{ordinal}\index{ordinal, \S\ref{sec:prelim_basic_prop}} 
is the set of all smaller ordinals, 
and a \uwave{cardinal}\index{cardinal, \S\ref{sec:prelim_basic_prop}} 
 is an initial ordinal. 
We use $\kappa, \mu$ for cardinals; $\lambda$
a for infinite cardinal;  $\alpha, \beta, \gamma, \theta$ for ordinal numbers. $\omega = \{0, 1, 2, \dots\}$ is the smallest infinite cardinal. 
$|A|$ denotes the \uwave{cardinality}\index{cardinal, \S\ref{sec:prelim_basic_prop}} 
 of a set $A$.
The \uwave{empty set}\index{empty set, \S\ref{sec:prelim_basic_prop}}  $\phi$ is the unique set $B$ such that $|B| = 0$.
$\mathbb{R}$ denotes the  
\uwave{real line}\index{real line, \S\ref{sec:prelim_basic_prop}}
$[0, 1]$ the   
\uwave{closed unit interval}\index{closed unit interval, \S\ref{sec:prelim_basic_prop}}  in $\mathbb{R}$.

For any sets $Z, Y$, we denote 
$\mathcal{P}(Z) = \{A : A \subset{Z}\}$; 
$Y^X= \{f: f$  is a function from $X$  into $Y$\}. 
If $f \in Y^X$, we write $f: X \to Y$.

If $f: X \to Y$, $A \subset X$ and $B \subset Y$, then $f(A): A \to Y$
 is defined by $f|_A(x)) = f(x)$ for each $x \in A$; 
 $f[A] = \{f(x): x \in A\}$. $f^{-1}B = f^{-1}[B] = \{x \in X: f(x) \in B\}$.

For a set $A$ and $n < \omega$,
let $[A]^{n} = \{ B \subset A: | B| = n \}$,
$[A]^{< \omega} = \cup \{ [A]^{n} : n < \omega \}$. 
Let $ \omega^0 = \{ \phi \}$. For  $n = 1, \dots, n$, let
$\omega^n = \{ f : f ; n \to \omega \} = \{ ( k_{0}, k_{1}, \dots, k_{n-1} ) :  k_i \in\omega, i < n\} $.
If $h < \omega$, let $\phi\oplus h = (h)$, 
$(k_0, \cdots, k_{n-1})\oplus h = (k_0, \cdots, k_{n-1}, h)$.	
Let $\omega^{<\omega} = \cup_{n<\omega}\omega^{n}$.
	
Let  $\mathcal{U} = \{ \mathcal{U}_\alpha\in \Delta\} $ be any family of subsets of  $X$. Without loss of generality we may assume the indexing set  $ \Delta $ be any set of ordinals. We call
	
	
	cardinality $|a|$ the power of $\mathcal{U}$, A \underline{$\lambda$-family} is a family with power $\le \lambda$. $\mathcal{U}$ is a \underline{cover} of $X$ if $\bigcup \mathcal{U} = X$. A \underline{$\lambda$-cover} is a cover with power $\le \lambda$.

\vspace{0.5em}
Let $\mathcal{U} = \{U_\alpha : \alpha \in \Delta\}$, $A \subset X$, $x \in X$. We denote

\begin{equation*}
	\mathcal{U} / A = \{U_\alpha \cap A : \alpha \in \Delta\}.
\end{equation*}

\begin{equation*}
	(\mathcal{U})_A = \{U_\alpha : \alpha \in \Delta, \ U_\alpha \cap A \neq \emptyset\}; \quad (\mathcal{U})_x = \{U_\alpha : \alpha \in \Delta, \ x \in U_\alpha\},
\end{equation*}

\begin{equation*}
	st(A, \mathcal{U}) = \bigcup (\mathcal{U})_A, \quad st(x, \mathcal{U}) = \bigcup (\mathcal{U})_x,
\end{equation*}

\begin{equation*}
	ind(\mathcal{U})_A = \{\alpha \in \Delta : U_\alpha \cap A \neq emptyset\}; quad ind(\mathcal{U})_x = \{\alpha \in \Delta : x \in U_\alpha\}
\end{equation*}

\vspace{0.5em}
By a \underline{space} we mean a topological space with no separation axioms $T_i$, $i=0,1,2$ at all.

\vspace{0.5em}
Throughout the following, $X$ denotes a space. For $A \subset X$, $\overline{A}$ denotes the \underline{closure} of $A$ and $A^\circ$ the \underline{interior} of $A$. For $x \in X$, let
\begin{equation*}
	top fn(x) = \{G : G \text{ is an open subset of } X \text{ with } x \in G\}.
\end{equation*}
