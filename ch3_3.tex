\section{Expandable spaces} \label{sec:3.3}


{\definition{ \label{def:3.3.1}}
\textup{(i)} $X$ is 
\uwave{$\lambda$-expandable}\index{$\lambda$-expandable| \S\ref{def:3.3.1}\hfill} 
\textup{([Krajewski 1971]\textsuperscript{\cite{Krajewski}})} $($resp. 
\uwave{$\lambda$-discretely expandable}$)$ 
\index{$\lambda$-discretely expandable| \S\ref{def:3.3.1}\hfill}
if every locally finite  $($resp. discrete$)$ $\lambda$-family 
of $($closed$)$ subsets of $X$ has a locally finite open expansion.

\textup{(ii)}  $X$ is 
\uwave{$\lambda$-almost expandable}\index{$\lambda$-almost expandable| \S\ref{def:3.3.1}\hfill} 
\textup{([Smith and Krajewski 1971]\textsuperscript{\cite{SmithKrajewski}})} $($resp. 
\uwave{$\lambda$-discretely almost expandable}$)$ 
\index{$\lambda$-discretely almost expandable| \S\ref{def:3.3.1}\hfill}
if every locally finite  $($resp. discrete$)$ $\lambda$-family 
of $($closed$)$ subsets of $X$ has a point-finite open expansion.

\textup{(iii)}  $X$ is 
\uwave{$\lambda$-$\theta^*$-expandable}
\index{$\lambda$-$\theta^*$-expandable| \S\ref{def:3.3.1}\hfill} 
\textup{([Smith 1976]\textsuperscript{\cite{Smith1976}})} 
$($Resp. 
\uwave{$\lambda$-discretely $\theta^*$-expandable}$)$ 
\index{$\lambda$-discretely $\theta^*$-expandable| \S\ref{def:3.3.1}\hfill} 
if every locally finite $($resp. discrete$)$ 
$\lambda$-family of $($closed$)$ subsets of $X$ has a $\theta^*$-sequence of 
open expansions.

\textup{(iv)}  $X$ is 
\uwave{$\lambda$-$\theta$-expandable}
\index{$\lambda$-$\theta$-expandable| \S\ref{def:3.3.1}\hfill} 
\textup{([Katuta 1975]\textsuperscript{\cite{Katuta1975}})} 
$($Resp. 
\uwave{$\lambda$-discretely $\theta$-expandable}$)$ 
\index{$\lambda$-discretely $\theta$-expandable| \S\ref{def:3.3.1}\hfill} 
if every locally finite $($resp. discrete$)$ 
$\lambda$-family of $($closed$)$ subsets of $X$ has a $\theta$-sequence of 
open expansions.

\textup{(v)}  $X$ is 
\uwave{$\lambda$-discretely subexpandable}
\index{$\lambda$-discretely subexpandable| \S\ref{def:3.3.1}\hfill} 
\textup{([Katuta 1975]\textsuperscript{\cite{Katuta1975}})} 
if every  discrete
$\lambda$-family of closed subsets of $X$ has a $\varepsilon$-sequence of 
open expansions.
}

{\theorem{\label{th:3.3.2}}    For any $X$, the following are equivalent:

\textup{(i)} $X$ is countably paracompact.

\textup{(ii)} Every countable increasing open cover of $X$ has a $\sigma$-cushioned open refinement.

\textup{(iii)} Every countable increasing open cover of $X$ has an open shrink.

\textup{(iv)} $X$ is countably expandable.
}


\begin{proof}
(i) $\Rightarrow$ (ii). Let $\mathscr{U} = \{U_n: n < \omega\}$ be an increasing 
open cover of a countably paracompact $X$. 
Let $\mathscr{V}$ be a locally finite  open refinement of $\mathscr{U}$. 
For $n < \omega$, let
$$G_n = \{x \in X: \exists \, W \in \operatorname{top}(x), \:
\operatorname{st}(W, \mathscr{V}) \subset U_n\}.$$

\noindent Then $\mathscr{G} = \{G_n: n < \omega\}$ is a cushioned 
open refinement of $\mathscr{U}$.

(ii) $\Rightarrow$ (iii). Let $\mathscr{U} = \{U_n: n < \omega\}$ be 
an increasing open cover of $X$. By (ii), 
let $\mathscr{V} = \bigcup_{k < \omega} \mathscr{V}_k$ be an 
open refinement of $\mathscr{U}$, where each 
$\mathscr{V}_k = \{V_{k,n}: n < \omega\}$ such that for each 
$\Gamma \subset \omega$, 
$\overline{\bigcup_{n \in \Gamma} V_{k,n}} \subset \bigcup_{n \in \Gamma} U_n, \forall \, k < \omega$. For $n < \omega$, let $W_n = \bigcup_{i=0}^n \bigcup_{j=0}^n V_{i,j}$. 
Then $\{W_n: n < \omega\}$ is an open shrink of $\mathscr{U}$.

(iii) $\Rightarrow$ (iv). Let $\mathscr{F} = \{F_n: n < \omega\}$ be a locally finite closed family. For $n < \omega$, let
$U_n = X \setminus \bigcup_{k > n} F_k$.
Then $\mathscr{U} = \{U_n: n < \omega\}$ is an increasing open cover of $X$. 
By (iii), there exists an open cover $\{H_n: n < \omega\}$ of $X$ such that 
$\overline{H}_n \subset U_n$. Let $G_0 = X$, 
$G_n = X\setminus \bigcup_{k<n}\overline{H_k}$ for $n\ge 1$.
Then $\{G_n: n < \omega\}$ is a locally finite  open expansion of $\mathscr{F}$. 
Thus $X$ is countably expandable.

(iv) $\Rightarrow$ (i). Let $\mathscr{U} = \{U_n: n < \omega\}$ be an open cover 
of a countably expandable space $X$. Let $B_0 = U_0$, 
$B_n = U_n \setminus \bigcup_{k < n} U_k$ for $n \ge 1$.
 Then $\mathscr{B} = \{B_n: n < \omega\}$ is a locally finite cover of $X$ with 
 $B_n \subset U_n$. Suppose $\mathscr{V}=\{\mathscr{V}_n: n < \omega\}$ 
 is a locally finite open expansion of $\mathscr{B}$ with 
 $B_n \subset V_n \subset U_n$ for each $n$. 
 Then $\mathscr{V}$ is a locally finite open refinement of $\mathscr{U}$, 
 so $X$ is countably paracompact.
\end{proof}

{\theorem{\label{th:3.3.3}}
For any space $X$, the following are equivalent:

\textup{(i)}  $X$ is countably metacompact.

\textup{(ii)}  Every countable increasing open cover of $X$ has a cushioned refinement.

\textup{(iii)}  Every countable increasing open cover of $X$ has a closed shrink.

\textup{(iv)} $X$ is countably almost expandable.
}

\begin{proof}
Use an analogous proof as Theorem \ref{th:3.3.2}.
\end{proof}

{\theorem{\label{th:3.3.4}} 
\textup{([Smith 1976]\textsuperscript{\cite{Smith1976}})}
$X$ is $\lambda$-expandable if and only if $X$ is 
$\lambda$-$\theta^*$-expandable.
}

\begin{proof} $(\Rightarrow)$ Clear.

$(\Leftarrow).$ 
Suppose $X$ is $\lambda$-$\theta^*$-expandable and 
$\mathscr{F} = \{F_\alpha:  \alpha < \lambda\}$ is a locally finite family of 
closed subsets of $X$. Let $\{\mathscr{G}_n: n < \omega\}$ be a 
$\theta^*$-sequence where each 
$\mathscr{G}_n = \{G_{n,\alpha}: \alpha < \lambda\}$ such that 
$F_\alpha \subset G_{n,\alpha}$ and $G_{n+1,\alpha} \subset G_{n,\alpha}$ 
for each $n, \alpha$.
For $n < \omega$, let
$$U_n = \{x \in X: \exists \, H \in \operatorname{top}(x) \textit{ such that } \vert{}\text{ind}(\mathscr{G}_n)_H\vert{} < \omega\}$$
\noindent Then $\{U_n: n < \omega\}$ is an increasing open cover of $X$. 
By Theorem \ref{th:4.1.9}, $X$ is countably paracompact. 
There exists a locally finite open cover $\{V_n: n < \omega\}$ of $X$ 
such that $\overline{V_n} \subset U_n$. 
For $\alpha < \lambda$, let $G_\alpha = \bigcup_{n < \omega} (V_n \cap G_{n,\alpha})$.
Then $\mathscr{G} = \{G_\alpha: \alpha < \lambda\}$ is an open expansion of $\mathscr{F}$. To show that $\mathscr{G}$ is locally finite, let $x \in X$.
Then $C = \{n < \omega: x \in \overline{V_n}\}$ is finite. 
Let $R = X \setminus \bigcup_{n \in\omega\setminus C} \overline{V_n}$. 
Then $R \in \operatorname{top}(x)$. For each $n \in C$, 
$x\in\overline{V_n} \subset U_n$, so there exists $H_n \in \operatorname{top}(x)$ 
such that $\vert{}\operatorname{ind}(\mathscr{G}_n)_{H_n}\vert{} < \omega$. Then
$H = R \bigcap(\bigcap_{n \in C} H_n )\in \operatorname{top}(x)$.
It is easy to check that 
$$\operatorname{ind}(\mathscr{G})_H \subset \bigcup_{n \in C} \operatorname{ind}(\mathscr{G}_n)_{H_n}.$$ 
\noindent Thus $\operatorname{ind}(\mathscr{G})_H$ is finite, 
so $\mathscr{G}$ is a locally finite open expansion of $\mathscr{F}$.
\end{proof}

 {\lemma{\label{lem:3.3.5}}
If $\mathscr{U} = \{U_\alpha:  \alpha \in \Delta\}$ is an open cover of $X$ and let 
$\tilde{U}_\alpha = U_\alpha \setminus \bigcup_{\gamma\in\Delta\setminus\{\alpha\}} U_\gamma$ 
for each $\alpha \in \Delta$, then $\tilde{\mathscr{U}} = \{\tilde{U}_\alpha:  \alpha \in \Delta\}$ 
is a discrete closed partial refinement of $\mathscr{U}$. \hfill\qedsymbol
}

 {\lemma{\label{lem:3.3.6}}
If $X$ is $\lambda$-discretely subexpandable, then for every open 
$\lambda$-cover $\mathscr{U}$ of $X$ there exists a $\sigma$-discrete closed partial refinement 
$\mathscr{P}$ and a $G_\delta$-set $A$ in $X$ such that 
$\bigcup \tilde{\mathscr{U}} \subset A \subset \bigcup \mathscr{P}$,
where $\tilde{\mathscr{U}}$ is defined as in the previous Lemma \ref{lem:3.3.5} .
}

\begin{proof} Suppose $\mathscr{U} = \{U_\alpha:  \alpha < \lambda\}$ be an 
open cover of a $\lambda$-discretely subexpandable space $X$. 
By the previous Lemma \ref{lem:3.3.5}, $\tilde{\mathscr{U}} = \{\tilde{U}_\alpha:  \alpha < \lambda\}$ 
is a discrete closed family with $\tilde{U}_\alpha \subset U_\alpha$. 
There exists an $\varepsilon$-sequence 
$\{\mathscr{G}_n = \{G_{\alpha,n}\}: n < \omega\}$ 
of open expansions of $\tilde{\mathscr{U}}$ such that 
$\tilde{U}_\alpha \subset G_{\alpha,n} \subset U_\alpha$ for each $\alpha, n$.
For each $n < \omega$, $\{\bigcup \tilde{\mathscr{U}}, X \setminus \bigcup \mathscr{G}_n\}$ is a discrete closed family in $X$. There exists a $\varepsilon$-sequence 
$\{\mathscr{L}_k = \{Q_{n,k}, W_{n,k}\}: k < \omega\}$ of open expansions such that 

$$\bigcup \tilde{\mathscr{U}} \subset Q_{n,k} \textit{ and }
X \setminus \bigcup \mathscr{G}_n \subset W_{n,k}.$$

\noindent For each $x \in X$, there exists $h < \omega$ such that $x$ is contained in 
at most one member of $\{Q_{n, h}, W_{n, h}\}$. 
Let $Q_n = \bigcap_{k < \omega} Q_{n,k}$, $W_n = \bigcap_{k < \omega} W_{n,k}$. We have:

\begin{gather}
\bigcup \tilde{\mathscr{U}} \subset Q_n, \;
X \setminus \bigcup \mathscr{G}_n \subset W_n \; \textit{ and } \;
Q_n \cap W_n = \emptyset \;
 \textit{ for } n < \omega.        \label{eq3.3.6.1}
\end{gather}

\noindent For $n, k < \omega$, let 
$$\mathscr{G}_{n,k} = \mathscr{G}_n \cup \{W_{n,k} \cap U_\alpha:  \alpha < \lambda\}.$$ 

\noindent Then each $\mathscr{G}_{n,k}$ is an open refinement of $\mathscr{U}$. 
By Lemma \ref{lem:3.3.5}, $\mathscr{P} = \bigcup_{n,k < \omega} \tilde{\mathscr{G}}_{n,k}$ is a $\sigma$-discrete closed partial refinement of $\mathscr{U}$. 
Let $A = \bigcap_{n < \omega} Q_n$. Then $A$ is a $G_\delta$-set and by (1), 
$\bigcup \tilde{\mathscr{U}} \subset A$.

To show that $A\subset \bigcup\mathscr{P}$, let $x \in A$. 
There exists $m < \omega$ such that $|\operatorname{ind}(\mathscr{G}_m)_x| \le 1$.
Since $x \in A \subset Q_m$, by \eqref{eq3.3.6.1}, $x \notin W_{m, h}$ for some $h < \omega$. 
Since $x \in \bigcup\mathscr{G}_{m,h}$, there exists $\alpha_0 < \lambda, x\in G_{\alpha_0, m}$, so
$\operatorname{ind}(\mathscr{G}_m)_x\subset \{\alpha_0\}$. It follows that

$$x \in G_{\alpha_0, m} \setminus \bigcup_{\gamma \ne \alpha_0} G_{\gamma, m} \setminus \bigcup_{\alpha < \lambda} (W_{m,k} \cap U_\alpha) = \tilde{G}_{\alpha_0, m} \in \tilde{\mathscr{G}}_{m,k} \subset \mathscr{P}.$$

\noindent Thus $A \subset \bigcup \mathscr{P}$.
\end{proof} 


{\theorem{\label{th:3.3.7}}  \textup{([Chaber 1979]\textsuperscript{\cite{Chaber}})}
In a $\lambda$-discretely subexpandable space, every point-finite open $\lambda$-cover has a $\sigma$-discrete closed refinement.
}

\begin{proof}
Suppose $\mathscr{U} = \{U_\alpha:  \alpha < \lambda\}$ be a point-finite open cover of a $\lambda$-discretely subexpandable space $X$. For $n < \omega$, let $F_0 = \emptyset$ and
$$F_n = \{x \in X:  \vert{}\operatorname{ind}(\mathscr{U})_x\vert{} \le n\}, \textit{ for } \ge 1$$
\noindent Then $\{F_n: n < \omega\}$ is a closed cover of $X$. We have the following:

\textsc{Claim.} For each $n < \omega$, there exists a $\sigma$-discrete closed partial refinement $\mathscr{P}_n$ and a $G_\delta$-set $A_n \subset X$ such that

$$F_n \subset A_n \subset \bigcup \mathscr{P}_n.$$

\noindent \textsc{Proof of Claim}. We use induction on $n < \omega$. Let $A_0 = \emptyset$ and $\mathscr{P}_0 = \{\emptyset\}$. Assume that $A_n$ and $\mathscr{P}_n$ have been defined. To define $A_{n+1}$ and $\mathscr{P}_{n+1}$, let $A_n = \bigcap_{k < \omega} G_k$, where each $G_k$ is open. 
For $k < \omega$, let
$$\mathscr{G}_k = \{ \cap_{\alpha\in S}U_\alpha: S\in[\lambda]^{n+1} \}
\bigcup \{Q_k\cap U_\alpha: \alpha < \lambda\}.$$

\noindent For each $x \in X$, if $x \in F_n$, then $x \in A_n \subset Q_k$, 
so $x \in Q_k\cap U_\alpha$ for some $\alpha < \omega$; if $x \notin F_n$, 
then $|\text{ind}(\mathscr{U})_x| > n$,
so $x\in \cap_{\alpha\in S} U_\alpha$ for some 
$S\in [\lambda]^{n+1}$. Hence
$\mathscr{G}_k$ is an open refinement of $\mathscr{U}$ with  
$|\mathscr{G}_k| \le \lambda$.
By Lemma \ref{lem:3.3.6}, we have:

(1) For each $k < \omega$, there exists a $\sigma$-discrete closed partial refinement $\mathscr{E}_k$ of $\mathscr{V}_k$ and a $G_\delta$-set $B_k$ such that $\bigcup \tilde{\mathscr{V}}_k \subset B_k \subset \bigcup \mathscr{E}_k$.

Let $A_{n+1} = \bigcap_{k < \omega} (B_k \cup G_k)$ and $\mathscr{P}_{n+1} = \mathscr{P}_n \cup \left( \bigcup_{k < \omega} \mathscr{E}_k \right)$. Then $A_{n+1}$ is a $G_\delta$-set and $\mathscr{P}_{n+1}$ is a $\sigma$-discrete closed partial refinement of $\mathscr{U}$. Let us see that:

(2) For each $k < \omega$, $F_{n+1} \setminus G_k \subset \bigcup \tilde{\mathscr{V}}_k$.

*Proof of (2).* Suppose $x \in F_{n+1} \setminus G_k$. Then $\vert{}\text{index}(\mathscr{U})_x\vert{} \le n+1$. Since $x \in F_{n+1}$ and $x \notin G_k$, there exists $S_0 \subset \lambda$ with $\vert{}S_0\vert{} = n+1$ such that $x \in \bigcap_{\alpha \in S_0} U_\alpha$. It follows that

$$x \in \bigcap_{\alpha \in S_0} U_\alpha \setminus \bigcup \left\{ \bigcap_{\alpha \in S} U_\alpha : S \neq S_0 \right\} \setminus \bigcup_{\alpha \in \lambda} (G_k \cap U_\alpha) = \tilde{\bigcap_{\alpha \in S_0} U_\alpha} \in \tilde{\mathscr{V}}_k[cite: 7]$$

So (2) holds.

By (2), for each $k < \omega$,

$$F_{n+1} \subset (F_{n+1} \setminus G_k) \cup G_k \subset \bigcup \tilde{\mathscr{V}}_k \cup G_k \subset B_k \cup G_k[cite: 7]$$

so $F_{n+1} \subset A_{n+1}$. To show that $A_{n+1} \subset \bigcup \mathscr{P}_{n+1}$, let $x \in A_{n+1}$. If $x \in A_n$, then $x \in \bigcup \mathscr{P}_n \subset \bigcup \mathscr{P}_{n+1}$. Assume $x \notin A_n$, then $x \notin G_k$ for some $k < \omega$. Since $x \in A_{n+1} \subset B_k \cup G_k$, $x \in B_k \subset \bigcup \mathscr{E}_k \subset \bigcup \mathscr{P}_{n+1}$. Hence $A_{n+1} \subset \bigcup \mathscr{P}_{n+1}$. This completes the proof of the induction.

Since $\bigcup_{n < \omega} F_n = X$, by the claim, the family $\mathscr{P} = \bigcup_{n < \omega} \mathscr{P}_n$ is a $\sigma$-discrete closed refinement of $\mathscr{U}$. 
\end{proof}
