\section{Expandable spaces} \label{sec:3-3}


{\definition{} \label{def:3-3-1}
\textup{(i)} $X$ is 
\uwave{$\lambda$-expandable}\index{$\lambda$-expandable| \S\ref{def:3-3-1}\hfill} 
\textup{(Krajewski [1971]\textsuperscript{\cite{Krajewski}})} $($resp. 
\uwave{$\lambda$-discretely expandable}
\index{$\lambda$-discretely expandable| \S\ref{def:3-3-1}\hfill}$)$ 
if every locally finite  $($resp. discrete$)$ $\lambda$-family 
of $($closed$)$ subsets of $X$ has a locally finite open expansion.

\textup{(ii)}  $X$ is 
\uwave{$\lambda$-almost expandable}\index{$\lambda$-almost expandable| \S\ref{def:3-3-1}\hfill} 
\textup{(Smith and Krajewski [1971]\textsuperscript{\cite{SmithKrajewski}})} $($resp. 
\uwave{$\lambda$-discretely almost expandable}$)$ 
\index{$\lambda$-discretely almost expandable| \S\ref{def:3-3-1}\hfill}
if every locally finite  $($resp. discrete$)$ $\lambda$-family 
of $($closed$)$ subsets of $X$ has a point-finite open expansion.

\textup{(iii)}  $X$ is 
\uwave{$\lambda$-$\theta^*$-expandable}
\index{$\lambda$-$\theta^*$-expandable| \S\ref{def:3-3-1}\hfill} 
\textup{(Smith [1976]\textsuperscript{\cite{Smith1976}})} 
$($Resp. 
\uwave{$\lambda$-discretely $\theta^*$-expandable}$)$ 
\index{$\lambda$-discretely $\theta^*$-expandable| \S\ref{def:3-3-1}\hfill} 
if every locally finite $($resp. discrete$)$ 
$\lambda$-family of $($closed$)$ subsets of $X$ has a $\theta^*$-sequence of 
open expansions.

\textup{(iv)}  $X$ is 
\uwave{$\lambda$-$\theta$-expandable}
\index{$\lambda$-$\theta$-expandable| \S\ref{def:3-3-1}\hfill} 
\textup{(Katuta [1975]\textsuperscript{\cite{Katuta1975}})} 
$($Resp. 
\uwave{$\lambda$-discretely $\theta$-expandable}$)$ 
\index{$\lambda$-discretely $\theta$-expandable| \S\ref{def:3-3-1}\hfill} 
if every locally finite $($resp. discrete$)$ 
$\lambda$-family of $($closed$)$ subsets of $X$ has a $\theta$-sequence of 
open expansions.

\textup{(v)}  $X$ is 
\uwave{$\lambda$-discretely subexpandable}
\index{$\lambda$-discretely subexpandable| \S\ref{def:3-3-1}\hfill} 
\textup{(Katuta [1975]\textsuperscript{\cite{Katuta1975}})} 
if every  discrete
$\lambda$-family of closed subsets of $X$ has a $\epsilon$-sequence of 
open expansions.
}

{\theorem{} \label{th:3-3-2} 
For any $X$, the following are equivalent:

\textup{(i)} $X$ is countably paracompact.

\textup{(ii)} Every countable increasing open cover of $X$ has a $\sigma$-cushioned open refinement.

\textup{(iii)} Every countable increasing open cover of $X$ has an open shrink.

\textup{(iv)} $X$ is countably expandable.
}


\begin{proof}
(i) $\Rightarrow$ (ii). Let $\mathscr{U} = \{U_n: n < \omega\}$ be an increasing 
open cover of a countably paracompact $X$. 
Let $\mathscr{V}$ be a locally finite  open refinement of $\mathscr{U}$. 
For $n < \omega$, let
$$G_n = \{x \in X: \exists \, W \in \operatorname{top}(x), \:
\operatorname{st}(W, \mathscr{V}) \subset U_n\}.$$

\noindent Then $\mathscr{G} = \{G_n: n < \omega\}$ is a cushioned 
open refinement of $\mathscr{U}$.

(ii) $\Rightarrow$ (iii). Let $\mathscr{U} = \{U_n: n < \omega\}$ be 
an increasing open cover of $X$. By (ii), 
let $\mathscr{V} = \bigcup_{k < \omega} \mathscr{V}_k$ be an 
open refinement of $\mathscr{U}$, where each 
$\mathscr{V}_k = \{V_{k,n}: n < \omega\}$ such that for each 
$\Gamma \subset \omega$, 
$\bigcup_{n \in \Gamma} V_{k,n} \subset \bigcup_{n \in \Gamma} U_n, \forall k < \omega$. For $n < \omega$, let $W_n = \bigcup_{i=0}^n \bigcup_{j=0}^n V_{i,j}$. 
Then $\{W_n: n < \omega\}$ is an open shrink of $\mathscr{U}$.

(iii) $\Rightarrow$ (iv). Let $\mathscr{F} = \{F_n: n < \omega\}$ be a locally finite closed family. For $n < \omega$, let
$U_n = X \setminus \bigcup_{k > n} F_k$.
Then $\mathscr{U} = \{U_n: n < \omega\}$ is an increasing open cover of $X$. 
By (iii), there exists an open cover $\{H_n: n < \omega\}$ of $X$ such that 
$\overline{H}_n \subset U_n$. Let $G_0 = X$, 
$G_n = X\setminus \bigcup_{k<n}\bar{H}_k$ for $n\ge 1$.
Then $\{G_n: n < \omega\}$ is a locally finite  open expansion of $\mathscr{F}$. 
Thus $X$ is countably expandable.

(iv) $\Rightarrow$ (i). Let $\mathscr{U} = \{U_n: n < \omega\}$ be an open cover 
of a countably expandable space $X$. Let $B_0 = U_0$, 
$B_n = U_n \setminus \bigcup_{k < n} U_k$ for $n \ge 1$.
 Then $\mathscr{B} = \{B_n: n < \omega\}$ is a locally finite cover of $X$ with 
 $B_n \subset U_n$. Suppose $\mathscr{V}=\{\mathscr{V}_n: n < \omega\}$ 
 is a locally finite open expansion of $\mathscr{B}$ with 
 $B_n \subset V_n \subset U_n$ for each $n$. 
 Then $\mathscr{V}$ is a locally finite open refinement of $\mathscr{U}$, 
 so $X$ is countably paracompact.
\end{proof}

3.3.3. Theorem

For any space $X$, the following are equivalent:

(i) $X$ is countably metacompact.

(ii) Every countable increasing open cover of $X$ has a cushioned refinement.

(iii) Every countable increasing open cover of $X$ has a closed shrinking.

(iv) $X$ is countably almost expandable.

Proof. Use an analogous proof as Theorem 3.3.2.

 3.3.4. Theorem (Smith, 1978)

$X$ is expandable if and only if $X$ is $\aleph_0$-expandable and countably paracompact.

Proof. $(\Rightarrow)$ Clear.

%$(\Leftarrow)$ Suppose $X$ is $\aleph_0$-expandable and $\mathcal{F} = \{F_\alpha \mid \alpha < \lambda\}$ is a LF family of closed subsets of $X$. Let $\{\mathcal{B}_n\}_{n < \omega}$ be a $\sigma$-sequence where each $\mathcal{B}_n = \{G_{n,\alpha} \mid \alpha < \lambda\}$ such that $F_\alpha \subset G_{n,\alpha}$ and $G_{n+1,\alpha} \subset G_{n,\alpha}$ for each $n, \alpha$.

For $n < \omega$, let

$$U_n = \{x \in X \mid \exists H \in \text{Top}(x) \text{ s.t. } \vert{}\text{index}(\mathcal{B}_n)_H\vert{} < \omega\}$$

Then $\{U_n\}_{n < \omega}$ is an increasing open cover of $X$. By Theorem 4.1.9, $X$ is countably paracompact. There is a LF open cover $\{V_n\}_{n < \omega}$ of $X$ such that $\overline{V_n} \subset U_n$. For $\alpha < \lambda$, let $W_\alpha = \bigcup_{n < \omega} (V_n \cap G_{n,\alpha})$.

Then $\mathcal{W} = \{W_\alpha \mid \alpha < \lambda\}$ is an open expansion of $\mathcal{F}$. To show that $\mathcal{W}$ is LF, let $x \in X$.

Then $C = \{n < \omega \mid x \in \overline{V_n}\}$ is finite. Let $R = X \setminus \bigcup_{n \notin C} \overline{V_n}$. Then $R \in \text{Top}(x)$. For each $n \in C$, $\overline{V_n} \subset U_n$, so there is $H_n \in \text{Top}(x)$ such that $\vert{}\text{index}(\mathcal{B}_n)_{H_n}\vert{} < \omega$. Then

$$H = R \cap \bigcap_{n \in C} H_n \in \text{Top}(x)$$

It is easy to check that $\text{index}(\mathcal{W})_H \subset \bigcup_{n \in C} \text{index}(\mathcal{B}_n)_{H_n}$. Thus $\vert{}\text{index}(\mathcal{W})_H\vert{}$ is finite, so $\mathcal{W}$ is a LF open expansion of $\mathcal{F}$.




 3.3.5. Lemma

If $\mathcal{V} = \{V_\alpha \mid \alpha \in \Delta\}$ is an open cover of $X$ and let $\tilde{V}_\alpha = V_\alpha \setminus \bigcup_{\gamma \ne \alpha} V_\gamma$ for each $\alpha \in \Delta$, then $\tilde{\mathcal{V}} = \{\tilde{V}_\alpha \mid \alpha \in \Delta\}$ is a discrete closed partial refinement of $\mathcal{V}$.

3.3.6. Lemma

If $X$ is $\kappa$-discretely subexpandable, then for every open $\kappa$-cover of $X$ there is a $\sigma$-discrete closed partial refinement $\mathcal{P}$ and a $G_\delta$-set $A$ in $X$ such that $\bigcup \tilde{\mathcal{V}} \subset A \subset \bigcup \mathcal{P}$.

Proof. Suppose $\mathcal{V} = \{V_\alpha \mid \alpha < \lambda\}$ be an open cover of a $\kappa$-discretely subexpandable space $X$. By Lemma 4.2.7, $\tilde{\mathcal{V}} = \{\tilde{V}_\alpha \mid \alpha < \lambda\}$ is a discrete closed family with $\tilde{V}_\alpha \subset V_\alpha$. There is a $\sigma$-sequence $\{\mathcal{G}_n = \{G_{\alpha,n}\}\}_{n < \omega}$ of open expansions of $\tilde{\mathcal{V}}$ such that $\tilde{V}_\alpha \subset G_{\alpha,n} \subset V_\alpha$ for each $\alpha, n$.

For each $n < \omega$, $\{\bigcup \tilde{\mathcal{V}}, X \setminus \bigcup \mathcal{G}_n\}$ is a discrete closed family in $X$. There is a $\sigma$-sequence $\{\mathcal{Q}_k = \{Q_{n,k}, W_{n,k}\}\}_{k < \omega}$ of open expansions such that $\bigcup \tilde{\mathcal{V}} \subset Q_{n,k}$ and $X \setminus \bigcup \mathcal{G}_n \subset W_{n,k}$.

For each $x \in X$, there is $k < \omega$ such that $x$ is contained in at most one member of $\{Q_{n,k}, W_{n,k}\}$. Let $Q_n = \bigcup_{k < \omega} Q_{n,k}$, $W_n = \bigcup_{k < \omega} W_{n,k}$. We have:

(1) For each $n < \omega$, $\bigcup \tilde{\mathcal{V}} \subset Q_n$, $X \setminus \bigcup \mathcal{G}_n \subset W_n$, and $Q_n \cap W_n = \emptyset$.

For $n, k < \omega$, let $\mathcal{G}_{n,k} = \mathcal{G}_n \cup \{W_{n,k} \cap V_\alpha \mid \alpha < \lambda\}$. Then each $\mathcal{G}_{n,k}$ is an open refinement of $\mathcal{V}$. By Lemma 4.2.7, $\mathcal{P} = \bigcup_{n,k < \omega} \tilde{\mathcal{B}}_{n,k}$ is a $\sigma$-discrete closed partial refinement of $\mathcal{V}$. Let $A = \bigcap_{n < \omega} Q_n$. Then $A$ is a $G_\delta$-set and by (1), $\bigcup \tilde{\mathcal{V}} \subset A$.

 3.3.7. Theorem (Chaber, 1979)

In a $\kappa$-discretely subexpandable space, every point-finite open $\kappa$-cover has a $\sigma$-discrete closed refinement.

Proof. Suppose $\mathcal{U} = \{U_\alpha \mid \alpha < \lambda\}$ be a point-finite open cover of a $\kappa$-discretely subexpandable space $X$. For $n < \omega$, let $F_0 = \emptyset$ and

$$F_n = \{x \in X \mid \vert{}\text{index}(\mathcal{U})_x\vert{} \le n\}, \quad n \ge 1[cite: 6]$$

Then $\{F_n\}_{n < \omega}$ is a closed cover of $X$. We have the following:

Claim. For each $n < \omega$, there is a $\sigma$-discrete closed partial refinement $\mathcal{P}_n$ and a $G_\delta$-set $A_n \subset X$ such that

By Lemma 4.2.7, we have:

(1) For each $k < \omega$, there is a $\sigma$-discrete closed partial refinement $\mathcal{E}_k$ of $\mathcal{V}_k$ and a $G_\delta$-set $B_k$ such that $\bigcup \tilde{\mathcal{V}}_k \subset B_k \subset \bigcup \mathcal{E}_k$.

Let $A_{n+1} = \bigcap_{k < \omega} (B_k \cup G_k)$ and $\mathcal{P}_{n+1} = \mathcal{P}_n \cup \left( \bigcup_{k < \omega} \mathcal{E}_k \right)$. Then $A_{n+1}$ is a $G_\delta$-set and $\mathcal{P}_{n+1}$ is a $\sigma$-discrete closed partial refinement of $\mathcal{U}$. Let us see that:

(2) For each $k < \omega$, $F_{n+1} \setminus G_k \subset \bigcup \tilde{\mathcal{V}}_k$.

*Proof of (2).* Suppose $x \in F_{n+1} \setminus G_k$. Then $\vert{}\text{index}(\mathcal{U})_x\vert{} \le n+1$. Since $x \in F_{n+1}$ and $x \notin G_k$, there is $S_0 \subset \lambda$ with $\vert{}S_0\vert{} = n+1$ such that $x \in \bigcap_{\alpha \in S_0} U_\alpha$. It follows that

$$x \in \bigcap_{\alpha \in S_0} U_\alpha \setminus \bigcup \left\{ \bigcap_{\alpha \in S} U_\alpha : S \neq S_0 \right\} \setminus \bigcup_{\alpha \in \lambda} (G_k \cap U_\alpha) = \tilde{\bigcap_{\alpha \in S_0} U_\alpha} \in \tilde{\mathcal{V}}_k[cite: 7]$$

So (2) holds.

By (2), for each $k < \omega$,

$$F_{n+1} \subset (F_{n+1} \setminus G_k) \cup G_k \subset \bigcup \tilde{\mathcal{V}}_k \cup G_k \subset B_k \cup G_k[cite: 7]$$

so $F_{n+1} \subset A_{n+1}$. To show that $A_{n+1} \subset \bigcup \mathcal{P}_{n+1}$, let $x \in A_{n+1}$. If $x \in A_n$, then $x \in \bigcup \mathcal{P}_n \subset \bigcup \mathcal{P}_{n+1}$. Hence $A_{n+1} \subset \bigcup \mathcal{P}_{n+1}$. This completes the proof of the induction.
