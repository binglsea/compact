\section{Expandable spaces} \label{sec:3-3}


{\definition{} \label{def:3-3-1}
\textup{(i)} $X$ is 
\uwave{$\lambda$-expandable}\index{$\lambda$-expandable| \S\ref{def:3-3-1}\hfill} 
\textup{(Krajewski [1971]\textsuperscript{\cite{Krajewski}})} $($resp. 
\uwave{$\lambda$-discretely expandable}
\index{$\lambda$-discretely expandable| \S\ref{def:3-3-1}\hfill}$)$ 
if every locally finite  $($resp. discrete$)$ $\lambda$-family 
of $($closed$)$ subsets of $X$ has a locally finite open expansion.

\textup{(ii)}  $X$ is 
\uwave{$\lambda$-almost expandable}\index{$\lambda$-almost expandable| \S\ref{def:3-3-1}\hfill} 
\textup{(Smith and Krajewski [1971]\textsuperscript{\cite{SmithKrajewski}})} $($resp. 
\uwave{$\lambda$-discretely almost expandable}$)$ 
\index{$\lambda$-discretely almost expandable| \S\ref{def:3-3-1}\hfill}
if every locally finite  $($resp. discrete$)$ $\lambda$-family 
of $($closed$)$ subsets of $X$ has a point-finite open expansion.

\textup{(iii)}  $X$ is 
\uwave{$\lambda$-$\theta^*$-expandable}
\index{$\lambda$-$\theta^*$-expandable| \S\ref{def:3-3-1}\hfill} 
\textup{(Smith [1976]\textsuperscript{\cite{Smith1976}})} 
$($Resp. 
\uwave{$\lambda$-discretely $\theta^*$-expandable}$)$ 
\index{$\lambda$-discretely $\theta^*$-expandable| \S\ref{def:3-3-1}\hfill} 
if every locally finite $($resp. discrete$)$ 
$\lambda$-family of $($closed$)$ subsets of $X$ has a $\theta^*$-sequence of 
open expansions.

\textup{(iv)}  $X$ is 
\uwave{$\lambda$-$\theta$-expandable}
\index{$\lambda$-$\theta$-expandable| \S\ref{def:3-3-1}\hfill} 
\textup{(Katuta [1975]\textsuperscript{\cite{Katuta1975}})} 
$($Resp. 
\uwave{$\lambda$-discretely $\theta$-expandable}$)$ 
\index{$\lambda$-discretely $\theta$-expandable| \S\ref{def:3-3-1}\hfill} 
if every locally finite $($resp. discrete$)$ 
$\lambda$-family of $($closed$)$ subsets of $X$ has a $\theta$-sequence of 
open expansions.

\textup{(v)}  $X$ is 
\uwave{$\lambda$-discretely subexpandable}
\index{$\lambda$-discretely subexpandable| \S\ref{def:3-3-1}\hfill} 
\textup{(Katuta [1975]\textsuperscript{\cite{Katuta1975}})} 
if every  discrete
$\lambda$-family of closed subsets of $X$ has a $\epsilon$-sequence of 
open expansions.
}

{\theorem{} \label{th:3-3-2} 
For any $X$, the following are equivalent:

\textup{(i)} $X$ is countably paracompact.

\textup{(ii)} Every countable increasing open cover of $X$ has a $\sigma$-cushioned open refinement.

\textup{(iii)} Every countable increasing open cover of $X$ has an open shrink.

\textup{(iv)}$X$ is countably expandable.
}


Proof

(i) $\Rightarrow$ (ii). Let $\mathscr{U} = \{U_n \mid n < \omega\}$ be an increasing open cover of a countably paracompact $X$. Let $\mathscr{V}$ be a locally finite (LF) open refinement of $\mathscr{U}$. For $n < \omega$, let

$$G_n = \{x \in X \mid \exists V \in \text{Top}(x) \text{ s.t. } \dots\}$$


Then $\{G_n \mid n < \omega\}$ is a cushioned open refinement of $\mathscr{U}$.

(ii) $\Rightarrow$ (iii). Let $\mathscr{U} = \{U_n \mid n < \omega\}$ be an increasing open cover of $X$. By (ii), let $\mathscr{V} = \bigcup_{k < \omega} \mathscr{V}_k$ be an open refinement of $\mathscr{U}$, where each $\mathscr{V}_k = \{V_{k,n} \mid n < \omega\}$ such that for each $N \subset \omega$, $\bigcup_{n \in N} V_{k,n} \subset \bigcup_{n \in N} U_n$ ($k < \omega$). For $n < \omega$, let $W_n = \bigcap_{i=0}^n \bigcap_{j=0}^n V_{i,j}$. Then $\{W_n \mid n < \omega\}$ is an open shrinking of $\mathscr{U}$.

(iii) $\Rightarrow$ (iv). Let $\mathscr{F} = \{F_n \mid n < \omega\}$ be a locally finite (LF) closed family. For $n < \omega$, let

$$U_n = X \setminus \bigcup_{k \ge n} F_k$$


Then $\mathscr{U} = \{U_n \mid n < \omega\}$ is an increasing open cover of $X$. By (iii), there is an open cover $\{H_n \mid n < \omega\}$ of $X$ such that $\overline{H}_n \subset U_n$. Let $G_0 = X$, $G_n = \dots$ is a LF open expansion of $\mathscr{F}$. Thus $X$ is countably expandable.

(iv) $\Rightarrow$ (i). Let $\mathscr{U} = \{U_n \mid n < \omega\}$ be an open cover of a countably expandable space $X$. Let $B_0 = U_0$, $B_n = U_n \setminus \bigcup_{k < n} U_k$ ($n \ge 1$). Then $\mathscr{B} = \{B_n \mid n < \omega\}$ is a LF cover of $X$ with $B_n \subset U_n$. Suppose $\{G_n \mid n < \omega\}$ is a LF open expansion of $\mathscr{B}$ with $B_n \subset G_n \subset U_n$ for each $n$. Then $\mathscr{V}$ is a LF open refinement of $\mathscr{U}$, so $X$ is countably paracompact.
