\section{Fully Normal Spaces} \label{sec:8.2}


{\theorem{\label{th:8.2.1}}
For any space $X$, the following are equivalent:

\textup{(i)} $X$ is  $\lambda$-fully normal.

\textup{(ii)} $X$ is $\lambda$-shrinkable and $\lambda$-screenable.

\textup{(iii)} $X$ is countably shrinkable and every open $\lambda$-cover of $X$ has a $\sigma$-star-finite open refinement.
}

\begin{proof}
(i)  $\Rightarrow$  (ii). By Corollary \ref{cor:2.1.8}.

(ii) $\Rightarrow$ (iii). Clear. 

(iii) $\Rightarrow$ (i). Assume (iii). Then $X$ is normal. 
By Theorem \ref{th:2.2.3}, let us show that $X$ is $\lambda$-paracompact. 
Suppose $\mathscr{U}$ is an open $\lambda$-cover of $X$. By (iii), 
$\mathscr{U}$ has an open refinement 
$\mathscr{V} = \bigcup_{n < \omega} \mathscr{V}_n$, 
where each $\mathscr{V}_n = \{V_{n,\alpha}: \alpha \in \Delta_n\}$ 
is star-finite.
Since $X$ is countably shrinkable, there exists an
open cover $\{S_n: n\in\omega\}$ such that 
$\overline{S_n}\subset\bigcup\mathscr{V}_n$
for each \(n < \omega\).
By Corollary \ref{cor:5.2.5}, 
\(X\) is countably paracompact. There exists
a locally finite open cover \(\mathscr{W}=\{W_n: n<\omega\}\) of \(X\) such that 
\(W_n\subset S_n\)
for each \(n<\omega\). Then \(\overline{W_n}\subset\bigcup\mathscr{V}_n\). 
We have

\textbf{(1)} \(\mathscr{G}_n=\{W_n\cap V_{n,\alpha}: \alpha\in\Delta_n\}\)
 is locally finite and open
for each \(n < \omega\).

\noindent\textsc{Proof}.
Let \(n<\omega\), \(x\in X\). If \(x\notin\bigcup{\mathscr{V}}_n\), 
then \(X\setminus\overline{W_n}\) is a
neighborhood of \(x\) which 
%has nonempty intersection with no member
is disjoint with any member
of \(\mathscr{G}_n\). If \(x\in\mathscr{V}_{n,\beta}\) for some \(\beta\in\Delta_n\),
then \(\{\alpha\in\Delta_n: V_{n,\beta}\cap V_{n,\alpha}\neq\emptyset\}\) is finite. Then the set
\(\{\alpha\in\Delta_n: V_{n,\beta}\cap (W_n\cap V_{n,\alpha})\neq\emptyset\}\) is finite, 
so \(\mathscr{G}_n\) is locally finite and (1) holds.

Let \(\mathscr{H}=\{H_n: n<\omega\}\) be a locally finite open cover of \(X\) such that
\(H_n\subset\bigcup \mathscr{G}_n\) for each \(n<\omega\). 
Let \(\mathscr{R}=\{H_n\cap W_n\cap V_{n,\alpha}: \alpha\in\Delta_n, n < \omega\}\).

\textbf{(2)} \(\mathscr{R}\) is a locally finite open refinement of \(\mathscr{U}\). Thus, 
\(X\) is $\lambda$-paracompact.

\noindent\textsc{Proof}. It is clear that \(\mathscr{R}\) is an open refinement of \(\mathscr{U}\). To show
that \(\mathscr{R}\) is locally finite, let \(x\in X\). Since \(\mathscr{H}\) is locally finite, 
there exists \(G_0\in\operatorname{top}(x)\) such
that \(D=\{n<\omega: G_0\cap H_n \ne \emptyset\}\) is finite. By (1), 
for each \(n\in D\)
there exists \(G_n\in\operatorname{top}(x)\) such that 
\(C_n=\{\alpha\in\Delta_n: G_n\cap(W_n\cap V_{n,\alpha})\neq\emptyset\}\)
is finite. Let \(G=G_0\cap(\bigcap_{n\in D}G_n)\) and let 

$$C=\{(n,\alpha): n<\omega,\alpha\in\Delta_n, G\cap H_n \cap W_n\cap V_{n,\alpha} \neq\emptyset\}.$$

\noindent Then $C\subset \bigcup_{n\in D}(\{n\}\times C_n)$, 
so \(C\) is finite and (2) holds.
\end{proof} 


\begin{corollary} \label{cor:8.2.2}
\textup{([Nagami 1955]\textsuperscript{\cite{Nagami}})}
Every screenable, normal, countably paracompact space is paracompact. \hfill\qedsymbol
\end{corollary}


{\theorem{\label{th:8.2.3}}
For any \(\lambda\) the following are equivalent:

\textup{(i)}  \(X\) is \(\lambda\)-fully normal.

\textup{(ii)}  \(X\) is \(\lambda\)-collectionwise normal and 
\(\lambda\)-$D\omega^2$-refinable.

\textup{(iii)}  \(X\) is normal \(\lambda\)-expandable and 
\(\lambda\)-$A\omega^2$-refinable.
}

\begin{proof}
By Theorem \ref{th:8.1.7}.
\end{proof}

{\corollary{} \label{cor:8.2.4}
For any \(\lambda\) the following are
equivalent:

\textup{(i)}  \(X\) is \(\lambda\)-fully normal.

\textup{(ii)}  \(X\) is \(\lambda\)-collectionwise normal and \(\lambda\)-subparacompact.

\textup{(iii)}  \(X\) is \(\lambda\)-collectionwise normal and \(\lambda\)-metacompact.

\textup{(iii)}  \(X\) is \(\lambda\)-collectionwise normal and 
\(\lambda\)-submetacompact.
}

\begin{proof}
 (i) \(\Rightarrow\) (ii) By Theorem \ref{th:8.2.3} and Corollary \ref{cor:2.1.8}.
 
(i) \(\Rightarrow\) (iii) By Theorem \ref{th:8.2.3}.

(ii) \(\Rightarrow\) (iv) and (iii) \(\Rightarrow\) (iv) Clear.
(iv) \(\Rightarrow\) (i) By Theorem \ref{th:8.2.3}.
\end{proof}

{\corollary{} \label{cor:8.2.5} Corollary. For any \(T_1\) collectionwise normal space \(X\),
the following are equivalent:

\textup{(i)} \(X\) is \(T_2\) paracompact.

\textup{(ii)}  \textup{([Michael 1955]\textsuperscript{\cite{Michael1955}}, 
	[Nagami 1955]\textsuperscript{\cite{Nagami}})}
\(X\) is metacompact.

\textup{(iii)}  \textup{([McAuley 1958]\textsuperscript{\cite{McAuley}})}
\(X\) is subparacompact.

\textup{(iv)}   \textup{([Worrell and Wicke 1965]]\textsuperscript{\cite{WorrellWicke}})}
([Worrell and Wicke 1965]). \(X\) is submetacompact.

\textup{(v)} \textup{([Liu 1977]\textsuperscript{\cite{Liu}})}
\(X\) is strictly quasi-paracompact.

\textup{(vi)} \textup{([Smith 1976]\textsuperscript{\cite{Smith1976}})}
\(X\) is weakly \(\bar{\theta}\)-refinable.\hfill\qedsymbol
}

{\theorem{\label{th:8.2.6}}
For any $X$, the following are equivalent:

\textup{(i)} \(X\) is \(\lambda\)-fully normal.

\textup{(ii)}  Every open \(\lambda\)-cover of \(X\) has an open star refinement.

\textup{(iii)}  Every open \(\lambda\)-cover of \(X\) has a semi-open local star refinement.

(iv) Every open \(\lambda\)-cover of \(X\) has a semi-open point-star refinement.
}

\begin{proof}
(i) \(\Rightarrow\) (ii) \(\Rightarrow\) (iii) \(\Rightarrow\) (iv). Clear.

(iv) \(\Rightarrow\) (i). Assume (iv). By Theorem \ref{th:8.1.5}, \(X\) is \(\lambda\)-collectionwise normal. By Theorem \ref{th:4.2.2}, \(X\) is 
$\lambda$-metacompact. By Corollary \ref{cor:8.2.4}, \(X\) is \(\lambda\)-fully normal. 
\end{proof}


{\theorem{\label{th:8.2.7}} 
For any$X$ the following are equivalent:

\textup{(i)} \(X\) is \(\lambda\)-fully normal.

\textup{(ii)}  Every open \(\lambda\)-cover \(\mathscr{U}\) of \(X\) has a locally finite
 and $\sigma$-discrete functionally open refinement \(\mathscr{V}\) such that \(\mathscr{\overline{V}}\) refines \(\mathscr{U}\).

\textup{(iii)}  Every open \(\lambda\)-cover of \(X\) has a closure-preserving 
closed refinement.

\textup{(iv)}  Every open \(\lambda\)-cover of \(X\) has a cushioned open refinement.

\textup{(v)} Every open \(\lambda\)-cover has a cushioned refinement.

\textup{(vi)} Every interior-preserving open \(\lambda\)-cover 
has a closure-preserving closed refinement.
}

\begin{proof} It is clear that (i) \(\Rightarrow\) (ii) \(\Rightarrow\) (iii) \(\Rightarrow\) (vi),
 (iii) \(\Rightarrow\) (v), and (ii) \(\Rightarrow\) (iv) \(\Rightarrow\) (v). 
 
 To see that (vi) \(\Rightarrow\) (i) and (v) \(\Rightarrow\) (i).
Assume (v) or (vi). By Theorem \ref{th:8.1.5}, \(X\) is \(\lambda\)-collectionwise
normal. By Theorem \ref{th:4.1.11}, \(X\) is \(\lambda\)-subparacompact. 
By Corollary \ref{cor:8.2.4}, \(X\) is \(\lambda\)-fully normal. 
\end{proof}

{\theorem{\label{th:8.2.8}}
For any $X$ the following are equivalent:

\textup{(i)}   \(X\) is \(\lambda\)-fully normal.

\textup{(ii)}   Every open \(\lambda\)-cover has a \(\sigma\)-discrete closed refinement \(\mathscr{F}\) such that \(\mathscr{F}^\circ\) covers \(X\).


\textup{(iii)}   Every open \(\lambda\)-cover has a \(\sigma\)-locally finite 
closed refinement \(\mathscr{F}\) such that \(\mathscr{F}^\circ\) covers \(X\).

\textup{(iv)}  \textup{([Jiang 1992]\textsuperscript{\cite{Jiang1992}})}
Every open \(\lambda\)-cover has a \(\sigma\)-cushioned open refinement.

\textup{(iii)}  Every interior-preserving open \(\lambda\)-cover has a
 \(\sigma\)-closure-preserving closed refinement \(\mathscr{F}\) 
 such that \(\mathscr{F}^\circ\) covers \(X\).
}

\begin{proof} (i) \(\Rightarrow\) (ii). By Theorem 8.2.7.

(ii) \(\Rightarrow\) (iii) $\Rightarrow$ (iv) 
and (ii) \(\Rightarrow\) (v). Clear.

(iv) \(\Rightarrow\) (i) By Theorem \ref{th:8.2.7} and Lemma \ref{lem:8.1.4}.

(v) \(\Rightarrow\) (i). Assume (v). By Theorem 4.1.11, \(X\) is \(\lambda\)-subparacompact. By Theorem 8.1.7, \(X\) is \(\lambda\)-collectionwise normal. By Corollary 8.2.4, \(X\) is \(\lambda\)-fully normal. 
\end{proof}


{\corollary{} \label{cor:8.2.9} 
\textup{(a)} \textup{([Michael 1953, 1957]\textsuperscript{\cite{Michael1953}}
	\textsuperscript{\cite{Michael1957}})}
For any \(T_1\) regular space \(X\), the following are equivalent:

\quad\textup{(i)} \(X\) is \(T_2\) paracompact.

\quad\textup{(ii)} Every open cover has a \(\sigma\)-discrete open refinement.

\quad\textup{(iii)} Every open cover has a \(\sigma\)-locally finite open refinement.

\quad\textup{(iv)} Every open cover has a \(\sigma\)-closure-preserving open refinement.

\textup{(b)}  \textup{([Michael 1959]\textsuperscript{\cite{Michael1959}}}
For any \(T_1\) space \(X\), the following are equivalent:

\quad\textup{(i)} \(X\) is \(T_2\) paracompact.

\quad\textup{(ii)}  Every open cover has a \(\sigma\)-cushioned open refinement.

\quad\textup{(iii)} Every open cover has a cushioned refinement.\hfill\qedsymbol
}

{\corollary{} \label{cor:8.2.10}
 \textup{([Morita 1962]\textsuperscript{\cite{Morita1962}})}
For any $X$ the following are equivalent:

\textup{(i)}  \(X\) is \(\lambda\)-fully normal.

\textup{(ii)}   Every open \(\lambda\)-cover has a locally finite closed refinement.

\textup{(iii)}   Every open \(\lambda\)-cover has a closure-preserving closed 
refinement.\hfill\qedsymbol
}


{\theorem{\label{th:8.2.11}}
For any $X$ the following are equivalent:

\textup{(i)}  \(X\) is \(\lambda\)-fully normal.

\textup{(ii)}  \textup{([Jiang 1992]\textsuperscript{\cite{Jiang1992}})} 
 \(X\) is \(\lambda\)-\(\varepsilon^*\)-refinable.

\textup{(iii)}   Every interior-preserving open \(\lambda\)-cover of \(X\) has an $\varepsilon^*$-sequence of interior-preserving  open covers of \(X\).

\textup{(iv)}  Every  interior-preserving open \(\lambda\)-cover of \(X\) has a local star refining sequence by  interior-preserving open covers.

\textup{(v)}  Every open \(\lambda\)-cover of \(X\) has a local star refining sequence by semi-open covers.
}

\begin{proof} 
(i) \(\Rightarrow\) (ii) and (iii). The proof 
is similar  to that for  Theorem \ref{th:8.1.9}.

(ii) \(\Rightarrow\) (v) and (iii) \(\Rightarrow\) (iv). Clear.

(iv) \(\Rightarrow\) (v) Assume (ii). By Theorem \ref{th:4.1.7}, $X$ is 
$\lambda$-submetacompact. The Theorem \ref{th:8.1.9}, $X$ is $\lambda$-collectionwise
normal. Then $X$ is $\lambda$-fully normal. 

(v) \(\Rightarrow\) (i) Assume (v). By Theorem \ref{th:4.1.11}, $X$ 
is $\lambda$-subparacompact. By Theorem \ref{th:8.1.9}, $X$ is
$\lambda$-collectionwise normal. Hence $X$ is $\lambda$-fully normal.
\end{proof}

 \textup{([Arhangelskii 1961]\textsuperscript{\cite{Arhangelski1961}}}
$X$ is $T_2$ paracompact  iff $X$ is $T_1$ and every open cover of $X$ has a local star refining sequence by open covers of $X$.\hfill\qedsymbol
