\section{Fully Normal Spaces} \label{sec:8.2}


{\theorem{\label{th:8.2.1}}
For any space $X$, the following are equivalent:

\textup{(i)} $X$ is  $\lambda$-fully normal.

\textup{(ii)} $X$ is $\lambda$-shrinkable and $\lambda$-screenable.

\textup{(iii)} $X$ is countably shrinkable and every open $\lambda$-cover of $X$ has a $\sigma$-star-finite open refinement.
}

\begin{proof}
(i)  $\Rightarrow$  (ii). By Corollary \ref{cor:2.1.8}.

(ii) $\Rightarrow$ (iii). Clear. 

(iii) $\Rightarrow$ (i). Assume (iii). Then $X$ is normal. 
By Theorem \ref{th:2.2.3} let us show that $X$ is paracompact. 
Suppose $\mathscr{U}$ is an open $\lambda$-cover of $X$. By (iii), 
$\mathscr{U}$ has an open refinement 
$\mathscr{V} = \bigcup_{n < \omega} \mathscr{V}_n$, 
where each $\mathscr{V}_n = \{V_{n,\alpha} \mid \alpha \in \Delta_n\}$ 
is star-finite.
Since \(\mathscr{V}\) is countably shrinkable, there exists an
open cover $\{S_n: n\in\omega\}$ such that 
$\overline{S_n}\subset\mathscr{V}_n$
for each \(n\in\omega\).
By Corollary \ref{cor:5.2.5},  % TO_DO
\(X\) is countably paracompact. There is
alocally finite open cover \(\mathscr{W}=\bigcup_{n\in\omega}\mathscr{W}_n\) of \(X\) such that \(\overline{\mathscr{W}}_n\subset\bigcup\mathscr{V}_n\)
for each \(n\in\omega\). Then \(\overline{\mathscr{W}}_n\subset\bigcup\mathscr{V}_n\). We have
\(\mathscr{G}_n=\{W_n\cap V_{n,\alpha}\mid\alpha\in\Delta_n\}\) islocally finite open
cover for each \(n\in\omega\).

Proof.
Let \(n<\omega\), \(x\in X\). If \(x\notin\overline{\mathscr{W}}_n\), then \(X\setminus\overline{\mathscr{W}}_n\) is a
neighborhood of \(x\) which has nonempty intersection with no member
of \(\mathscr{G}_n\). If \(x\in\mathscr{V}_{n,\beta}\) for some \(\beta\in\Delta_n\),
then \(\{\alpha\in\Delta_n\mid V_{n,\beta}\cap V_{n,\alpha}\neq\emptyset\}\) is finite. Then the set
\(\{\alpha\in\Delta_n\mid (V_{n,\beta}\cap W_n)\cap(V_{n,\alpha}\cap W_n)\neq\emptyset\}\) is finite, so \(\mathscr{G}_n\) islocally finite and (1)
holds.

Let \(\mathscr{R}_n=\{R_{n,\gamma}\mid\gamma\in\Gamma_n\}\) be alocally finite open cover of \(X\) such that
\(\overline{\mathscr{R}}_n\prec\mathscr{G}_n\) for each \(n\in\omega\). Let \(\mathscr{R}=\bigcup_{n\in\omega}\{R_{n,\gamma}\cap W_n\cap V_{n,\alpha}\mid\gamma\in\Gamma_n,\alpha\in\Delta_n\}\).

(2) \(\mathscr{R}\) is alocally finite open refinement of \(\mathscr{U}\). Thus, \(X\) is paracompact.

Proof. It is clear that \(\mathscr{R}\) is an open refinement of \(\mathscr{U}\). To show
that \(\mathscr{R}\) is locally finite, let \(x\in X\). Since \(\mathscr{W}\) is locally finite, there is \(D_0\in[\omega]^{<\omega}\) such
that \(D_0=\{n\in\omega\mid x\in\overline{\mathscr{W}}_n\}\) is finite. By (1), for each \(n\in D_0\)
there is \(G_n\in\mathscr{N}(x)\) such that \(C_n=\{\alpha\in\Delta_n\mid G_n\cap(W_n\cap V_{n,\alpha})\neq\emptyset\}\)
is finite. Let \(G=R_0\cap\bigcap_{n\in D_0}G_n\) and let \(C=\{(n,\gamma,\alpha)\mid n\in\omega, \gamma\in\Gamma_n, \alpha\in\Delta_n, G\cap(R_{n,\gamma}\cap W_n\cap V_{n,\alpha})\neq\emptyset\}\).
Then \(C\subset\bigcup_{n\in D_0}(\{n\}\times\Gamma_n\times C_n)\), so \(C\) is finite and (2) holds.
\end{proof} 


\begin{corollary} \label{cor:8.2.2}
\textup{([Nagami 1955]\textsuperscript{\cite{Nagami}})}
Every screenable, normal, countably paracompact space is paracompact.
\end{corollary}

(iv) (Worrell and Wicke [1965]). \(X\) is submetacompact.

(vi) (Smith [ ]). \(X\) is weakly \(\bar{\theta}\)-refinable. \(\square\)
