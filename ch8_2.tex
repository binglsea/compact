\section{Fully Normal Spaces} \label{sec:8.2}


{\theorem{\label{th:8.2.1}}
For any space $X$, the following are equivalent:

\textup{(i)} $X$ is  $\lambda$-fully normal.

\textup{(ii)} $X$ is $\lambda$-shrinkable and $\lambda$-screenable.

\textup{(iii)} $X$ is countably shrinkable and every open $\lambda$-cover of $X$ has a $\sigma$-star-finite open refinement.
}

\begin{proof}
(i)  $\Rightarrow$  (ii). By Corollary \ref{cor:2.1.8}.

(ii) $\Rightarrow$ (iii). Clear. 

(iii) $\Rightarrow$ (i). Assume (iii). Then $X$ is normal. 
By Theorem 2.2.3, let us show that $X$ is paracompact. 
Suppose $\mathcal{U}$ be an open $\kappa$-cover of $X$. By (iii), 
$\mathcal{U}$ has an open refinement 
$\mathcal{W} = \bigcup_{n < \omega} \mathcal{W}_n$, 
where each $\mathcal{W}_n = \{V_{n,\alpha} \mid \alpha \in \Delta_n\}$ 
is star-finite...
\end{proof}
